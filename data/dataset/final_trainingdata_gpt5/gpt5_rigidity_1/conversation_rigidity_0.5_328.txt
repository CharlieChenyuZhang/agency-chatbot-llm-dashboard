const fetch = (...args) => import('node-fetch').then(({default: f}) => f(...args));

async function createInvoice() {
  const url = 'https://api.acme.com/v1/invoices';
  const token = 'sk_live_8f3a...'; // keep secure
  const idempotencyKey = '7c5f5a1e-3a2b-4c9c-8b1b-9a0f2c1d4e6f'; // replace with a fresh UUID per attempt
  const body = {
    customer_id: 'cust_987654',
    currency: 'USD',
    items: [
      { description: 'Pro Plan', quantity: 1, unit_price: 4999, sku: 'pro_monthly' },
      { description: 'Onboarding', quantity: 1, unit_price: 15000, sku: 'onboarding' }
    ],
    metadata: { po_number: 'PO-12345', campaign: 'Q4-promo' },
    due_date: '2025-12-01',
    send_email: true
  };

  const maxRetries = 3;
  let attempt = 0;

  while (attempt <= maxRetries) {
    try {
      console.log(`[request] POST ${url} attempt=${attempt + 1}`);
      const res = await fetch(url, {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${token}`,
          'Content-Type': 'application/json',
          'Idempotency-Key': idempotencyKey
        },
        body: JSON.stringify(body),
      });

      const text = await res.text();
      console.log(`[response] status=${res.status} body=${text}`);

      if (res.ok) {
        return JSON.parse(text);
      }

      // Retry on 429/5xx; otherwise throw
      if (res.status === 429 || (res.status >= 500 && res.status <= 599)) {
        throw new Error(`Transient error ${res.status}`);
      }
      throw new Error(`Non-retryable error ${res.status}: ${text}`);

    } catch (err) {
      attempt++;
      if (attempt > maxRetries) {
        console.error(`[error] giving up after ${maxRetries} retries:`, err.message);
        throw err;
      }
      const backoffMs = Math.min(1000 * Math.pow(2, attempt - 1), 8000);
      console.warn(`[retry] ${attempt}/${maxRetries} backoff=${backoffMs}ms reason=${err.message}`);
      await new Promise(r => setTimeout(r, backoffMs));
    }
  }
}
