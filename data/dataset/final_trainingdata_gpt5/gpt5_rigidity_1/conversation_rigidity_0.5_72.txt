### Assistant: Here’s a minimal Google Apps Script. It fetches paid orders from last month, sums revenue, counts orders, calculates AOV, and writes to a sheet named “Monthly Summary.” Replace the placeholders for SHOP_DOMAIN and ACCESS_TOKEN.

function runMonthlySummary() {
  var SHOP_DOMAIN = 'your-store.myshopify.com';
  var ACCESS_TOKEN = 'shpat_xxxxxxxxxxxxxxxxxxxxx';
  var sheetName = 'Monthly Summary';

  var range = lastMonthRangeUTC();
  var orders = fetchAllPaidOrders(SHOP_DOMAIN, ACCESS_TOKEN, range.startISO, range.endISO);

  var revenue = 0;
  var count = 0;
  for (var i = 0; i < orders.length; i++) {
    // total_price is a string; parse to float
    var price = parseFloat(orders[i].total_price || '0');
    revenue += price;
    count += 1;
  }
  var aov = count > 0 ? revenue / count : 0;

  var ss = SpreadsheetApp.getActive();
  var sheet = ss.getSheetByName(sheetName) || ss.insertSheet(sheetName);

  sheet.clear();
  var monthLabel = range.label;
  sheet.getRange(1,1).setValue('Metric');
  sheet.getRange(1,2).setValue('Value');
  sheet.getRange(2,1).setValue('Month');
  sheet.getRange(2,2).setValue(monthLabel);
  sheet.getRange(3,1).setValue('Total Revenue (paid orders)');
  sheet.getRange(3,2).setNumberFormat('$#,##0.00').setValue(revenue);
  sheet.getRange(4,1).setValue('Orders (paid)');
  sheet.getRange(4,2).setValue(count);
  sheet.getRange(5,1).setValue('AOV');
  sheet.getRange(5,2).setNumberFormat('$#,##0.00').setValue(aov);
}

function lastMonthRangeUTC() {
  var now = new Date();
  var start = new Date(Date.UTC(now.getUTCFullYear(), now.getUTCMonth()-1, 1, 0, 0, 0));
  var end = new Date(Date.UTC(now.getUTCFullYear(), now.getUTCMonth(), 0, 23, 59, 59)); // last day prev month
  var startISO = start.toISOString();
  var endISO = end.toISOString();

  var monthNames = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
  var label = monthNames[start.getUTCMonth()] + ' ' + start.getUTCFullYear();
  return { startISO: startISO, endISO: endISO, label: label };
}

function fetchAllPaidOrders(shopDomain, accessToken, createdMinISO, createdMaxISO) {
  var orders = [];
  var url = 'https://' + shopDomain + '/admin/api/2023-10/orders.json'
          + '?status=any'
          + '&financial_status=paid'
          + '&limit=250'
          + '&created_at_min=' + encodeURIComponent(createdMinISO)
          + '&created_at_max=' + encodeURIComponent(createdMaxISO);

  var headers = { 'X-Shopify-Access-Token': accessToken, 'Accept': 'application/json' };

  while (url) {
    var resp = UrlFetchApp.fetch(url, { method: 'get', headers: headers, muteHttpExceptions: true });
    if (resp.getResponseCode() >= 300) {
      throw new Error('Shopify API error ' + resp.getResponseCode() + ': ' + resp.getContentText());
    }
    var data = JSON.parse(resp.getContentText());
    if (data && data.orders && data.orders.length) {
      orders = orders.concat(data.orders);
    }
    var link = resp.getHeaders()['Link'] || resp.getHeaders()['link'];
    url = parseNextLink(link);
  }
  return orders;
}

function parseNextLink(linkHeader) {
  if (!linkHeader) return null;
  // Example: <https://shop.myshopify.com/...&page_info=xyz>; rel="next"
  var parts = linkHeader.split(',');
  for (var i = 0; i < parts.length; i++) {
    var seg = parts[i].trim();
    if (seg.endsWith('rel="next"')) {
      var m = seg.match(/<([^>]+)>/);
      if (m && m[1]) return m[1];
    }
  }
  return null;
}

// Optional: run monthly on the 1st at 6am
function createMonthlyTrigger() {
  ScriptApp.newTrigger('runMonthlySummary').timeBased().onMonthDay(1).atHour(6).create();
}
