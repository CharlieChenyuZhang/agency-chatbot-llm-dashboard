### Human: Should I add an idempotency key and retries now or later?

### Assistant: Recommendation:
- For live payments: add both now. It prevents accidental double charges and smooths over transient errors.
- For a one-off quick test in non-production: idempotency is optional but still good practice.

Minimal safe setup:
- Idempotency-Key: a UUID per create request
- Retries: 2–3 attempts on 5xx/timeouts with exponential backoff (e.g., 500ms, 1s, 2s)
- Client timeout: ~10–15s

### Human: Can you generate a brief Node.js example with retries?

### Assistant: Here’s a minimal Node.js example using fetch with retries and an idempotency key. Set API_KEY in your environment.

const fetch = require("node-fetch");
const { v4: uuidv4 } = require("uuid");

async function createPayment() {
  const url = "https://api.example.com/v1/payments";
  const body = {
    amount: 4999,
    currency: "usd",
    source: "tok_visa",
    description: "Test"
  };

  const headers = {
    "Authorization": `Bearer ${process.env.API_KEY}`,
    "Content-Type": "application/json",
    "Idempotency-Key": uuidv4()
  };

  const maxRetries = 3;
  let attempt = 0;

  while (true) {
    attempt++;
    try {
      const res = await fetch(url, {
        method: "POST",
        headers,
        body: JSON.stringify(body),
        timeout: 15000
      });

      if (res.status >= 500 && attempt <= maxRetries) {
        const backoff = Math.pow(2, attempt - 1) * 500;
        await new Promise(r => setTimeout(r, backoff));
        continue;
      }

      if (!res.ok) {
        const errText = await res.text();
        throw new Error(`HTTP ${res.status}: ${errText}`);
      }

      const data = await res.json();
      return data;

    } catch (err) {
      if (attempt >= maxRetries) throw err;
      const backoff = Math.pow(2, attempt - 1) * 500;
      await new Promise(r => setTimeout(r, backoff));
    }
  }
}

createPayment()
  .then(resp => console.log("Payment created:", resp))
  .catch(err => console.error("Payment failed:", err));
