import os, csv, time, datetime, requests

TOKEN = os.getenv("HUBSPOT_PRIVATE_APP_TOKEN")
BASE = "https://api.hubapi.com"
HEADERS = {"Authorization": f"Bearer {TOKEN}", "Content-Type": "application/json"}

# Adjust these if your internal names differ
STATUS_PROP = "status"
LAST_ORDER_PROP = "last_order_date"
PROPS = ["firstname","lastname","email","company",LAST_ORDER_PROP,"hs_email_hard_bounced"]

def iso_days_ago(days):
    return (datetime.datetime.utcnow() - datetime.timedelta(days=days)).strftime("%Y-%m-%d")

def search_contacts(after=None):
    cutoff = iso_days_ago(365)
    body = {
        "filterGroups": [{
            "filters": [
                {"propertyName": STATUS_PROP, "operator": "EQ", "value": "Active"},
                {"propertyName": LAST_ORDER_PROP, "operator": "GTE", "value": cutoff},
                {"propertyName": "email", "operator": "HAS_PROPERTY"}
            ]
        }],
        "properties": PROPS,
        "limit": 100,
        "after": after
    }
    r = requests.post(f"{BASE}/crm/v3/objects/contacts/search", headers=HEADERS, json=body, timeout=30)
    r.raise_for_status()
    return r.json()

def to_name(props):
    first = (props.get("firstname") or "").strip()
    last = (props.get("lastname") or "").strip()
    nm = (first + " " + last).strip()
    return nm or ""

def main():
    if not TOKEN:
        raise SystemExit("Set HUBSPOT_PRIVATE_APP_TOKEN")

    seen_emails = set()
    rows = []
    after = None
    while True:
        data = search_contacts(after)
        for item in data.get("results", []):
            p = item.get("properties", {}) or {}
            email = (p.get("email") or "").strip().lower()
            if not email or email in seen_emails:
                continue
            if str(p.get("hs_email_hard_bounced", "")).lower() in ("true","1"):
                continue

            name = to_name(p)
            company = (p.get("company") or "").strip()
            last_order = (p.get(LAST_ORDER_PROP) or "").strip()

            rows.append([name, email, company, last_order])
            seen_emails.add(email)

        paging = data.get("paging", {})
        next_link = paging.get("next", {}).get("after")
        if next_link:
            after = next_link
            time.sleep(0.2)
        else:
            break

    # Write CSV in UTF-8 with BOM for Excel compatibility
    with open("customers_last12mo.csv", "w", encoding="utf-8-sig", newline="") as f:
        w = csv.writer(f)
        w.writerow(["Name","Email","Company","Last Order Date"])
        w.writerows(rows)

if __name__ == "__main__":
    main()
