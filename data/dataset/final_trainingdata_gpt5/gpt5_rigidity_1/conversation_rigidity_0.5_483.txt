PowerShell script: Exact duplicate finder (dry-run by default)
- Saves results to a timestamped CSV in WorkDir.
- Picks the newest (LastWriteTime) as canonical.
- Moves duplicates to Quarantine when -Execute is used.
- Skips common junk/system folders and your exclusions.

PowerShell:
param(
  [Parameter(Mandatory=$true)][string[]]$Roots,
  [Parameter(Mandatory=$true)][string]$WorkDir,
  [Parameter(Mandatory=$true)][string]$QuarantinePath,
  [switch]$Execute = $false,
  [int]$MinAgeDays = 0,
  [string[]]$ExcludeDirs = @(".git","node_modules","$Recycle.Bin","System Volume Information","OneDriveTemp","Sync Issues",".tmp"),
  [string[]]$IncludeExtensions = @(".pdf",".doc",".docx",".ppt",".pptx",".xls",".xlsx",".txt",".rtf",".jpg",".jpeg",".png",".tif",".tiff",".gif",".heic"),
  [string]$HashAlgorithm = "SHA256"
)

# Prep
$ErrorActionPreference = "Stop"
New-Item -ItemType Directory -Force -Path $WorkDir | Out-Null
New-Item -ItemType Directory -Force -Path $QuarantinePath | Out-Null
$ts = Get-Date -Format "yyyyMMdd_HHmmss"
$report = Join-Path $WorkDir "dedupe_report_$ts.csv"
"Action,Hash,CanonicalPath,DuplicatePath,Reason" | Out-File -Encoding UTF8 $report

function IsExcludedDir($path, $exList) {
  foreach ($ex in $exList) {
    if ($path -like "*\$ex\*" -or $(Split-Path $path -Leaf) -eq $ex) { return $true }
  }
  return $false
}

Write-Host "Scanning files..."
$cutoff = (Get-Date).AddDays(-$MinAgeDays)
$allFiles = foreach ($root in $Roots) {
  Get-ChildItem -Path $root -File -Recurse -Force -ErrorAction SilentlyContinue |
    Where-Object {
      -not (IsExcludedDir $_.FullName $ExcludeDirs) -and
      ($IncludeExtensions -contains $_.Extension.ToLowerInvariant()) -and
      $_.LastWriteTime -le $cutoff
    }
}

Write-Host ("Files considered: {0}" -f $allFiles.Count)

# Group by size first to reduce hashing work
$bySize = $allFiles | Group-Object Length | Where-Object { $_.Count -gt 1 }

$hashMap = @{}
$progress = 0
foreach ($group in $bySize) {
  $progress++
  Write-Progress -Activity "Hashing potential duplicates" -Status ("Group {0}/{1}" -f $progress, $bySize.Count) -PercentComplete (($progress/$bySize.Count)*100)
  foreach ($f in $group.Group) {
    try {
      $h = Get-FileHash -Algorithm $HashAlgorithm -LiteralPath $f.FullName
      if (-not $hashMap.ContainsKey($h.Hash)) { $hashMap[$h.Hash] = New-Object System.Collections.Generic.List[System.IO.FileInfo] }
      $hashMap[$h.Hash].Add($f)
    } catch {
      "Skip,,,$($f.FullName),HashError" | Out-File -Append -Encoding UTF8 $report
    }
  }
}

Write-Host "Selecting canonical files..."
$actions = New-Object System.Collections.Generic.List[object]
foreach ($kv in $hashMap.GetEnumerator()) {
  $files = $kv.Value
  if ($files.Count -lt 2) { continue }
  $canonical = $files | Sort-Object LastWriteTime -Descending, FullName -Ascending | Select-Object -First 1
  foreach ($dup in ($files | Where-Object { $_.FullName -ne $canonical.FullName })) {
    $actions.Add([PSCustomObject]@{
      Hash = $kv.Key
      Canonical = $canonical.FullName
      Duplicate = $dup.FullName
      Reason = "Exact duplicate (newest kept)"
    })
  }
}

Write-Host ("Exact duplicate sets found: {0} duplicates to quarantine" -f $actions.Count)

# Dry run report
foreach ($a in $actions) {
  "Quarantine,$($a.Hash),$($a.Canonical),$($a.Duplicate),$($a.Reason)" | Out-File -Append -Encoding UTF8 $report
}

if (-not $Execute) {
  Write-Host "Dry-run complete. Review CSV:"
  Write-Host $report
  Write-Host "When ready, rerun with -Execute to move duplicates to quarantine."
  return
}

# Execute: move duplicates to quarantine (preserve relative structure safely)
Write-Host "Executing moves to quarantine..."
foreach ($a in $actions) {
  try {
    $rel = $a.Duplicate.Substring((Split-Path $Roots[0] -Parent).Length) 2>$null
  } catch { $rel = [IO.Path]::GetFileName($a.Duplicate) }
  $targetDir = Join-Path $QuarantinePath (Split-Path $rel -Parent)
  New-Item -ItemType Directory -Force -Path $targetDir | Out-Null
  $targetPath = Join-Path $targetDir (Split-Path $a.Duplicate -Leaf)

  # If name collision in quarantine, append short hash
  if (Test-Path $targetPath) {
    $short = $a.Hash.Substring(0,10)
    $base = [IO.Path]::GetFileNameWithoutExtension($targetPath)
    $ext = [IO.Path]::GetExtension($targetPath)
    $targetPath = Join-Path $targetDir ($base + ".__dup_" + $short + $ext)
  }

  Move-Item -LiteralPath $a.Duplicate -Destination $targetPath -Force
}
Write-Host "Done. Quarantined duplicates:"
Write-Host $QuarantinePath
Write-Host "CSV:"
Write-Host $report

PowerShell: Year/Topic refactor (dry-run by default)
param(
  [Parameter(Mandatory=$true)][string[]]$Roots,
  [Parameter(Mandatory=$true)][string]$DestRoot,
  [Parameter(Mandatory=$true)][string]$WorkDir,
  [switch]$Execute = $false
)

$ErrorActionPreference = "Stop"
New-Item -ItemType Directory -Force -Path $WorkDir | Out-Null
$ts = Get-Date -Format "yyyyMMdd_HHmmss"
$plan = Join-Path $WorkDir "refactor_plan_$ts.csv"
"Action,OldPath,NewPath,Year,Topic,SourceOfYear" | Out-File -Encoding UTF8 $plan

# Simple topic map: keyword -> Topic
$topicMap = @{
  "tax"="Taxes"; "irs"="Taxes"; "w2"="Taxes"; "1099"="Taxes";
  "invoice"="Receipts"; "receipt"="Receipts"; "paid"="Receipts";
  "insurance"="Insurance"; "claim"="Insurance"; "policy"="Insurance";
  "medical"="Medical"; "doctor"="Medical"; "rx"="Medical";
  "travel"="Travel"; "flight"="Travel"; "itinerary"="Travel";
  "school"="School"; "class"="School"; "homework"="School";
  "work"="Work"; "project"="Work"; "meeting"="Work";
}

function GuessTopic([string]$path) {
  $name = (Split-Path $path -Leaf).ToLowerInvariant()
  $parents = (Split-Path $path -Parent).ToLowerInvariant()
  foreach ($k in $topicMap.Keys) {
    if ($name -like "*$k*" -or $parents -like "*$k*") { return $topicMap[$k] }
  }
  return "Misc"
}

function GetImageDateTaken([string]$file) {
  # Try exiftool if present
  $ex = Get-Command exiftool -ErrorAction SilentlyContinue
  if ($null -ne $ex) {
    try {
      $out = & exiftool -s -s -s -DateTimeOriginal "$file" 2>$null
      if ($out) {
        $dt = [datetime]::Parse($out)
        return ,@($dt, "EXIF:DateTimeOriginal")
      }
    } catch {}
  }
  # Fallback to LastWriteTime
  $fi = Get-Item -LiteralPath $file
  return ,@($fi.LastWriteTime, "LastWriteTime")
}

$files = foreach ($root in $Roots) {
  Get-ChildItem -Path $root -File -Recurse -Force -ErrorAction SilentlyContinue
}

foreach ($f in $files) {
  $ext = $f.Extension.ToLowerInvariant()
  $isImage = @(".jpg",".jpeg",".png",".tif",".tiff",".gif",".heic",".bmp") -contains $ext

  $year = $null; $src = $null
  if ($isImage) {
    $r = GetImageDateTaken $f.FullName
    $year = $r[0].Year
    $src = $r[1]
  } else {
    $year = $f.LastWriteTime.Year
    $src = "LastWriteTime"
  }

  $topic = GuessTopic $f.FullName
  $yearStr = "{0:D4}" -f $year

  $newDir = Join-Path $DestRoot (Join-Path $yearStr $topic)
  $newPath = Join-Path $newDir $f.Name

  "Plan,$($f.FullName),$newPath,$yearStr,$topic,$src" | Out-File -Append -Encoding UTF8 $plan
}

if (-not $Execute) {
  Write-Host "Dry-run complete. Review plan CSV:"
  Write-Host $plan
  Write-Host "Tweak the topic map as needed, then rerun with -Execute to move files."
  return
}

# Execute moves (create dirs, avoid collisions by appending counter)
Import-Csv -Path $plan | ForEach-Object {
  $old = $_.OldPath
  $new = $_.NewPath
  $dir = Split-Path $new -Parent
  New-Item -ItemType Directory -Force -Path $dir | Out-Null

  $target = $new
  $base = [IO.Path]::GetFileNameWithoutExtension($target)
  $ext = [IO.Path]::GetExtension($target)
  $i = 1
  while (Test-Path $target) {
    $target = Join-Path $dir ($base + " (" + $i + ")" + $ext)
    $i++
  }
  Move-Item -LiteralPath $old -Destination $target -Force
}

    -DestRoot "C:\Users\you\OneDrive\Library" `
