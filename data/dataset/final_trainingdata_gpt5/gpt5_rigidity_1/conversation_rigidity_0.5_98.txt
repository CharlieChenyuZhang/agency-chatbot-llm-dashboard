#!/usr/bin/env python3
import argparse, os, sys, hashlib, time, subprocess, json, csv
from datetime import datetime
from pathlib import Path
import shutil

import pandas as pd

try:
    from PIL import Image
    import imagehash
except Exception:
    Image = None
    imagehash = None

try:
    import PyPDF2
except Exception:
    PyPDF2 = None

try:
    import docx
except Exception:
    docx = None

try:
    import tlsh
except Exception:
    tlsh = None

try:
    import ssdeep
except Exception:
    ssdeep = None


ALLOWED_EXT = {'.pdf','.doc','.docx','.txt','.jpg','.jpeg','.png','.tif','.tiff','.heic'}

def shasum(path):
    h = hashlib.sha256()
    with open(path, 'rb') as f:
        for chunk in iter(lambda: f.read(1024*1024), b''):
            h.update(chunk)
    return h.hexdigest()

def run_exiftool_date(path):
    try:
        out = subprocess.check_output(['exiftool','-j','-DateTimeOriginal','-CreateDate',str(path)], stderr=subprocess.DEVNULL, timeout=10)
        data = json.loads(out.decode('utf-8', 'ignore'))[0]
        for key in ['DateTimeOriginal','CreateDate']:
            if key in data:
                return parse_exif_date(data[key])
    except Exception:
        pass
    return None

def parse_exif_date(s):
    # exiftool formats like "2020:05:31 14:22:10"
    try:
        return datetime.strptime(s.split('+')[0], '%Y:%m:%d %H:%M:%S')
    except Exception:
        return None

def pdf_metadata_date(path):
    # Try PyPDF2 info
    if PyPDF2 is None: return None
    try:
        with open(path, 'rb') as f:
            r = PyPDF2.PdfReader(f)
            info = r.metadata or {}
            # Keys like '/CreationDate', '/ModDate' may exist
            for key in ['/ModDate','/CreationDate']:
                if key in info:
                    d = parse_pdf_date(str(info[key]))
                    if d: return d
    except Exception:
        pass
    return None

def parse_pdf_date(s):
    # PDF date format: D:YYYYMMDDHHmmSSOHH'mm'
    try:
        if s.startswith('D:'):
            s = s[2:]
        # just take first 14 digits if present
        fmt = '%Y%m%d%H%M%S'
        digits = ''.join(ch for ch in s if ch.isdigit())
        if len(digits) >= 14:
            return datetime.strptime(digits[:14], fmt)
    except Exception:
        return None
    return None

def docx_dates(path):
    if docx is None: return None
    try:
        d = docx.Document(path)
        cp = d.core_properties
        for dt in [cp.modified, cp.created]:
            if isinstance(dt, datetime):
                return dt
    except Exception:
        pass
    return None

def best_date_for(path):
    suf = path.suffix.lower()
    if suf in {'.jpg','.jpeg','.png','.tif','.tiff','.heic'}:
        d = run_exiftool_date(path)
        if d: return d
    if suf == '.pdf':
        d = pdf_metadata_date(path)
        if d: return d
        # fallback to exiftool for PDFs (sometimes has CreateDate)
        d = run_exiftool_date(path)
        if d: return d
    if suf in {'.doc','.docx'}:
        d = docx_dates(path)
        if d: return d
    # fallback to mtime
    try:
        return datetime.fromtimestamp(path.stat().st_mtime)
    except Exception:
        return None

def extract_text_pdf(path):
    # prefer pdftotext for speed and accuracy
    try:
        out = subprocess.check_output(['pdftotext','-q',str(path),'-'], stderr=subprocess.DEVNULL, timeout=20)
        return out.decode('utf-8','ignore')
    except Exception:
        # fallback: PyPDF2 text (less reliable)
        if PyPDF2 is None: return ''
        try:
            text = []
            with open(path,'rb') as f:
                r = PyPDF2.PdfReader(f)
                for page in r.pages[:20]:
                    try:
                        text.append(page.extract_text() or '')
                    except Exception:
                        pass
            return '\n'.join(text)
        except Exception:
            return ''

def extract_text_docx(path):
    if docx is None: return ''
    try:
        d = docx.Document(path)
        return '\n'.join(p.text for p in d.paragraphs)
    except Exception:
        return ''

def fuzzy_signatures(text):
    text = text if isinstance(text,str) else ''
    sd = ssdeep.hash(text) if ssdeep else ''
    th = ''
    if tlsh:
        try:
            h = tlsh.hash(text.encode('utf-8','ignore'))
            th = h if h not in (None,'TNULL') else ''
        except Exception:
            pass
    return sd, th

def image_phash(path):
    if not Image or not imagehash: return ''
    try:
        with Image.open(path) as im:
            im = im.convert('RGB')
            return str(imagehash.phash(im))
    except Exception:
        return ''

def immediate_project(path, source_roots):
    # If parent equals any source root, project = Misc; else parent name
    parent = path.parent
    for root in source_roots:
        try:
            if parent.samefile(root):
                return 'Misc'
        except Exception:
            pass
    return parent.name or 'Misc'

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--sources', nargs='+', required=True)
    ap.add_argument('--staging', required=True)
    ap.add_argument('--dry-run', action='store_true')
    args = ap.parse_args()

    sources = [Path(os.path.expanduser(s)).resolve() for s in args.sources]
    staging = Path(os.path.expanduser(args.staging)).resolve()
    reports = staging / 'reports'
    output = staging / 'output'
    reports.mkdir(parents=True, exist_ok=True)
    output.mkdir(parents=True, exist_ok=True)

    records = []
    source_roots = sources

    # Collect files
    all_files = []
    for root in sources:
        for dirpath, _, filenames in os.walk(root):
            for fn in filenames:
                p = Path(dirpath) / fn
                if p.suffix.lower() in ALLOWED_EXT:
                    all_files.append(p)

    # Inventory
    for p in all_files:
        try:
            st = p.stat()
            size = st.st_size
            mtime = st.st_mtime
            sha = shasum(p)
            bd = best_date_for(p) or datetime.fromtimestamp(mtime)
            year = str(bd.year) if bd else 'Unknown'
            proj = immediate_project(p, source_roots)
            # Near-dup signals
            suf = p.suffix.lower()
            ss = tl = ph = ''
            if suf == '.pdf':
                text = extract_text_pdf(p)
                ss, tl = fuzzy_signatures(text)
            elif suf in {'.doc','.docx','.txt'}:
                text = extract_text_docx(p) if suf != '.txt' else p.read_text(errors='ignore')[:500000]
                ss, tl = fuzzy_signatures(text)
            elif suf in {'.jpg','.jpeg','.png','.tif','.tiff','.heic'}:
                ph = image_phash(p)

            # Proposed destination path (no write yet)
            dest_dir = output / year / proj
            dest_path = dest_dir / p.name

            records.append({
                'path': str(p),
                'size': size,
                'mtime': mtime,
                'mtime_iso': datetime.fromtimestamp(mtime).isoformat(),
                'sha256': sha,
                'best_date_iso': bd.isoformat() if bd else '',
                'year': year,
                'project': proj,
                'proposed_dest': str(dest_path),
                'ssdeep': ss,
                'tlsh': tl,
                'phash': ph
            })
        except Exception as e:
            print(f'WARN: failed {p}: {e}', file=sys.stderr)

    df = pd.DataFrame(records)

    # Duplicates by sha256
    dup_groups = []
    keep_rows = []
    discard_rows = []
    if not df.empty:
        for sha, g in df.groupby('sha256'):
            # newest mtime first
            g2 = g.sort_values('mtime', ascending=False)
            keeper = g2.iloc[0]
            keep_rows.append(keeper)
            for _, row in g2.iloc[1:].iterrows():
                discard_rows.append(row)
            dup_groups.append((sha, len(g2)))

    # Write reports
    inv_path = reports / 'inventory.csv'
    dup_path = reports / 'duplicates_exact.csv'
    moves_path = reports / 'proposed_moves.csv'
    near_path = reports / 'near_signals.csv'

    df.to_csv(inv_path, index=False)

    dup_rows = []
    for r in keep_rows:
        dup_rows.append({'sha256': r['sha256'], 'status':'keep', 'path': r['path'], 'mtime_iso': r['mtime_iso']})
    for r in discard_rows:
        dup_rows.append({'sha256': r['sha256'], 'status':'discard', 'path': r['path'], 'mtime_iso': r['mtime_iso']})
    pd.DataFrame(dup_rows).to_csv(dup_path, index=False)

    # Proposed moves: Only for keepers we plan to place in output; discards will be linked to their keeper for later deletion/archival
    move_rows = []
    keep_paths = set(r['path'] for r in keep_rows)
    for _, r in df.iterrows():
        action = 'keep_and_move' if r['path'] in keep_paths else 'duplicate_discard'
        move_rows.append({'src': r['path'], 'dest': r['proposed_dest'], 'action': action})
    pd.DataFrame(move_rows).to_csv(moves_path, index=False)

    pd.DataFrame(df[['path','ssdeep','tlsh','phash']]).to_csv(near_path, index=False)

    print(f'Inventory: {len(df)} files')
    print(f'Duplicate groups: {sum(1 for _ in df.groupby("sha256"))}')
    kept = len(keep_rows)
    discarded = len(discard_rows)
    print(f'Keepers: {kept}, Discards (exact dups): {discarded}')
    print(f'Wrote: {inv_path}')
    print(f'Wrote: {dup_path}')
    print(f'Wrote: {moves_path}')
    print(f'Wrote: {near_path}')
    if args.dry_run:
        print('Dry run only. No files moved.')

if __name__ == '__main__':
    main()
