How the reallocation works:
- If any category’s Available < 0, the script pulls from donor categories in ascending Priority until it’s zero or you hit the monthly cap.

Apps Script (Extensions > Apps Script > paste). At top of file, set:
- REALLOCATION_CAP = 150 (max dollars to move per calendar month)
- DONOR_PRIORITY comes from Categories!Priority
- PROTECTED = rows where Protect = "Y"

Script (plain text):
const REALLOCATION_CAP = 150; // dollars per calendar month

function reallocateButton() {
  const ss = SpreadsheetApp.getActive();
  const catSheet = ss.getSheetByName('Categories');
  const budSheet = ss.getSheetByName('Budget');

  const monthKey = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM');
  const props = PropertiesService.getDocumentProperties();
  const usedKey = 'used_cap_' + monthKey;
  let used = Number(props.getProperty(usedKey) || 0);
  let remainingCap = Math.max(0, REALLOCATION_CAP - used);

  if (remainingCap <= 0) {
    SpreadsheetApp.getUi().alert('Reallocation cap reached for ' + monthKey);
    return;
  }

  // Read Categories
  const catValues = catSheet.getDataRange().getValues().slice(1); // skip header
  const protectedSet = new Set(catValues.filter(r => String(r[3]).toUpperCase() === 'Y').map(r => String(r[0])));

  // Read Budget: Category, StartOfMonth, Assigned, Activity, Available
  const bRange = budSheet.getDataRange();
  const bValues = bRange.getValues();
  const header = bValues[0];
  const rows = bValues.slice(1);

  const colIndex = (name) => header.indexOf(name);
  const idxCat = colIndex('Category');
  const idxAvail = colIndex('Available');

  if (idxCat < 0 || idxAvail < 0) throw new Error('Budget sheet missing Category or Available columns');

  // Build maps
  const donors = [];
  const seekers = [];
  for (let i = 0; i < rows.length; i++) {
    const cat = String(rows[i][idxCat]);
    const avail = Number(rows[i][idxAvail]) || 0;
    const rowNum = i + 2; // account for header
    const catMeta = catValues.find(r => String(r[0]) === cat);
    const priority = catMeta ? Number(catMeta[2]) || 99 : 99;
    const isProtected = protectedSet.has(cat);

    if (avail > 0 && !isProtected) donors.push({cat, avail, rowNum, priority});
    if (avail < 0) seekers.push({cat, need: -avail, rowNum});
  }

  donors.sort((a, b) => a.priority - b.priority); // lower priority number = donate first

  // Reallocate
  let toWrite = {}; // rowNum -> newAvailable
  let capUsedNow = 0;

  // snapshot current availables
  const availMap = new Map(rows.map((r, i) => [i + 2, Number(r[idxAvail]) || 0]));

  for (const s of seekers) {
    let need = s.need;
    if (remainingCap <= 0) break;

    for (const d of donors) {
      if (need <= 0 || remainingCap <= 0) break;
      if (d.avail <= 0) continue;

      const transfer = Math.min(d.avail, need, remainingCap);
      d.avail -= transfer;
      need -= transfer;
      remainingCap -= transfer;
      capUsedNow += transfer;

      // reduce donor available
      const dAvail = (toWrite[d.rowNum] ?? availMap.get(d.rowNum)) - transfer;
      toWrite[d.rowNum] = dAvail;

      // increase seeker available
      const sAvail = (toWrite[s.rowNum] ?? availMap.get(s.rowNum)) + transfer;
      toWrite[s.rowNum] = sAvail;
    }
  }

  // Write back
  const availCol = idxAvail + 1;
  const updates = Object.entries(toWrite);
  updates.forEach(([rowStr, newAvail]) => {
    budSheet.getRange(Number(rowStr), availCol).setValue(newAvail);
  });

  props.setProperty(usedKey, String(used + capUsedNow));

  SpreadsheetApp.getUi().alert('Reallocated $' + capUsedNow.toFixed(2) + ' this month. Remaining cap: $' + remainingCap.toFixed(2));
}

const INCOME_PCTS = {
  'Rent': 0.40,
  'Groceries': 0.15,
  'Utilities': 0.10,
  'Transportation': 0.10,
  'Emergency Fund': 0.10,
  'Dining': 0.08,
  'Fun': 0.07
};

function incomeAssign() {
  const ui = SpreadsheetApp.getUi();
  const resp = ui.prompt('Income Assignment', 'Enter net income amount to assign:', ui.ButtonSet.OK_CANCEL);
  if (resp.getSelectedButton() !== ui.Button.OK) return;
  const amt = Number(resp.getResponseText());
  if (!amt || amt <= 0) { ui.alert('Invalid amount'); return; }

  const ss = SpreadsheetApp.getActive();
  const bud = ss.getSheetByName('Budget');
  const data = bud.getDataRange().getValues();
  const header = data[0];
  const rows = data.slice(1);

  const idxCat = header.indexOf('Category');
  const idxAssigned = header.indexOf('Assigned');
  if (idxCat < 0 || idxAssigned < 0) throw new Error('Budget sheet needs Category and Assigned');

  const pctSum = Object.values(INCOME_PCTS).reduce((a,b)=>a+b,0);
  if (Math.abs(pctSum - 1) > 0.001) {
    ui.alert('Warning: INCOME_PCTS do not sum to 100%. Proceeding anyway.');
  }

  let assignedTotal = 0;
  for (let i = 0; i < rows.length; i++) {
    const cat = String(rows[i][idxCat]);
    const pct = INCOME_PCTS[cat];
    if (!pct) continue;
    const add = Math.round(amt * pct * 100) / 100;
    rows[i][idxAssigned] = Number(rows[i][idxAssigned] || 0) + add;
    assignedTotal += add;
  }

  // Write back Assigned column
  for (let i = 0; i < rows.length; i++) {
    bud.getRange(i + 2, idxAssigned + 1).setValue(rows[i][idxAssigned]);
  }

  ui.alert('Assigned $' + assignedTotal.toFixed(2) + ' across categories.');
}
