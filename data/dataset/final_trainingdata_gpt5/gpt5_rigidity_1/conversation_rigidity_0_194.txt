[PowerShell script starts]
param(
  [string]$SourceRoot = "D:\Docs",
  [string]$DestRoot = "E:\TeamDocs",
  [string]$ManifestPath = ".\move_manifest.csv",
  [switch]$WhatIfMode = $true,
  [bool]$DeleteDuplicates = $false
)

Set-StrictMode -Version Latest
$ErrorActionPreference = "Stop"

function Get-FileHashSha256($path) {
  return (Get-FileHash -Path $path -Algorithm SHA256).Hash
}

function Guess-ClientName($path) {
  $generic = @("downloads","misc","temp","tmp","archive","old","backup")
  $dir = Split-Path -Parent $path
  while ($dir -and (Split-Path $dir -Leaf)) {
    $leaf = (Split-Path $dir -Leaf)
    if (-not ($generic -contains $leaf.ToLower())) { return $leaf }
    $dir = Split-Path -Parent $dir
  }
  return "_UnknownClient"
}

$files = Get-ChildItem -Path $SourceRoot -Recurse -File -Force
$manifest = New-Object System.Collections.Generic.List[object]
$seen = @{}  # hash -> canonical path

foreach ($f in $files) {
  try {
    $hash = Get-FileHashSha256 $f.FullName
    $year = $f.LastWriteTime.Year
    $client = Guess-ClientName $f.FullName
    $destDir = Join-Path (Join-Path $DestRoot $year) $client
    $destPath = Join-Path $destDir $f.Name

    if (-not (Test-Path $destDir)) {
      if ($WhatIfMode) {
        "Would create dir: $destDir" | Out-Host
      } else {
        New-Item -ItemType Directory -Force -Path $destDir | Out-Null
      }
    }

    if ($seen.ContainsKey($hash)) {
      $canonical = $seen[$hash]
      $manifest.Add([pscustomobject]@{
        Action = $(if ($DeleteDuplicates) {"DeleteDupe"} else {"HardlinkDupe"})
        From   = $f.FullName
        To     = $canonical
        Dest   = $destPath
        Hash   = $hash
      })

      if ($WhatIfMode) {
        if ($DeleteDuplicates) {
          "Would delete duplicate: $($f.FullName) (same as $canonical)" | Out-Host
        } else {
          "Would replace $($f.FullName) with hardlink to $canonical" | Out-Host
        }
      } else {
        if ($DeleteDuplicates) {
          Remove-Item -LiteralPath $f.FullName -Force
        } else {
          # Replace dupe with hardlink pointing to canonical
          $tmp = "$($f.FullName).tmp_link"
          if (Test-Path $tmp) { Remove-Item $tmp -Force }
          New-Item -ItemType HardLink -Path $tmp -Target $canonical | Out-Null
          Remove-Item -LiteralPath $f.FullName -Force
          Rename-Item -Path $tmp -NewName (Split-Path $f.FullName -Leaf)
        }
      }
      continue
    }

    # First time we see this content: move to destination and set as canonical
    $seen[$hash] = $destPath
    $manifest.Add([pscustomobject]@{
      Action = "MoveCanonical"
      From   = $f.FullName
      To     = $destPath
      Dest   = $destPath
      Hash   = $hash
    })

    if ($WhatIfMode) {
      "Would move: $($f.FullName) -> $destPath" | Out-Host
    } else {
      if (Test-Path $destPath) {
        # If a file already exists at destination with same hash, treat as dupe of canonical
        $existingHash = (Get-FileHash -Path $destPath -Algorithm SHA256).Hash
        if ($existingHash -eq $hash) {
          if ($DeleteDuplicates) {
            Remove-Item -LiteralPath $f.FullName -Force
          } else {
            $tmp = "$($f.FullName).tmp_link"
            if (Test-Path $tmp) { Remove-Item $tmp -Force }
            New-Item -ItemType HardLink -Path $tmp -Target $destPath | Out-Null
            Remove-Item -LiteralPath $f.FullName -Force
            Rename-Item -Path $tmp -NewName (Split-Path $f.FullName -Leaf)
          }
        } else {
          # Name collision, different content: rename with hash suffix
          $ext = [IO.Path]::GetExtension($destPath)
          $base = [IO.Path]::GetFileNameWithoutExtension($destPath)
          $destPath = Join-Path (Split-Path $destPath -Parent) ($base + "_" + $hash.Substring(0,8) + $ext)
          Move-Item -LiteralPath $f.FullName -Destination $destPath -Force
          $seen[$hash] = $destPath
        }
      } else {
        Move-Item -LiteralPath $f.FullName -Destination $destPath -Force
      }
    }
  } catch {
    Write-Warning "Skipped $($f.FullName): $($_.Exception.Message)"
  }
}

# Export manifest
$manifest | Export-Csv -NoTypeInformation -Path $ManifestPath -Force

if ($WhatIfMode) {
  "Dry run complete. Review $ManifestPath, then rerun with -WhatIfMode:$false when satisfied." | Out-Host
} else {
  "Applied changes. Manifest written to $ManifestPath." | Out-Host
}
