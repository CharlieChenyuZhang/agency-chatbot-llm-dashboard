Script contents:
#!/usr/bin/env bash
set -euo pipefail
shopt -s nullglob

DIR="$HOME/Desktop/Screenshots"
PROJECT_RAW="Onboarding"
DRY_RUN=1           # 1 = preview, 0 = apply
RESET_PER_DAY=0     # 1 = numbering resets each date, 0 = continuous
ON_COLLISION="suffix"  # suffix | fail | skip

# Sanitize project name for filesystem safety
PROJECT=$(printf '%s' "$PROJECT_RAW" | sed -E 's#[/:*?"<>|]#-#g; s/ +/ /g; s/^-+|-+$//g')

# Build (epoch<TAB>path) list
lines=()
for f in "$DIR"/*.png; do
  b=$(stat -f "%B" "$f" 2>/dev/null || echo 0)
  if [ "${b:-0}" -le 0 ]; then b=$(stat -f "%m" "$f"); fi
  printf -v line '%s\t%s' "$b" "$f"
  lines+=("$line")
done

IFS=$'\n' sorted=($(printf '%s\n' "${lines[@]}" | LC_ALL=C sort -n -k1,1 -k2,2))
seq=0
prev_date=""

for line in "${sorted[@]}"; do
  epoch=${line%%$'\t'*}
  file=${line#*$'\t'}
  date_str=$(date -r "$epoch" +%Y-%m-%d)

  if [ "$RESET_PER_DAY" -eq 1 ] && [ "$date_str" != "$prev_date" ]; then
    seq=0
    prev_date="$date_str"
  fi
  ((seq++))

  ext="${file##*.}"
  base="$date_str - $PROJECT - $(printf '%03d' "$seq").$ext"
  target="$DIR/$base"

  if [ -e "$target" ] && [ "$file" != "$target" ]; then
    case "$ON_COLLISION" in
      fail) echo "Collision: $target exists. Aborting."; exit 1 ;;
      skip) echo "Skipping (exists): $file -> $target"; continue ;;
      suffix)
        n=1
        while [ -e "$target" ] && [ "$file" != "$target" ]; do
          ((n++))
          target="$DIR/$date_str - $PROJECT - $(printf '%03d' "$seq") ($n).$ext"
        done
        ;;
    esac
  fi

  if [ "$DRY_RUN" -eq 1 ]; then
    printf 'Would rename: %s -> %s\n' "$file" "$target"
  else
    mv -n -- "$file" "$target"
    printf 'Renamed: %s -> %s\n' "$file" "$target"
  fi
done
