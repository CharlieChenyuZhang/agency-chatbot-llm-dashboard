// charge.js
import crypto from 'node:crypto';

const API_KEY = process.env.API_KEY;
if (!API_KEY) {
  console.error('Set API_KEY env var first');
  process.exit(1);
}

// Example input payload as your system might produce
const input = {
  userId: '123',
  amount: 5.0,           // dollars as float
  currency: 'usd',
  metadata: { note: 'Order #A10042', source: 'web' }
};

function toMinorUnits(amount) {
  // Avoid float rounding by using strings when possible
  const asStr = String(amount);
  const [whole, frac = ''] = asStr.split('.');
  const fracPadded = (frac + '00').slice(0, 2);
  return parseInt(whole, 10) * 100 + parseInt(fracPadded || '0', 10);
}

async function sleep(ms) { return new Promise(r => setTimeout(r, ms)); }

async function createCharge(payload) {
  const url = 'https://api.example.com/v1/charge';
  const idempotencyKey = crypto.randomUUID();

  const outgoing = {
    userId: payload.userId,
    amount: Number.isInteger(payload.amount) ? payload.amount : toMinorUnits(payload.amount),
    currency: String(payload.currency).toUpperCase(),
    metadata: payload.metadata
  };

  const body = JSON.stringify(outgoing);

  let attempt = 0;
  const maxAttempts = 4;

  while (attempt < maxAttempts) {
    attempt++;
    const res = await fetch(url, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${API_KEY}`,
        'Content-Type': 'application/json',
        'Idempotency-Key': idempotencyKey
      },
      body
    }).catch(e => ({ ok: false, status: 0, text: async () => e.message }));

    if (res.ok) {
      const data = await res.json();
      console.log('Charge succeeded:', { id: data.id, amount: data.amount, currency: data.currency });
      return data;
    }

    const text = await res.text();
    console.warn(`Attempt ${attempt} failed (status ${res.status}): ${text.replace(API_KEY, '***')}`);

    if (res.status >= 500 || res.status === 0) {
      // Retry with exponential backoff + jitter
      const backoff = Math.min(1000 * 2 ** (attempt - 1), 8000) + Math.floor(Math.random() * 250);
      await sleep(backoff);
      continue;
    }

    // Non-retryable
    throw new Error(`Charge failed: ${res.status} ${text}`);
  }
  throw new Error('Charge failed after retries');
}

createCharge(input).catch(err => {
  console.error('Final error:', err.message);
  process.exit(1);
});
