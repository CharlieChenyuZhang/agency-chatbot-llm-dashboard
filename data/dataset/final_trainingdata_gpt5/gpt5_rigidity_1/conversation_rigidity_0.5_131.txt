Macro code (exports only the “Sales” sheet):

Sub ExportSalesToCSV()
    Dim sht As Worksheet, rng As Range, r As Long, c As Long
    Dim lastRow As Long, lastCol As Long
    Dim v, s As String, line As String
    Dim savePath As Variant
    Dim stm As Object ' ADODB.Stream
    
    On Error Resume Next
    Set sht = ThisWorkbook.Worksheets("Sales")
    On Error GoTo 0
    If sht Is Nothing Then
        MsgBox "Sheet 'Sales' not found.", vbExclamation
        Exit Sub
    End If
    
    With sht.UsedRange
        lastRow = .Rows(.Rows.Count).Row
        lastCol = .Columns(.Columns.Count).Column
    End With
    Set rng = sht.Range(sht.Cells(1, 1), sht.Cells(lastRow, lastCol))
    
    savePath = Application.GetSaveAsFilename(InitialFileName:="Sales.csv", FileFilter:="CSV Files (*.csv), *.csv")
    If savePath = False Then Exit Sub
    
    Set stm = CreateObject("ADODB.Stream")
    stm.Type = 2 'text
    stm.Charset = "utf-8"
    stm.Open
    
    Application.ScreenUpdating = False
    
    For r = 1 To rng.Rows.Count
        line = ""
        For c = 1 To rng.Columns.Count
            v = rng.Cells(r, c).Value
            If IsEmpty(v) Then
                s = ""
            ElseIf IsDate(v) Then
                s = Format(v, "yyyy-mm-dd")
            ElseIf IsNumeric(v) And rng.Cells(r, c).NumberFormat <> "@" Then
                s = CStr(v)
                s = Replace(s, Application.International(xlDecimalSeparator), ".")
            Else
                s = CStr(v)
            End If
            s = Replace(s, """", """""")
            If InStr(1, s, ";") > 0 Or InStr(1, s, """") > 0 Or InStr(1, s, vbCr) > 0 Or InStr(1, s, vbLf) > 0 Or Left$(s, 1) = " " Or Right$(s, 1) = " " Then
                s = """" & s & """"
            End If
            If c > 1 Then line = line & ";"
            line = line & s
        Next c
        stm.WriteText line & vbCrLf
    Next r
    
    stm.SaveToFile savePath, 2
    stm.Close
    Set stm = Nothing
    
    Application.ScreenUpdating = True
    MsgBox "Export complete: " & savePath, vbInformation
End Sub

This reads values (not formulas), formats dates, uses dot decimals, semicolons as delimiter, and writes UTF-8.
