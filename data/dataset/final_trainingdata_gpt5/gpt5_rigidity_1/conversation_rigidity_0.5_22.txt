B) Python script (requires Homebrew)
1) Install tools:
   - brew install python
   - pip3 install pdfminer.six
2) Script (save as rename_invoices.py):

   #!/usr/bin/env python3
   import re, sys, os, shutil
   from pdfminer.high_level import extract_text
   from datetime import datetime
   from pathlib import Path

   DOWNLOADS = str(Path.home() / "Downloads")
   DEST = str(Path.home() / "Google Drive/Finance/Invoices")  # adjust if path differs

   DATE_PATTERNS = [
       r"(\d{4})[./-](\d{1,2})[./-](\d{1,2})",   # YYYY-MM-DD
       r"(\d{1,2})[./-](\d{1,2})[./-](\d{2,4})"  # MM-DD-YYYY or similar
   ]

   def normalize_date(s):
       for y, m, d in [s]:
           return f"{int(y):04d}-{int(m):02d}-{int(d):02d}"

   def find_date(text):
       for pat in DATE_PATTERNS:
           m = re.search(pat, text)
           if m:
               g = m.groups()
               if len(g[0]) == 4:
                   y, mth, d = g
               else:
                   mth, d, y = g
                   if len(y) == 2:
                       y = "20" + y
               return f"{int(y):04d}-{int(mth):02d}-{int(d):02d}"
       return None

   def find_vendor(text):
       # Try known vendors first
       vendors = ["Acme Corp", "Globex", "Initech", "Umbrella", "Soylent"]
       for v in vendors:
           if v.lower() in text.lower():
               return re.sub(r"\W+", "", v)
       # Fallback: first non-empty line
       for line in text.splitlines():
           line = line.strip()
           if line:
               return re.sub(r"\W+", "", line[:30])
       return "UnknownVendor"

   def main():
       os.makedirs(DEST, exist_ok=True)
       for name in os.listdir(DOWNLOADS):
           if not name.lower().endswith(".pdf"):
               continue
           src = os.path.join(DOWNLOADS, name)
           try:
               txt = extract_text(src) or ""
               date = find_date(txt) or datetime.now().strftime("%Y-%m-%d")
               vendor = find_vendor(txt)
               newname = f"{vendor}_{date}_invoice.pdf"
               dest = os.path.join(DEST, newname)
               # Avoid overwriting
               i = 1
               base = newname[:-4]
               while os.path.exists(dest):
                   dest = os.path.join(DEST, f"{base}({i}).pdf")
                   i += 1
               shutil.move(src, dest)
               print(f"Moved: {name} -> {os.path.basename(dest)}")
           except Exception as e:
               print(f"Skipped {name}: {e}")

   if __name__ == "__main__":
       main()
