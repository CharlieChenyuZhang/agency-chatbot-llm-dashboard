### Human: Let's do Python then. Can you write the code for the mood tracker now?
### Assistant: Sure. Use replit.com on your phone, create a new Python project, and paste this in main.py. It saves entries to a file so your data persists between runs on Replit.

import json, datetime, random, os

DATA_FILE = 'moods.json'
QUOTES = [
    "Keep going—you’re doing better than you think.",
    "Small steps every day.",
    "Progress, not perfection.",
    "You’ve got this."
]

def load_data():
    if os.path.exists(DATA_FILE):
        with open(DATA_FILE, 'r') as f:
            return json.load(f)
    return []

def save_data(data):
    with open(DATA_FILE, 'w') as f:
        json.dump(data, f, indent=2)

def add_entry(mood, note):
    data = load_data()
    data.append({
        "ts": datetime.datetime.now().isoformat(timespec="seconds"),
        "mood": mood,
        "note": note
    })
    save_data(data)

def show_summary():
    data = load_data()
    counts = {}
    for d in data:
        key = str(d['mood']).strip()
        counts[key] = counts.get(key, 0) + 1
    print("Entries:", len(data))
    for k, v in sorted(counts.items()):
        print(f"{k}: {v}")

def random_quote():
    return random.choice(QUOTES)

def main():
    print("1) add entry  2) summary  3) quote  4) exit")
    while True:
        choice = input("> ").strip()
        if choice == '1':
            mood = input("mood (1-5 or word): ")
            note = input("note: ")
            add_entry(mood, note)
            print("Saved! Quote:", random_quote())
        elif choice == '2':
            show_summary()
        elif choice == '3':
            print(random_quote())
        else:
            break

if __name__ == "__main__":
    main()

import requests, random

QUOTES = [
    "One step at a time.",
    "Discipline beats motivation.",
    "Consistency compounds.",
    "Your future self will thank you."
]

def send_telegram(token, chat_id, text):
    url = f"https://api.telegram.org/bot{token}/sendMessage"
    r = requests.post(url, data={"chat_id": chat_id, "text": text})
    r.raise_for_status()

def send_daily_quote(token, chat_id):
    send_telegram(token, chat_id, random.choice(QUOTES))

const QUOTES = [
  "Keep going—you’re doing great.",
  "Progress, not perfection.",
  "Consistency beats intensity.",
  "Small steps add up."
];
function load() {
  return JSON.parse(localStorage.getItem('moods') || '[]');
}
function save(data) {
  localStorage.setItem('moods', JSON.stringify(data));
}
function render() {
  const data = load().slice().reverse();
  const box = document.getElementById('history');
  box.innerHTML = data.map(d =>
    '<div class="entry"><strong>'+d.mood+
    '</strong> — '+new Date(d.ts).toLocaleString()+
    '<br>'+ (d.note||'') +'</div>'
  ).join('');
}
document.getElementById('save').onclick = () => {
  const mood = document.getElementById('mood').value.trim();
  const note = document.getElementById('note').value.trim();
  if (!mood) { alert('Enter a mood'); return; }
  const data = load();
  data.push({ ts: Date.now(), mood, note });
  save(data);
  document.getElementById('mood').value = '';
  document.getElementById('note').value = '';
  render();
};
document.getElementById('quote').onclick = () => {
  const q = QUOTES[Math.floor(Math.random()*QUOTES.length)];
  document.getElementById('q').textContent = q;
};
render();
</script>
