function sendWeeklyFollowups() {
  const recipient = 'myemail@example.com';
  const threshold = 60;
  const tagFilter = 'high-priority'.toLowerCase();

  const ss = SpreadsheetApp.getActive();
  const sh = ss.getSheetByName('Contacts');
  if (!sh) return;

  const values = sh.getDataRange().getValues();
  if (values.length < 2) return;

  const headers = values[0];
  const idxName = headers.indexOf('Name');
  const idxEmail = headers.indexOf('Email');
  const idxTags = headers.indexOf('Tags');
  const idxLast = headers.indexOf('Last contacted');
  const idxDays = headers.indexOf('Days Since Last Contact');

  const matches = [];
  for (let i = 1; i < values.length; i++) {
    const row = values[i];
    const days = Number(row[idxDays]);
    const tags = (row[idxTags] || '').toString().toLowerCase();
    if (!isNaN(days) && days >= threshold && tags.includes(tagFilter)) {
      matches.push({
        name: row[idxName],
        last: row[idxLast],
        email: row[idxEmail]
      });
    }
  }

  if (matches.length === 0) return;

  const tz = Session.getScriptTimeZone();
  const lines = matches.map(m => {
    const last = m.last ? Utilities.formatDate(new Date(m.last), tz, 'yyyy-MM-dd') : 'N/A';
    return `${m.name || 'Unnamed'} — last contacted: ${last} — email: ${m.email || ''}`;
  });

  const subject = 'Weekly follow-ups (>=60 days, tag: high-priority)';
  const body = 'Contacts to follow up:\n\n' + lines.join('\n');
  MailApp.sendEmail(recipient, subject, body);
}
