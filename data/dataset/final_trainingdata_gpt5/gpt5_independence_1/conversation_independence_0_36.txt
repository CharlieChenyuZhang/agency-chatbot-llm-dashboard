import sys
import os

FILE = "tasks.txt"

def load_tasks():
    if not os.path.exists(FILE):
        return []
    with open(FILE, "r", encoding="utf-8") as f:
        return [line.strip() for line in f if line.strip()]

def save_tasks(tasks):
    with open(FILE, "w", encoding="utf-8") as f:
        for t in tasks:
            f.write(t + "\n")

def main():
    if len(sys.argv) < 2:
        print("Usage: python todo.py [add <task> | list | done <index>]")
        return

    cmd = sys.argv[1].lower()
    tasks = load_tasks()

    if cmd == "add":
        if len(sys.argv) < 3:
            print("Please provide a task to add.")
            return
        task = " ".join(sys.argv[2:])
        tasks.append(task)
        save_tasks(tasks)
        print(f"Added: {task}")
    elif cmd == "list":
        if not tasks:
            print("No tasks yet.")
        else:
            for i, t in enumerate(tasks, start=1):
                print(f"{i}. {t}")
    elif cmd == "done":
        if len(sys.argv) < 3 or not sys.argv[2].isdigit():
            print("Please provide the task index to mark done.")
            return
        idx = int(sys.argv[2]) - 1
        if 0 <= idx < len(tasks):
            completed = tasks.pop(idx)
            save_tasks(tasks)
            print(f"Completed: {completed}")
        else:
            print("Invalid index.")
    else:
        print("Unknown command.")

if __name__ == "__main__":
    main()
