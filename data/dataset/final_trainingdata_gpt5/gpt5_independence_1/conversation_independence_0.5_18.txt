#!/usr/bin/env python3
import argparse, hashlib, json, os, shutil, sys, time
from pathlib import Path

CATEGORIES = {
    "docs": {".pdf", ".doc", ".docx", ".txt", ".md", ".rtf", ".xlsx", ".pptx"},
    "photos": {".jpg", ".jpeg", ".png", ".heic", ".gif", ".tif", ".tiff"},
    "archives": {".zip", ".tar", ".gz", ".bz2", ".rar", ".7z"}
}
EXCLUDE_DIRS = {"node_modules", ".git", ".Trash", "__pycache__"}
EXCLUDE_FILES = {".DS_Store"}
SIZE_SKIP_BYTES = 5 * 1024 * 1024 * 1024  # 5 GB

def sha256(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1024*1024), b""):
            h.update(chunk)
    return h.hexdigest()

def categorize(ext):
    e = ext.lower()
    for cat, exts in CATEGORIES.items():
        if e in exts:
            return cat
    return "docs" if e else "docs"

def norm_name(name):
    # Simple normalization: Title Case and single spaces
    stem, ext = os.path.splitext(name)
    stem = " ".join(stem.replace("_", " ").replace("-", " ").split()).title()
    return f"{stem}{ext}"

def within_same_volume(a, b):
    try:
        return os.stat(a).st_dev == os.stat(b).st_dev
    except FileNotFoundError:
        return False

def walk_files(root):
    for dirpath, dirnames, filenames in os.walk(root):
        dirnames[:] = [d for d in dirnames if d not in EXCLUDE_DIRS and not d.startswith(".")]
        for fn in filenames:
            if fn in EXCLUDE_FILES:
                continue
            p = Path(dirpath) / fn
            if p.is_file():
                yield p

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--source", required=True)
    ap.add_argument("--target", required=True)
    ap.add_argument("--dry-run", action="store_true")
    ap.add_argument("--apply", action="store_true")
    args = ap.parse_args()
    if args.dry_run == args.apply:
        print("Choose exactly one: --dry-run or --apply"); sys.exit(2)

    src = Path(args.source).expanduser().resolve()
    tgt = Path(args.target).expanduser().resolve()
    reports = tgt / "_Reports"
    quarantine = tgt / "_Quarantine"
    for p in (reports, quarantine):
        p.mkdir(parents=True, exist_ok=True)

    plan = {
        "generated_at": time.time(),
        "source": str(src),
        "target": str(tgt),
        "files": [],
        "dupes_by_hash": {},
        "actions": [],
        "skipped_large": [],
        "errors": []
    }

    by_hash = {}
    total = 0
    for p in walk_files(src):
        try:
            if p.stat().st_size > SIZE_SKIP_BYTES:
                plan["skipped_large"].append(str(p)); continue
            h = sha256(p)
            by_hash.setdefault(h, []).append(p)
            total += 1
        except Exception as e:
            plan["errors"].append({"path": str(p), "error": str(e)})

    plan["dupes_by_hash"] = {k: [str(x) for x in v] for k, v in by_hash.items() if len(v) > 1}

    # Build move/link actions
    for h, paths in by_hash.items():
        canonical = min(paths, key=lambda p: (p.stat().st_mtime, len(str(p))))  # prefer latest? we’ll override below
        # We actually want latest; invert min by using negative mtime
        canonical = max(paths, key=lambda p: p.stat().st_mtime)

        # Determine category from extension
        ext = canonical.suffix.lower()
        cat = categorize(ext)
        cat_dir = {"docs": "Docs", "photos": "Photos", "archives": "Archives"}[cat]

        # Destination path
        dest_dir = tgt / cat_dir
        dest_dir.mkdir(parents=True, exist_ok=True)
        dest_name = norm_name(canonical.name)
        dest_path = dest_dir / dest_name

        # Ensure uniqueness for non-identical-name collisions
        i = 2
        base = dest_path
        while dest_path.exists():
            stem, ext = os.path.splitext(base.name)
            dest_path = dest_dir / f"{stem} - v{i}{ext}"
            i += 1

        # Plan move for canonical
        plan["actions"].append({
            "type": "move_canonical",
            "from": str(canonical),
            "to": str(dest_path),
            "hash": h
        })

        # Plan hardlink replacements for duplicates
        for p in paths:
            if p == canonical: continue
            plan["actions"].append({
                "type": "replace_with_hardlink",
                "target_canonical": str(dest_path),
                "replace_path": str(p),
                "hash": h
            })

    # Write reports
    (reports / "plan.json").write_text(json.dumps(plan, indent=2))
    # Human summary
    dup_count = sum(len(v) - 1 for v in by_hash.values() if len(v) > 1)
    summary = [
        f"Total files scanned: {total}",
        f"Exact duplicate sets: {len(plan['dupes_by_hash'])}",
        f"Total duplicate instances (to replace): {dup_count}",
        f"Large files skipped: {len(plan['skipped_large'])}",
        f"Errors: {len(plan['errors'])}",
        "",
        f"Planned actions: {len(plan['actions'])} (use plan.json for details)"
    ]
    (reports / "summary.txt").write_text("\n".join(summary))

    print("\n".join(summary))

    if args.apply:
        applied = 0
        for a in plan["actions"]:
            try:
                if a["type"] == "move_canonical":
                    srcp = Path(a["from"])
                    dstp = Path(a["to"])
                    dstp.parent.mkdir(parents=True, exist_ok=True)
                    shutil.move(str(srcp), str(dstp))
                    applied += 1
                elif a["type"] == "replace_with_hardlink":
                    repl = Path(a["replace_path"])
                    canon = Path(a["target_canonical"])
                    # Remove duplicate and replace with hardlink pointing to canonical
                    if repl.exists():
                        os.remove(repl)
                    if within_same_volume(canon, repl.parent):
                        os.link(canon, repl)
                    else:
                        # Fallback: copy if cross-volume
                        shutil.copy2(canon, repl)
                    applied += 1
            except Exception as e:
                plan["errors"].append({"action": a, "error": str(e)})
        (reports / "apply_errors.json").write_text(json.dumps(plan["errors"], indent=2))
        print(f"Applied actions: {applied}. Errors: {len(plan['errors'])}")
if __name__ == "__main__":
    main()
