create_invoice() {
  local endpoint="https://api.example.com/v1/invoices"
  local token="$1"            # e.g., sk_live_...
  local customer_id="$2"      # e.g., cust_123
  local amount="$3"           # e.g., 4750
  local currency="$4"         # e.g., USD
  local source="${5:-dashboard}"  # metadata.source default

  if ! command -v jq >/dev/null || ! command -v openssl >/dev/null; then
    echo "Requires jq and openssl" >&2
    return 1
  fi

  # Build canonical payload
  local payload
  payload=$(jq -nc --arg cid "$customer_id" --arg cur "$currency" --arg src "$source" --argjson amt "$amount" \
    '{customer_id:$cid,amount:$amt,currency:$cur,metadata:{source:$src}}' | jq -c .)

  # Deterministic Idempotency-Key: sha256(payload) truncated to 32 chars
  local idempotency
  idempotency=$(printf "%s" "$payload" | openssl dgst -sha256 -r | awk '{print $1}' | cut -c1-32)

  # Logging
  local shown_token="[REDACTED]"
  if [ "$SHOW_TOKEN" = "1" ]; then shown_token="$token"; fi
  echo "POST $endpoint"
  echo "Authorization: Bearer $shown_token"
  echo "Idempotency-Key: $idempotency"
  echo "Payload: $payload"

  # Send
  curl -sS -X POST "$endpoint" \
    -H "Authorization: Bearer $token" \
    -H "Idempotency-Key: $idempotency" \
    -H "Content-Type: application/json" \
    -H "Accept: application/json" \
    -H "User-Agent: AcmeClient/1.4" \
    -H "X-Debug: true" \
    --data "$payload" \
    --max-time 10
}
