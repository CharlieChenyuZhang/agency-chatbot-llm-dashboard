2) Save this Python script as, say, nyc_top_checker.py:

   import os, json, requests, sys
   from pathlib import Path

   SUBREDDIT = "nyc"
   REDDIT_URL = f"https://www.reddit.com/r/{SUBREDDIT}/top.json?t=week&limit=1&raw_json=1"
   UA = "nyc-top-checker/1.0 (contact: you@example.com)"
   STATE_FILE = Path.home() / ".nyc_top_last_id"
   EXCLUDE_NSFW = True

   def get_top_post():
       r = requests.get(REDDIT_URL, headers={"User-Agent": UA}, timeout=15)
       r.raise_for_status()
       data = r.json()["data"]["children"][0]["data"]
       return {
           "id": data["id"],
           "title": data["title"],
           "score": data["score"],
           "num_comments": data["num_comments"],
           "permalink": "https://www.reddit.com" + data["permalink"],
           "over_18": data.get("over_18", False),
           "thumbnail": data.get("thumbnail"),
       }

   def post_to_slack(webhook, title, score, comments, permalink, thumbnail=None):
       # Basic text plus optional image block
       blocks = []
       if thumbnail and isinstance(thumbnail, str) and thumbnail.startswith("http"):
           blocks.append({
               "type": "image",
               "image_url": thumbnail,
               "alt_text": "thumbnail"
           })
       blocks.append({
           "type": "section",
           "text": {
               "type": "mrkdwn",
               "text": f":trophy: New top post in r/{SUBREDDIT} (This week)\n*{title}*\nScore: {score} | Comments: {comments}\n{permalink}"
           }
       })
       resp = requests.post(webhook, json={"blocks": blocks}, timeout=15)
       resp.raise_for_status()

   def main():
       webhook = os.environ.get("SLACK_WEBHOOK_URL")
       if not webhook:
           print("SLACK_WEBHOOK_URL not set", file=sys.stderr)
           sys.exit(1)

       top = get_top_post()
       if EXCLUDE_NSFW and top["over_18"]:
           # If the top post is NSFW, skip notifying and do not update state
           sys.exit(0)

       last_id = STATE_FILE.read_text().strip() if STATE_FILE.exists() else None
       if top["id"] == last_id:
           # No change
           sys.exit(0)

       # New top post detected
       post_to_slack(webhook, top["title"], top["score"], top["num_comments"], top["permalink"], top.get("thumbnail"))

       # Update state
       STATE_FILE.write_text(top["id"])

   if __name__ == "__main__":
       main()
