```python
import random
from collections import deque
import numpy as np
import pygame
import torch
import torch.nn as nn
import torch.optim as optim

# ---------- Game config ----------
BLOCK_SIZE = 20
GRID_W, GRID_H = 20, 20  # 20x20 grid
SPEED = 0  # training: no delay

# ---------- Snake Game Environment ----------
class SnakeGame:
    def __init__(self):
        pygame.init()
        self.w = GRID_W * BLOCK_SIZE
        self.h = GRID_H * BLOCK_SIZE
        self.display = pygame.display.set_mode((self.w, self.h))
        pygame.display.set_caption('Snake RL')
        self.clock = pygame.time.Clock()
        self.reset()

    def reset(self):
        self.direction = (1, 0)  # moving right
        self.head = (GRID_W // 2, GRID_H // 2)
        self.snake = [self.head,
                      (self.head[0] - 1, self.head[1]),
                      (self.head[0] - 2, self.head[1])]
        self.score = 0
        self._place_food()
        self.frame_iteration = 0
        return self._get_state()

    def _place_food(self):
        while True:
            x = random.randint(0, GRID_W - 1)
            y = random.randint(0, GRID_H - 1)
            self.food = (x, y)
            if self.food not in self.snake:
                break

    def _get_state(self):
        head_x, head_y = self.head
        dx, dy = self.direction

        # positions around the head
        left_dir = (-dy, dx)
        right_dir = (dy, -dx)

        def danger_at(dir_vec):
            nx = head_x + dir_vec[0]
            ny = head_y + dir_vec[1]
            return (nx < 0 or nx >= GRID_W or
                    ny < 0 or ny >= GRID_H or
                    (nx, ny) in self.snake[1:])

        danger_straight = danger_at(self.direction)
        danger_left = danger_at(left_dir)
        danger_right = danger_at(right_dir)

        # food location relative to head
        food_left = self.food[0] < head_x
        food_right = self.food[0] > head_x
        food_up = self.food[1] < head_y
        food_down = self.food[1] > head_y

        state = [
            int(danger_straight),
            int(danger_left),
            int(danger_right),
            int(dx == -1),  # moving left
            int(dx == 1),   # moving right
            int(dy == -1),  # moving up
            int(dy == 1),   # moving down
            int(food_left),
            int(food_right),
            int(food_up),
            int(food_down),
        ]
        return np.array(state, dtype=np.float32)

    def step(self, action):
        # action: 0 = straight, 1 = right, 2 = left
        self.frame_iteration += 1
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                pygame.quit()
                quit()

        dx, dy = self.direction
        if action == 1:  # turn right
            dx, dy = (dy, -dx)
        elif action == 2:  # turn left
            dx, dy = (-dy, dx)
        self.direction = (dx, dy)

        new_head = (self.head[0] + dx, self.head[1] + dy)
        self.head = new_head
        self.snake.insert(0, self.head)

        reward = 0
        done = False

        # collision
        if (self.head[0] < 0 or self.head[0] >= GRID_W or
                self.head[1] < 0 or self.head[1] >= GRID_H or
                self.head in self.snake[1:]):
            done = True
            reward = -10
            return self._get_state(), reward, done, self.score

        # food
        if self.head == self.food:
            self.score += 1
            reward = 10
            self._place_food()
        else:
            self.snake.pop()

        # optional: die if stuck too long
        if self.frame_iteration > 100 * len(self.snake):
            done = True
            reward = -10

        return self._get_state(), reward, done, self.score

    def render(self):
        self.display.fill((0, 0, 0))
        # draw snake
        for x, y in self.snake:
            pygame.draw.rect(self.display, (0, 255, 0),
                             pygame.Rect(x * BLOCK_SIZE, y * BLOCK_SIZE,
                                         BLOCK_SIZE, BLOCK_SIZE))
        # draw food
        fx, fy = self.food
        pygame.draw.rect(self.display, (255, 0, 0),
                         pygame.Rect(fx * BLOCK_SIZE, fy * BLOCK_SIZE,
                                     BLOCK_SIZE, BLOCK_SIZE))
        pygame.display.flip()
        self.clock.tick(SPEED)

# ---------- DQN Model ----------
class DQN(nn.Module):
    def __init__(self, input_dim, hidden_dim, output_dim):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, output_dim)
        )

    def forward(self, x):
        return self.net(x)

# ---------- Agent with Replay Memory ----------
class Agent:
    def __init__(self, state_dim, action_dim, gamma=0.9, lr=0.001,
                 max_mem=50_000, batch_size=100):
        self.gamma = gamma
        self.lr = lr
        self.action_dim = action_dim
        self.epsilon = 1.0
        self.eps_min = 0.01
        self.eps_decay = 0.995
        self.batch_size = batch_size

        self.memory = deque(maxlen=max_mem)
        self.device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
        self.policy_net = DQN(state_dim, 128, action_dim).to(self.device)
        self.target_net = DQN(state_dim, 128, action_dim).to(self.device)
        self.target_net.load_state_dict(self.policy_net.state_dict())
        self.target_net.eval()

        self.optimizer = optim.Adam(self.policy_net.parameters(), lr=lr)
        self.loss_fn = nn.MSELoss()
        self.learn_step = 0
        self.target_update_freq = 1000

    def remember(self, state, action, reward, next_state, done):
        self.memory.append((state, action, reward, next_state, done))

    def act(self, state):
        if random.random() < self.epsilon:
            return random.randint(0, self.action_dim - 1)
        state = torch.tensor(state, dtype=torch.float32, device=self.device).unsqueeze(0)
        with torch.no_grad():
            q_values = self.policy_net(state)
        return int(torch.argmax(q_values).item())

    def replay(self):
        if len(self.memory) < self.batch_size:
            return

        batch = random.sample(self.memory, self.batch_size)
        states, actions, rewards, next_states, dones = zip(*batch)

        states = torch.tensor(states, dtype=torch.float32, device=self.device)
        actions = torch.tensor(actions, dtype=torch.int64, device=self.device).unsqueeze(1)
        rewards = torch.tensor(rewards, dtype=torch.float32, device=self.device).unsqueeze(1)
        next_states = torch.tensor(next_states, dtype=torch.float32, device=self.device)
        dones = torch.tensor(dones, dtype=torch.float32, device=self.device).unsqueeze(1)

        q_pred = self.policy_net(states).gather(1, actions)
        with torch.no_grad():
            q_next = self.target_net(next_states).max(1)[0].unsqueeze(1)
            q_target = rewards + self.gamma * q_next * (1 - dones)

        loss = self.loss_fn(q_pred, q_target)
        self.optimizer.zero_grad()
        loss.backward()
        self.optimizer.step()

        self.learn_step += 1
        if self.learn_step % self.target_update_freq == 0:
            self.target_net.load_state_dict(self.policy_net.state_dict())

        # decay epsilon
        if self.epsilon > self.eps_min:
            self.epsilon *= self.eps_decay

# ---------- Training Loop ----------
def train(num_episodes=500):
    game = SnakeGame()
    state_dim = len(game.reset())
    action_dim = 3  # straight, right, left
    agent = Agent(state_dim, action_dim)

    scores = []
    for ep in range(num_episodes):
        state = game.reset()
        done = False
        total_reward = 0

        while not done:
            # comment out render() to speed up training
            # game.render()

            action = agent.act(state)
            next_state, reward, done, score = game.step(action)
            agent.remember(state, action, reward, next_state, done)
            agent.replay()

            state = next_state
            total_reward += reward

        scores.append(score)
        avg_score = sum(scores[-50:]) / min(len(scores), 50)
        print(f"Episode {ep+1} | Score: {score} | Avg(50): {avg_score:.2f} | Eps: {agent.epsilon:.3f}")

    pygame.quit()

if __name__ == "__main__":
    train(num_episodes=1000)
```

- Simplified (small state, no convolution, simple rewards).

It may need many episodes to learn decent play, and performance will vary. But you can run this, see it improve (if you enable `render()` occasionally), and then iterate.
