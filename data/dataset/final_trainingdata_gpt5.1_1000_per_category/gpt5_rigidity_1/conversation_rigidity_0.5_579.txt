### Human: Let’s make them stay after refresh. I really want to see persistence.
### Assistant: Persistence is a great next step, but it’s a bit more complex than what we’ve done so far. Still, we can do a straightforward version.

We’ll:
- Store the notes in an **array** in JavaScript.
- Save that array as JSON in `localStorage`.
- Load from `localStorage` when the page starts.

This is slightly advanced for a total beginner, so I’ll keep the code tidy and label the “magic parts” so you know what to revisit later.

Here’s a version that does persistence:

```javascript
const noteInput = document.getElementById("noteInput");
const addNoteButton = document.getElementById("addNoteButton");
const notesList = document.getElementById("notesList");

let notes = [];

// Load notes from localStorage when the page loads
function loadNotes() {
  const saved = localStorage.getItem("notes");
  if (saved) {
    notes = JSON.parse(saved);
    notes.forEach((text) => {
      createNoteElement(text);
    });
  }
}

function saveNotes() {
  localStorage.setItem("notes", JSON.stringify(notes));
}

function createNoteElement(text) {
  const li = document.createElement("li");
  li.textContent = text;

  li.addEventListener("click", function () {
    // Remove from array
    notes = notes.filter((noteText) => noteText !== text);
    saveNotes();

    // Remove from page
    notesList.removeChild(li);
  });

  notesList.appendChild(li);
}

addNoteButton.addEventListener("click", function () {
  const text = noteInput.value.trim();

  if (text === "") {
    alert("Please type something before adding a note.");
    return;
  }

  notes.push(text);
  saveNotes();

  createNoteElement(text);

  noteInput.value = "";
  noteInput.focus();
});

// Initialize
loadNotes();
```

Short explanation of the “magic parts”:
- `localStorage.setItem` and `getItem`: Save and load data in the browser.
- `JSON.stringify` / `JSON.parse`: Turn the array into a string (to save) and back into an array (to use).
