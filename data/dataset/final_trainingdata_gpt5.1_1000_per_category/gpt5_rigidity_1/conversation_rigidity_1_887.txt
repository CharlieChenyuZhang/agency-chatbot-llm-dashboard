awk '{
  sum=$1; $1="";
  sub(/^ +/,"",$0);
  path=$0;
  if (sum in seen) {
    print path
  } else {
    seen[sum]=path
  }
}' /tmp/all_files_with_checksums.txt > /tmp/duplicate_files.txt

while IFS= read -r file; do
  [ -f "$file" ] || continue
  newest="$file"
  checksum="$(shasum "$file" | awk "{print \$1}")"
  while IFS= read -r other; do
    [ -f "$other" ] || continue
    [ "$other" = "$file" ] && continue
    c2="$(shasum "$other" | awk "{print \$1}")"
    [ "$c2" != "$checksum" ] && continue
    if [ "$other" -nt "$newest" ]; then
      newest="$other"
    fi
  done < <(awk -v c="$checksum" '$1==c{$1="";sub(/^ +/,"",$0);print $0}' /tmp/all_files_with_checksums.txt)
  while IFS= read -r other; do
    [ -f "$other" ] || continue
    [ "$other" = "$newest" ] && continue
    c2="$(shasum "$other" | awk "{print \$1}")"
    [ "$c2" != "$checksum" ] && continue
    rm "$other"
  done < <(awk -v c="$checksum" '$1==c{$1="";sub(/^ +/,"",$0);print $0}' /tmp/all_files_with_checksums.txt)
done < /tmp/duplicate_files.txt
