First, here’s a script that expects a list of customers in `customers.json` and produces `customers.csv` with standard fields.

```python
#!/usr/bin/env python3
import json
import csv
from pathlib import Path

INPUT_FILE = Path("customers.json")
OUTPUT_FILE = Path("customers.csv")

# Define the fields/columns you want in the CSV and their JSON paths.
FIELDS = [
    ("customer_id", "id"),
    ("first_name", "first_name"),
    ("last_name", "last_name"),
    ("email", "email"),
    ("phone", "phone"),
    ("country", "address.country"),
    ("city", "address.city"),
    ("signup_date", "signup_date"),
    ("last_activity", "last_activity"),
    ("total_orders", "stats.total_orders"),
    ("total_spend", "stats.total_spend"),
    ("status", "status"),
]

def get_nested_value(obj, path, default=""):
    parts = path.split(".")
    for part in parts:
        if isinstance(obj, dict) and part in obj:
            obj = obj[part]
        else:
            return default
    return obj if obj is not None else default

def main():
    if not INPUT_FILE.exists():
        raise FileNotFoundError(f"{INPUT_FILE} not found")

    with INPUT_FILE.open("r", encoding="utf-8") as f:
        data = json.load(f)

    # If your JSON is like {"customers": [...]}, adjust here:
    if isinstance(data, dict) and "customers" in data:
        customers = data["customers"]
    else:
        customers = data  # assume it's already a list

    with OUTPUT_FILE.open("w", newline="", encoding="utf-8") as csvfile:
        writer = csv.writer(csvfile)
        header = [label for (label, _) in FIELDS]
        writer.writerow(header)

        for customer in customers:
            row = [get_nested_value(customer, path) for (_, path) in FIELDS]
            writer.writerow(row)

    print(f"Wrote {len(customers)} customers to {OUTPUT_FILE}")

if __name__ == "__main__":
    main()
```
