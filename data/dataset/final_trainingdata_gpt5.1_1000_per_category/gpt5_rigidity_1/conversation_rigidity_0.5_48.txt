Below is a single Python example that:
- Fetches museums from a specified Wikipedia category.
- Uses OpenStreetMap’s Nominatim service for basic geocoding (light usage only, you should respect their usage policy).
- Outputs a Markdown table.

You’ll need `requests` installed (`pip install requests`).

```python
import requests
import time

WIKI_API_URL = "https://en.wikipedia.org/w/api.php"
CATEGORY_TITLE = "Category:Museums_in_Pittsburgh"
USER_AGENT = "PghMuseumListBot/0.1 (your_email@example.com)"

def get_category_members(category_title):
    members = []
    cmcontinue = None
    while True:
        params = {
            "action": "query",
            "list": "categorymembers",
            "cmtitle": category_title,
            "cmlimit": 50,
            "format": "json"
        }
        if cmcontinue:
            params["cmcontinue"] = cmcontinue
        r = requests.get(WIKI_API_URL, params=params, headers={"User-Agent": USER_AGENT})
        r.raise_for_status()
        data = r.json()
        members.extend(data["query"]["categorymembers"])
        if "continue" in data:
            cmcontinue = data["continue"]["cmcontinue"]
        else:
            break
    return members

def geocode_place(name):
    url = "https://nominatim.openstreetmap.org/search"
    params = {
        "q": f"{name}, Pittsburgh, PA",
        "format": "json",
        "limit": 1
    }
    r = requests.get(url, params=params, headers={"User-Agent": USER_AGENT})
    r.raise_for_status()
    data = r.json()
    if not data:
        return None, None, None
    result = data[0]
    return result.get("display_name"), result.get("lat"), result.get("lon")

def main():
    museums = get_category_members(CATEGORY_TITLE)
    print("## Museums in Pittsburgh (from Wikipedia category)\n")
    print("> Automatically fetched from Wikipedia category and geocoded via OpenStreetMap Nominatim. Please verify manually.\n")
    print("| Museum | Address / Location (from OSM) | Latitude | Longitude | Wikipedia Page |")
    print("|--------|-------------------------------|----------|-----------|----------------|")
    for m in museums:
        title = m["title"]
        address, lat, lon = geocode_place(title)
        wiki_url = f"https://en.wikipedia.org/wiki/{title.replace(' ', '_')}"
        print(f"| {title} | {address or ''} | {lat or ''} | {lon or ''} | {wiki_url} |")
        time.sleep(1)  # be polite to the OSM servers

if __name__ == "__main__":
    main()
```
