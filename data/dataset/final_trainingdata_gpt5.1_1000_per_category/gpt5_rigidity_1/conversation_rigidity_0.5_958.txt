**2. Create `generate_readme_sections.py`:**

This script will print the two Markdown tables so you can paste them into the README, or you can eventually automate updating the README structure around them.

```python
import json
from pathlib import Path

DATA_FILE = Path("oscars.json")

def load_data():
    with DATA_FILE.open() as f:
        return json.load(f)

def oscar_winners_table(oscar_winners):
    lines = []
    lines.append("| Release Year | Oscar Ceremony Year | Film            | Oscars Won | Categories |")
    lines.append("|-------------|---------------------|-----------------|------------|-----------|")
    for entry in sorted(oscar_winners, key=lambda e: e["release_year"]):
        categories = ", ".join(entry["categories"])
        lines.append(
            f"| {entry['release_year']}        | {entry['ceremony_year']}                | {entry['film']} | {entry['wins']}          | {categories} |"
        )
    return "\n".join(lines)

def nominations_summary_table(nominations):
    lines = []
    lines.append("| Release Year | Film               | Nominations | Wins | Notes |")
    lines.append("|-------------|--------------------|------------|------|-------|")
    for entry in sorted(nominations, key=lambda e: e["release_year"]):
        # You can refine 'Notes' manually later if you want
        lines.append(
            f"| {entry['release_year']}        | {entry['film']:<18} | {entry['nominations']}          | {entry['wins']}    |  |"
        )
    return "\n".join(lines)

def main():
    data = load_data()
    print("## Oscar-Winning Christopher Nolan Films\n")
    print(oscar_winners_table(data["oscar_winners"]))
    print("\n\n## Oscar Nominations for Christopher Nolan Films\n")
    print("Summary of Academy Award nominations and wins for films directed by Christopher Nolan.\n")
    print(nominations_summary_table(data["nominations_summary"]))

if __name__ == "__main__":
    main()
```
