### Human: Let’s do plain HTML and res.sendFile like you suggest. Can you give me that server.js skeleton now?

### Assistant: Here’s a lean skeleton that matches the structure I prefer. You can adapt paths as needed.

```js
// server.js
const path = require('path');
const express = require('express');
const Database = require('better-sqlite3');

const app = express();
const PORT = 3000;

// Middleware
app.use(express.urlencoded({ extended: true }));
app.use(express.json());
app.use(express.static(path.join(__dirname, 'public')));

// Database setup
const db = new Database(path.join(__dirname, 'data', 'workouts.db'));

// Create table if not exists
db.prepare(`
  CREATE TABLE IF NOT EXISTS workouts (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    type TEXT NOT NULL,
    start_time TEXT NOT NULL,
    end_time TEXT NOT NULL,
    duration_minutes REAL NOT NULL,
    notes TEXT,
    created_at TEXT NOT NULL
  )
`).run();

// Routes
app.get('/', (req, res) => {
  res.redirect('/today');
});

app.get('/today', (req, res) => {
  res.sendFile(path.join(__dirname, 'views', 'today.html'));
});

app.post('/workouts', (req, res) => {
  const { type, start_time, end_time, duration_minutes, notes } = req.body;
  const created_at = new Date().toISOString();

  const stmt = db.prepare(`
    INSERT INTO workouts (type, start_time, end_time, duration_minutes, notes, created_at)
    VALUES (?, ?, ?, ?, ?, ?)
  `);
  stmt.run(type, start_time, end_time, duration_minutes, notes || '', created_at);

  res.redirect('/history');
});

app.get('/history', (req, res) => {
  res.sendFile(path.join(__dirname, 'views', 'history.html'));
});

app.get('/api/workouts', (req, res) => {
  const rows = db.prepare(`
    SELECT * FROM workouts ORDER BY created_at DESC
  `).all();
  res.json(rows);
});

app.listen(PORT, () => {
  console.log(`Workout tracker running at http://localhost:${PORT}`);
});
```

I’ve deliberately kept logic simple and pushed any “fancy” display (like aggregations and charts) to the frontend for later, so you actually get the basics working first.
