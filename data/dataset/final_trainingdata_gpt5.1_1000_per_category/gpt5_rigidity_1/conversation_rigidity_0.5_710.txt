Here’s a compact script (single file):

```python
import random
from collections import deque
import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
import gymnasium as gym

# Hyperparameters
GAMMA = 0.99
LR = 1e-3
BATCH_SIZE = 64
BUFFER_SIZE = 50_000
MIN_REPLAY_SIZE = 1_000
EPS_START = 1.0
EPS_END = 0.05
EPS_DECAY = 500  # smaller -> faster decay
TARGET_UPDATE_FREQ = 1000  # steps
TRAINING_STEPS = 50_000

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

class QNetwork(nn.Module):
    def __init__(self, obs_dim, n_actions):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(obs_dim, 128),
            nn.ReLU(),
            nn.Linear(128, 128),
            nn.ReLU(),
            nn.Linear(128, n_actions)
        )

    def forward(self, x):
        return self.net(x)

def main():
    env = gym.make("CartPole-v1")
    obs, _ = env.reset(seed=0)

    obs_dim = env.observation_space.shape[0]
    n_actions = env.action_space.n

    policy_net = QNetwork(obs_dim, n_actions).to(device)
    target_net = QNetwork(obs_dim, n_actions).to(device)
    target_net.load_state_dict(policy_net.state_dict())
    target_net.eval()

    optimizer = optim.Adam(policy_net.parameters(), lr=LR)

    replay_buffer = deque(maxlen=BUFFER_SIZE)

    # Fill replay buffer with random experience
    while len(replay_buffer) < MIN_REPLAY_SIZE:
        action = env.action_space.sample()
        next_obs, reward, terminated, truncated, _ = env.step(action)
        done = terminated or truncated
        replay_buffer.append((obs, action, reward, next_obs, done))
        obs = next_obs
        if done:
            obs, _ = env.reset()

    obs, _ = env.reset(seed=1)
    episode_reward = 0.0

    for step in range(TRAINING_STEPS):
        # Epsilon-greedy action selection
        epsilon = EPS_END + (EPS_START - EPS_END) * np.exp(-step / EPS_DECAY)
        if random.random() < epsilon:
            action = env.action_space.sample()
        else:
            obs_t = torch.as_tensor(obs, dtype=torch.float32, device=device).unsqueeze(0)
            q_values = policy_net(obs_t)
            action = int(torch.argmax(q_values, dim=1).item())

        next_obs, reward, terminated, truncated, _ = env.step(action)
        done = terminated or truncated

        replay_buffer.append((obs, action, reward, next_obs, done))
        obs = next_obs
        episode_reward += reward

        if done:
            obs, _ = env.reset()
            print(f"Step {step}, episode reward: {episode_reward}")
            episode_reward = 0.0

        # Sample a batch and update
        batch = random.sample(replay_buffer, BATCH_SIZE)
        obs_batch, act_batch, rew_batch, next_obs_batch, done_batch = zip(*batch)

        obs_batch = torch.as_tensor(obs_batch, dtype=torch.float32, device=device)
        act_batch = torch.as_tensor(act_batch, dtype=torch.int64, device=device).unsqueeze(-1)
        rew_batch = torch.as_tensor(rew_batch, dtype=torch.float32, device=device).unsqueeze(-1)
        next_obs_batch = torch.as_tensor(next_obs_batch, dtype=torch.float32, device=device)
        done_batch = torch.as_tensor(done_batch, dtype=torch.float32, device=device).unsqueeze(-1)

        # Compute target Q-values
        with torch.no_grad():
            next_q_values = target_net(next_obs_batch)
            max_next_q_values = next_q_values.max(dim=1, keepdim=True)[0]
            target_q_values = rew_batch + GAMMA * (1 - done_batch) * max_next_q_values

        # Compute current Q-values
        q_values = policy_net(obs_batch)
        action_q_values = q_values.gather(1, act_batch)

        loss = nn.functional.mse_loss(action_q_values, target_q_values)

        optimizer.zero_grad()
        loss.backward()
        optimizer.step()

        # Periodically update target network
        if step % TARGET_UPDATE_FREQ == 0:
            target_net.load_state_dict(policy_net.state_dict())

    env.close()

if __name__ == "__main__":
    main()
```
