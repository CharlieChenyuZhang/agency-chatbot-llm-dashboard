```python
import os
from mutagen import File

ROOT = os.path.expanduser("~/Music/Organized")  # adjust if needed
MIN_BITRATE = 192000  # 192 kbps in bits per second
LOG_PATH = os.path.expanduser("~/deleted_low_bitrate.log")
AUDIO_EXT = {".mp3", ".m4a", ".flac", ".ogg", ".wav", ".aac"}

def is_audio(path):
    return os.path.splitext(path)[1].lower() in AUDIO_EXT

def main():
    deleted = 0
    with open(LOG_PATH, "a", encoding="utf-8") as log:
        for dirpath, _, filenames in os.walk(ROOT):
            for name in filenames:
                path = os.path.join(dirpath, name)
                if not is_audio(path):
                    continue
                try:
                    audio = File(path, easy=False)
                    if not audio:
                        continue
                    info = getattr(audio, "info", None)
                    if not info or not hasattr(info, "bitrate") or not info.bitrate:
                        continue
                    if info.bitrate < MIN_BITRATE:
                        os.remove(path)
                        log.write(f"{path}\t{info.bitrate}\n")
                        deleted += 1
                except Exception as e:
                    # Skip problematic files but don’t crash
                    continue
    print(f"Deleted {deleted} files. Details in {LOG_PATH}")

if __name__ == "__main__":
    main()
```

```python
import os
import hashlib

ROOT = os.path.expanduser("~/Music/Organized")
LOG_PATH = os.path.expanduser("~/deleted_duplicates.log")
AUDIO_EXT = {".mp3", ".m4a", ".flac", ".ogg", ".wav", ".aac"}

CHUNK_SIZE = 1024 * 1024  # 1 MB

def is_audio(path):
    return os.path.splitext(path)[1].lower() in AUDIO_EXT

def file_hash(path):
    h = hashlib.md5()
    with open(path, "rb") as f:
        while chunk := f.read(CHUNK_SIZE):
            h.update(chunk)
    return h.hexdigest()

def main():
    seen = {}
    deleted = 0
    with open(LOG_PATH, "a", encoding="utf-8") as log:
        for dirpath, _, filenames in os.walk(ROOT):
            for name in filenames:
                path = os.path.join(dirpath, name)
                if not is_audio(path):
                    continue
                try:
                    h = file_hash(path)
                except Exception:
                    # Skip unreadable files
                    continue

                if h in seen:
                    # Duplicate: delete this one
                    try:
                        os.remove(path)
                        log.write(f"DELETED\t{path}\tDUP_OF\t{seen[h]}\n")
                        deleted += 1
                    except Exception:
                        continue
                else:
                    seen[h] = path
    print(f"Deleted {deleted} duplicate files. Details in {LOG_PATH}")

if __name__ == "__main__":
    main()
```

Save and exit.

- Delete subsequent duplicates.
- Log `DELETED <path> DUP_OF <kept-path>` in `deleted_duplicates.log`.
