```python
import csv
from datetime import datetime
from pathlib import Path

DATA_FILE = Path("subway_delays.csv")

FIELDS = [
    "date",
    "line",
    "origin",
    "destination",
    "planned_departure",
    "actual_departure",
    "actual_arrival",
    "delay_minutes",
    "notes",
]

def init_file():
    if not DATA_FILE.exists():
        with DATA_FILE.open("w", newline="", encoding="utf-8") as f:
            writer = csv.DictWriter(f, fieldnames=FIELDS)
            writer.writeheader()

def add_entry():
    print("Add a new commute entry:")
    date = input("Date (YYYY-MM-DD, blank for today): ").strip()
    if not date:
        date = datetime.now().strftime("%Y-%m-%d")

    line = input("Line (e.g., A, 4, Q): ").strip()
    origin = input("Origin station: ").strip()
    destination = input("Destination station: ").strip()
    planned = input("Planned departure time (HH:MM): ").strip()
    actual_dep = input("Actual departure time (HH:MM): ").strip()
    actual_arr = input("Actual arrival time (HH:MM): ").strip()
    notes = input("Notes (optional): ").strip()

    def to_minutes(t):
        h, m = map(int, t.split(":"))
        return h * 60 + m

    delay = to_minutes(actual_dep) - to_minutes(planned)

    row = {
        "date": date,
        "line": line,
        "origin": origin,
        "destination": destination,
        "planned_departure": planned,
        "actual_departure": actual_dep,
        "actual_arrival": actual_arr,
        "delay_minutes": delay,
        "notes": notes,
    }

    with DATA_FILE.open("a", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=FIELDS)
        writer.writerow(row)

    print("Entry saved.")

def summary_by_line():
    stats = {}
    with DATA_FILE.open("r", newline="", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        for row in reader:
            line = row["line"]
            delay = int(row["delay_minutes"])
            stats.setdefault(line, {"total_delay": 0, "count": 0})
            stats[line]["total_delay"] += delay
            stats[line]["count"] += 1

    print("Average departure delay by line (minutes):")
    for line, s in stats.items():
        avg = s["total_delay"] / s["count"]
        print(f"{line}: {avg:.1f}")

def main():
    init_file()
    print("1) Add entry")
    print("2) Summary by line")
    choice = input("Choose an option: ").strip()
    if choice == "1":
        add_entry()
    elif choice == "2":
        summary_by_line()
    else:
        print("Unknown option.")

if __name__ == "__main__":
    main()
```
