4. In the new module, paste this code:

```vba
Sub ExportSheetsToCsvUtf8()
    Dim ws As Worksheet
    Dim wb As Workbook
    Dim savePath As String
    Dim baseName As String
    Dim fullFileName As String

    Set wb = ThisWorkbook
    savePath = wb.Path
    If savePath = "" Then
        MsgBox "Please save the workbook first.", vbExclamation
        Exit Sub
    End If

    baseName = Left(wb.Name, InStrRev(wb.Name, ".") - 1)

    For Each ws In wb.Worksheets
        If ws.Visible = xlSheetVisible Then
            fullFileName = savePath & "\" & baseName & "_" & ws.Name & ".csv"
            ws.Copy
            With ActiveWorkbook
                .SaveAs Filename:=fullFileName, _
                        FileFormat:=xlCSVUTF8, _
                        CreateBackup:=False
                .Close SaveChanges:=False
            End With
        End If
    Next ws

    MsgBox "Export complete. Files saved to: " & savePath, vbInformation
End Sub
```
