```python
import numpy as np

class GridWorldEnv:
    def __init__(self, rows=5, cols=5, max_steps=50):
        self.rows = rows
        self.cols = cols
        self.max_steps = max_steps

        # Define special cells
        self.start = (4, 0)
        self.goal = (0, 4)
        self.walls = {(1, 1), (1, 2), (2, 1)}

        # Actions: 0=up, 1=right, 2=down, 3=left
        self.action_space_n = 4
        self.observation_space_n = rows * cols

        self.reset()

    def _to_state(self, pos):
        r, c = pos
        return r * self.cols + c

    def _from_state(self, state):
        r = state // self.cols
        c = state % self.cols
        return (r, c)

    def reset(self):
        self.pos = self.start
        self.steps = 0
        return self._to_state(self.pos)

    def step(self, action):
        assert 0 <= action < self.action_space_n

        self.steps += 1
        r, c = self.pos

        # Proposed next position
        if action == 0:    # up
            new_pos = (r - 1, c)
        elif action == 1:  # right
            new_pos = (r, c + 1)
        elif action == 2:  # down
            new_pos = (r + 1, c)
        else:              # left
            new_pos = (r, c - 1)

        # Check bounds and walls
        out_of_bounds = not (0 <= new_pos[0] < self.rows and 0 <= new_pos[1] < self.cols)
        into_wall = new_pos in self.walls

        if out_of_bounds or into_wall:
            # Invalid move: stay in place, stronger penalty
            new_pos = self.pos
            reward = -0.05
        else:
            reward = -0.01  # step penalty

        self.pos = new_pos

        # Check for goal
        done = False
        if self.pos == self.goal:
            reward = 1.0
            done = True
        elif self.steps >= self.max_steps:
            done = True

        next_state = self._to_state(self.pos)
        info = {}

        return next_state, reward, done, info

    def render(self):
        grid = [['.' for _ in range(self.cols)] for _ in range(self.rows)]
        for (wr, wc) in self.walls:
            grid[wr][wc] = '#'
        gr, gc = self.goal
        grid[gr][gc] = 'G'
        r, c = self.pos
        grid[r][c] = 'A'

        for row in grid:
            print(' '.join(row))
        print()

# Q-learning agent
def q_learning(
    env,
    episodes=500,
    alpha=0.1,
    gamma=0.99,
    epsilon_start=1.0,
    epsilon_end=0.1,
    epsilon_decay_episodes=300
):
    n_states = env.observation_space_n
    n_actions = env.action_space_n
    Q = np.zeros((n_states, n_actions))

    def get_epsilon(episode):
        if episode >= epsilon_decay_episodes:
            return epsilon_end
        frac = episode / epsilon_decay_episodes
        return epsilon_start + frac * (epsilon_end - epsilon_start)

    rewards_per_episode = []

    for ep in range(episodes):
        state = env.reset()
        done = False
        total_reward = 0.0
        epsilon = get_epsilon(ep)

        while not done:
            # Epsilon-greedy policy
            if np.random.rand() < epsilon:
                action = np.random.randint(n_actions)
            else:
                action = np.argmax(Q[state])

            next_state, reward, done, info = env.step(action)

            # Q-learning update
            best_next = np.max(Q[next_state])
            td_target = reward + gamma * best_next
            td_error = td_target - Q[state, action]
            Q[state, action] += alpha * td_error

            state = next_state
            total_reward += reward

        rewards_per_episode.append(total_reward)

        if (ep + 1) % 50 == 0:
            avg_reward = np.mean(rewards_per_episode[-50:])
            print(f"Episode {ep+1}/{episodes}, eps={epsilon:.2f}, avg_reward(last50)={avg_reward:.3f}")

    return Q, rewards_per_episode

if __name__ == "__main__":
    env = GridWorldEnv()
    Q, rewards = q_learning(env)
```
