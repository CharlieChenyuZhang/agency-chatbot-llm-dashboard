```powershell
$base = "C:\Users\YourName\Pictures\_MasterPhotos"
$inbox = Join-Path $base "_Inbox_RawImports"

Get-ChildItem -Path $inbox -File | ForEach-Object {
    $file = $_

    # Prefer DateTaken from metadata if available (for photos),
    # otherwise fall back to CreationTime
    $date = $file.CreationTime
    try {
        $shell = New-Object -ComObject Shell.Application
        $folder = $shell.Namespace($file.DirectoryName)
        $item   = $folder.ParseName($file.Name)
        $dateTakenStr = $folder.GetDetailsOf($item, 12)  # "Date taken" column, may vary by system locale
        if ($dateTakenStr) {
            $parsed = [datetime]::Parse($dateTakenStr)
            if ($parsed.Year -gt 1980) {
                $date = $parsed
            }
        }
    } catch {}

    $year = $date.Year
    $monthNum = "{0:00}" -f $date.Month
    $monthFolderName = "$year-$monthNum"

    $targetYearFolder  = Join-Path $base $year
    $targetMonthFolder = Join-Path $targetYearFolder $monthFolderName

    if (!(Test-Path $targetYearFolder)) {
        New-Item -ItemType Directory -Path $targetYearFolder | Out-Null
    }
    if (!(Test-Path $targetMonthFolder)) {
        New-Item -ItemType Directory -Path $targetMonthFolder | Out-Null
    }

    $destination = Join-Path $targetMonthFolder $file.Name
    Move-Item -Path $file.FullName -Destination $destination
}
```
