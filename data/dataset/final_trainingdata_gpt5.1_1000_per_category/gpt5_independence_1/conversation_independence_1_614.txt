   ```ts
       throw err; // bubble up so HTTP layer returns 500
     }
   }
   ```

2. **Add retry with smaller batch size**
   - On specific transient DB errors (`ECONNRESET`, `ETIMEDOUT`), retry once with smaller batch sizes and a read-only replica:

   ```ts
   const MAX_RETRIES = 1;

     for (let attempt = 0; attempt <= MAX_RETRIES; attempt++) {
       try {
         return await streamMonthlyReport(params, { batchSize: attempt === 0 ? 2000 : 500 });
   ```

  ```ts
  const rememberMe = Boolean(req.body.rememberMe);

  const refreshTokenTtlMs = rememberMe
    ? 30 * 24 * 60 * 60 * 1000 // 30 days
    : 24 * 60 * 60 * 1000;     // 1 day

  res.cookie("refreshToken", token, {
    httpOnly: true,
    secure: true,
    sameSite: "lax",
    maxAge: refreshTokenTtlMs,
  });

  res.cookie("session", sessionId, {
    httpOnly: true,
    secure: true,
    sameSite: "lax",
    maxAge: rememberMe ? refreshTokenTtlMs : undefined,
  });
  ```
