```python
    st.subheader("Basic Info")
    st.write(f"Rows: {df.shape[0]}, Columns: {df.shape[1]}")
    st.write("Column types:")
    st.write(df.dtypes)

    st.subheader("Summary Statistics (Numeric Columns)")
    st.write(df.describe())

    numeric_cols = df.select_dtypes(include="number").columns.tolist()
    if numeric_cols:
        st.subheader("Distribution of a Numeric Column")
        selected_col = st.selectbox("Choose a column", numeric_cols)

        fig, ax = plt.subplots()
        df[selected_col].hist(ax=ax, bins=30)
        ax.set_title(f"Distribution of {selected_col}")
        st.pyplot(fig)
    else:
        st.info("No numeric columns found for plotting.")
```

```python
import streamlit as st
import pandas as pd
import matplotlib.pyplot as plt

st.set_page_config(page_title="Quick Data Profiler", layout="wide")

# Sidebar
st.sidebar.title("Quick Data Profiler")
st.sidebar.write(
    """
    A simple tool for fast exploratory analysis of CSV files.

    **Created by:** Your Name – Data Analyst  
    **Use cases:**  
    - Quick sanity checks on new datasets  
    - Exploring columns and distributions  
    - Preparing for deeper analysis in notebooks
    """
)

st.sidebar.markdown("---")
st.sidebar.write("Tips:")
st.sidebar.write("- For large files, sample before upload for best performance.")
st.sidebar.write("- Ensure your CSV has a header row.")

# Main content
st.title("Quick Data Profiler")
st.write("Upload a CSV file to get instant summary statistics and basic charts.")

uploaded_file = st.file_uploader("Upload your CSV file", type=["csv"])

if uploaded_file is not None:
    df = pd.read_csv(uploaded_file)

    n_rows, n_cols = df.shape

    if n_rows > 1_000_000:
        st.warning(
            f"Large dataset detected: {n_rows:,} rows. "
            "The app may be slow. Consider sampling the data first."
        )

    st.subheader("Preview of Data")
    st.dataframe(df.head())

    st.subheader("Basic Info")
    col1, col2 = st.columns(2)
    with col1:
        st.write(f"**Rows:** {n_rows:,}")
        st.write(f"**Columns:** {n_cols:,}")
    with col2:
        st.write("**Column types:**")
        st.write(df.dtypes)

    st.subheader("Summary Statistics (Numeric Columns)")
    if df.select_dtypes(include="number").shape[1] > 0:
        st.write(df.describe())
    else:
        st.info("No numeric columns found for summary statistics.")

    numeric_cols = df.select_dtypes(include="number").columns.tolist()
    if numeric_cols:
        st.subheader("Numeric Column Exploration")
        selected_col = st.selectbox("Choose a numeric column", numeric_cols)

        plot_type = st.radio(
            "Select plot type", ["Histogram", "Boxplot"], horizontal=True
        )

        fig, ax = plt.subplots()
        if plot_type == "Histogram":
            df[selected_col].hist(ax=ax, bins=30)
            ax.set_title(f"Distribution of {selected_col}")
        else:
            ax.boxplot(df[selected_col].dropna(), vert=True)
            ax.set_title(f"Boxplot of {selected_col}")
            ax.set_xticklabels([selected_col])

        st.pyplot(fig)
    else:
        st.info("No numeric columns found for plotting.")
else:
    st.info("Upload a CSV file using the uploader above to begin.")
```
