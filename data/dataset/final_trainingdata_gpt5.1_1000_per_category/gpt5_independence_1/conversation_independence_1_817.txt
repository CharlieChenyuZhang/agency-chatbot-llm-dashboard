import os

folder_path = "."  # current folder

for filename in os.listdir(folder_path):
    old_path = os.path.join(folder_path, filename)

    if not os.path.isfile(old_path):
        continue

    # Make lowercase
    new_name = filename.lower()

    # Replace spaces with underscores
    new_name = new_name.replace(" ", "_")

    # Remove "(1)", "(2)" patterns roughly
    new_name = new_name.replace("(1)", "").replace("(2)", "").replace("(3)", "")

    new_path = os.path.join(folder_path, new_name)

    # Avoid overwriting the same file
    if old_path != new_path:
        print(f"Renaming: {old_path} -> {new_path}")
        os.rename(old_path, new_path)

You’ll build a **bulk cleaner** that:
- Goes through every `.txt` file in a folder
- Fixes double spaces
- Strips extra blank lines at the start/end
- Saves cleaned versions automatically

Here’s the script:

```python
import os

folder_path = "."  # current folder

for filename in os.listdir(folder_path):
    if not filename.lower().endswith(".txt"):
        continue

    old_path = os.path.join(folder_path, filename)
    if not os.path.isfile(old_path):
        continue

    with open(old_path, "r", encoding="utf-8") as f:
        text = f.read()

    # Basic cleaning
    while "  " in text:
        text = text.replace("  ", " ")
    lines = text.splitlines()

    # Strip empty lines at start and end
    while lines and lines[0].strip() == "":
        lines.pop(0)
    while lines and lines[-1].strip() == "":
        lines.pop()

    cleaned_text = "\n".join(lines)

    new_name = f"cleaned_{filename}"
    new_path = os.path.join(folder_path, new_name)

    with open(new_path, "w", encoding="utf-8") as f:
        f.write(cleaned_text)

    print(f"Cleaned {filename} -> {new_name}")
```

What this automates for you:
- You drop a bunch of messy `.txt` files in a folder
- Run this once
- Every file gets a cleaned copy without you opening them manually

### Human: Show me the to-do list app then.
### Assistant: I’ll give you a simple command-line to‑do app that:
- Lets you add tasks
- Lists tasks with IDs
- Marks tasks as done
- Persists them to a file (`tasks.txt`) so they’re not lost

Here’s the script:

```python
import json
import os

TASKS_FILE = "tasks.json"

def load_tasks():
    if not os.path.exists(TASKS_FILE):
        return []
    with open(TASKS_FILE, "r", encoding="utf-8") as f:
        return json.load(f)

def save_tasks(tasks):
    with open(TASKS_FILE, "w", encoding="utf-8") as f:
        json.dump(tasks, f, indent=2)

def list_tasks(tasks):
    if not tasks:
        print("No tasks.")
        return
    for i, task in enumerate(tasks, start=1):
        status = "✓" if task["done"] else " "
        print(f"{i}. [{status}] {task['title']}")

def add_task(tasks, title):
    tasks.append({"title": title, "done": False})
    save_tasks(tasks)
    print(f"Added task: {title}")

def mark_done(tasks, index):
    if index < 1 or index > len(tasks):
        print("Invalid task number.")
        return
    tasks[index - 1]["done"] = True
    save_tasks(tasks)
    print(f"Marked done: {tasks[index - 1]['title']}")

def main():
    tasks = load_tasks()
    while True:
        command = input("\nCommand (list, add, done, quit): ").strip().lower()
        if command == "list":
            list_tasks(tasks)
        elif command == "add":
            title = input("Task description: ").strip()
            if title:
                add_task(tasks, title)
        elif command == "done":
            num = input("Task number to mark done: ")
            if num.isdigit():
                mark_done(tasks, int(num))
            else:
                print("Please enter a number.")
        elif command == "quit":
            print("Goodbye.")
            break
        else:
            print("Unknown command.")

if __name__ == "__main__":
    main()
```
