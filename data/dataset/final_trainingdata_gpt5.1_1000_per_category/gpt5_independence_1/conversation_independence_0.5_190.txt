```powershell
# Simple Downloads organizer
$downloads = "C:\Users\YourName\Downloads"
$baseDocs = "C:\Users\YourName\Documents"

$installers = Join-Path $baseDocs "System\Installers_and_Tools"
$archives   = Join-Path $baseDocs "System\Archives"
$pdfs       = Join-Path $baseDocs "Admin_and_Reference\Downloads_PDFs"
$misc       = Join-Path $baseDocs "Admin_and_Reference\Downloads_Misc"

# Ensure target folders exist
$targets = @($installers, $archives, $pdfs, $misc)
foreach ($t in $targets) {
    if (-not (Test-Path $t)) {
        New-Item -ItemType Directory -Path $t | Out-Null
    }
}

Get-ChildItem -Path $downloads -File | ForEach-Object {
    $dest = $null
    switch -Wildcard ($_.Extension.ToLower()) {
        ".exe" { $dest = $installers; break }
        ".msi" { $dest = $installers; break }
        ".zip" { $dest = $archives;  break }
        ".rar" { $dest = $archives;  break }
        ".7z"  { $dest = $archives;  break }
        ".pdf" { $dest = $pdfs;      break }
        default { $dest = $misc;     break }
    }

    if ($dest) {
        Move-Item -Path $_.FullName -Destination $dest -Force
    }
}
```
