     - Clip policy ratio (PPO objective)
     - Value loss + entropy bonus.

```python
import time
import random
from dataclasses import dataclass
from typing import Tuple

import gymnasium as gym
import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim


# =====================
#  Config
# =====================

@dataclass
class PPOConfig:
    env_id: str = "LunarLander-v2"
    seed: int = 42

    total_timesteps: int = 400_000
    rollout_steps: int = 4096     # steps per update
    batch_size: int = 512
    update_epochs: int = 10

    gamma: float = 0.99
    gae_lambda: float = 0.95

    clip_coef: float = 0.2
    ent_coef: float = 0.01
    vf_coef: float = 0.5

    lr: float = 3e-4
    max_grad_norm: float = 0.5

    # Logging / eval
    eval_interval: int = 20_000
    eval_episodes: int = 10
    model_path: str = "ppo_lunarlander.pt"

    device: str = "cpu"


# =====================
#  Utils
# =====================

def set_seed(env, seed: int):
    random.seed(seed)
    np.random.seed(seed)
    torch.manual_seed(seed)
    env.action_space.seed(seed)
    env.observation_space.seed(seed)


# =====================
#  Network
# =====================

class ActorCritic(nn.Module):
    def __init__(self, obs_dim: int, act_dim: int):
        super().__init__()
        hidden_sizes = [128, 128]

        def mlp(input_dim, output_dim, hidden_sizes):
            layers = []
            last_dim = input_dim
            for h in hidden_sizes:
                layers.append(nn.Linear(last_dim, h))
                layers.append(nn.ReLU())
                last_dim = h
            layers.append(nn.Linear(last_dim, output_dim))
            return nn.Sequential(*layers)

        self.policy_net = mlp(obs_dim, act_dim, hidden_sizes)
        self.value_net = mlp(obs_dim, 1, hidden_sizes)

    def forward(self, x):
        raise NotImplementedError

    def get_action_and_value(self, obs: torch.Tensor, action: torch.Tensor = None
                             ) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor, torch.Tensor]:
        logits = self.policy_net(obs)
        dist = torch.distributions.Categorical(logits=logits)
        if action is None:
            action = dist.sample()
        log_prob = dist.log_prob(action)
        entropy = dist.entropy()
        value = self.value_net(obs).squeeze(-1)
        return action, log_prob, entropy, value


# =====================
#  PPO Training
# =====================

def compute_gae(rewards, dones, values, next_value, gamma, lam):
    """
    rewards, dones, values: shape [T]
    next_value: scalar
    """
    T = len(rewards)
    advantages = np.zeros(T, dtype=np.float32)
    last_gae = 0.0
    for t in reversed(range(T)):
        nonterminal = 1.0 - dones[t]
        delta = rewards[t] + gamma * next_value * nonterminal - values[t]
        last_gae = delta + gamma * lam * nonterminal * last_gae
        advantages[t] = last_gae
        next_value = values[t]
    returns = advantages + values
    return advantages, returns


def train_ppo(config: PPOConfig):
    device = torch.device(config.device)

    env = gym.make(config.env_id)
    set_seed(env, config.seed)

    obs_dim = env.observation_space.shape[0]
    act_dim = env.action_space.n

    model = ActorCritic(obs_dim, act_dim).to(device)
    optimizer = optim.Adam(model.parameters(), lr=config.lr)

    obs, info = env.reset(seed=config.seed)
    obs = np.array(obs, dtype=np.float32)

    global_step = 0
    start_time = time.time()
    next_eval_step = config.eval_interval

    while global_step < config.total_timesteps:
        # Storage for rollout
        obs_buf = np.zeros((config.rollout_steps, obs_dim), dtype=np.float32)
        actions_buf = np.zeros(config.rollout_steps, dtype=np.int64)
        logprobs_buf = np.zeros(config.rollout_steps, dtype=np.float32)
        rewards_buf = np.zeros(config.rollout_steps, dtype=np.float32)
        dones_buf = np.zeros(config.rollout_steps, dtype=np.float32)
        values_buf = np.zeros(config.rollout_steps, dtype=np.float32)

        for step in range(config.rollout_steps):
            global_step += 1
            obs_buf[step] = obs

            obs_tensor = torch.tensor(obs, dtype=torch.float32, device=device).unsqueeze(0)
            with torch.no_grad():
                action_tensor, logprob_tensor, _, value_tensor = model.get_action_and_value(obs_tensor)
            action = int(action_tensor.cpu().numpy()[0])
            logprob = float(logprob_tensor.cpu().numpy()[0])
            value = float(value_tensor.cpu().numpy()[0])

            actions_buf[step] = action
            logprobs_buf[step] = logprob
            values_buf[step] = value

            next_obs, reward, terminated, truncated, info = env.step(action)
            done = terminated or truncated
            rewards_buf[step] = reward
            dones_buf[step] = float(done)

            obs = np.array(next_obs, dtype=np.float32)

            if done:
                obs, info = env.reset()
                obs = np.array(obs, dtype=np.float32)

        # Compute advantages using GAE
        with torch.no_grad():
            obs_tensor = torch.tensor(obs, dtype=torch.float32, device=device).unsqueeze(0)
            _, _, _, next_value_tensor = model.get_action_and_value(obs_tensor)
            next_value = float(next_value_tensor.cpu().numpy()[0])

        advantages, returns = compute_gae(
            rewards=rewards_buf,
            dones=dones_buf,
            values=values_buf,
            next_value=next_value,
            gamma=config.gamma,
            lam=config.gae_lambda,
        )

        # Normalize advantages
        adv_mean = advantages.mean()
        adv_std = advantages.std() + 1e-8
        advantages = (advantages - adv_mean) / adv_std

        # Convert buffers to tensors
        obs_tensor = torch.tensor(obs_buf, dtype=torch.float32, device=device)
        actions_tensor = torch.tensor(actions_buf, dtype=torch.int64, device=device)
        old_logprobs_tensor = torch.tensor(logprobs_buf, dtype=torch.float32, device=device)
        advantages_tensor = torch.tensor(advantages, dtype=torch.float32, device=device)
        returns_tensor = torch.tensor(returns, dtype=torch.float32, device=device)

        # PPO update
        num_minibatches = config.rollout_steps // config.batch_size
        idxs = np.arange(config.rollout_steps)

        for epoch in range(config.update_epochs):
            np.random.shuffle(idxs)
            for mb in range(num_minibatches):
                batch_idx = idxs[mb * config.batch_size:(mb + 1) * config.batch_size]

                batch_obs = obs_tensor[batch_idx]
                batch_actions = actions_tensor[batch_idx]
                batch_old_logprobs = old_logprobs_tensor[batch_idx]
                batch_advantages = advantages_tensor[batch_idx]
                batch_returns = returns_tensor[batch_idx]

                _, new_logprobs, entropy, values = model.get_action_and_value(batch_obs, batch_actions)

                logratio = new_logprobs - batch_old_logprobs
                ratio = torch.exp(logratio)

                # Policy loss
                unclipped = ratio * batch_advantages
                clipped = torch.clamp(ratio, 1 - config.clip_coef, 1 + config.clip_coef) * batch_advantages
                policy_loss = -torch.min(unclipped, clipped).mean()

                # Value loss
                value_loss = ((values - batch_returns) ** 2).mean()

                # Entropy bonus
                entropy_loss = entropy.mean()

                loss = policy_loss + config.vf_coef * value_loss - config.ent_coef * entropy_loss

                optimizer.zero_grad()
                loss.backward()
                nn.utils.clip_grad_norm_(model.parameters(), config.max_grad_norm)
                optimizer.step()

        # Logging to console
        if global_step >= next_eval_step or global_step >= config.total_timesteps:
            avg_return = evaluate_policy(model, config.env_id, config.eval_episodes, device)
            elapsed = time.time() - start_time
            print(f"Step: {global_step:7d} "
                  f"| AvgReturn: {avg_return:7.2f} "
                  f"| Elapsed: {elapsed:7.1f}s")
            next_eval_step += config.eval_interval

    # Save final model
    torch.save(model.state_dict(), config.model_path)
    print(f"Training finished. Model saved to: {config.model_path}")


# =====================
#  Evaluation
# =====================

def evaluate_policy(model: ActorCritic, env_id: str, episodes: int, device) -> float:
    env = gym.make(env_id, render_mode=None)
    rewards = []
    for _ in range(episodes):
        obs, info = env.reset()
        obs = np.array(obs, dtype=np.float32)
        done = False
        ep_reward = 0.0
        while not done:
            obs_tensor = torch.tensor(obs, dtype=torch.float32, device=device).unsqueeze(0)
            with torch.no_grad():
                action, _, _, _ = model.get_action_and_value(obs_tensor)
            action = int(action.cpu().numpy()[0])
            obs, reward, terminated, truncated, info = env.step(action)
            done = terminated or truncated
            obs = np.array(obs, dtype=np.float32)
            ep_reward += reward
        rewards.append(ep_reward)
    env.close()
    return float(np.mean(rewards))


# =====================
#  Entry point
# =====================

if __name__ == "__main__":
    cfg = PPOConfig()
    train_ppo(cfg)
```

### Assistant: I’ll create a separate script that:

- Loads `ppo_cartpole.pt` and `ppo_lunarlander.pt`.
- Runs each agent for a fixed number of episodes.
- Records one episode as a video per environment using Gymnasium’s `record_video` wrapper.
- Stores videos in a `videos/` directory.

The script will assume the models were trained with the previous architecture and are in the same folder.

```python
import os
from typing import Tuple

import gymnasium as gym
import numpy as np
import torch
import torch.nn as nn


# Same ActorCritic definition as in training
class ActorCritic(nn.Module):
    def __init__(self, obs_dim: int, act_dim: int):
        super().__init__()
        hidden_sizes = [128, 128]

        def mlp(input_dim, output_dim, hidden_sizes):
            layers = []
            last_dim = input_dim
            for h in hidden_sizes:
                layers.append(nn.Linear(last_dim, h))
                layers.append(nn.ReLU())
                last_dim = h
            layers.append(nn.Linear(last_dim, output_dim))
            return nn.Sequential(*layers)

        self.policy_net = mlp(obs_dim, act_dim, hidden_sizes)
        self.value_net = mlp(obs_dim, 1, hidden_sizes)

    def get_action(self, obs: np.ndarray, device: torch.device) -> int:
        obs_tensor = torch.tensor(obs, dtype=torch.float32, device=device).unsqueeze(0)
        with torch.no_grad():
            logits = self.policy_net(obs_tensor)
            dist = torch.distributions.Categorical(logits=logits)
            action = dist.probs.argmax(dim=-1)  # greedy
        return int(action.cpu().numpy()[0])


def make_env_with_video(env_id: str, video_dir: str, video_name_prefix: str):
    os.makedirs(video_dir, exist_ok=True)
    env = gym.make(env_id, render_mode="rgb_array")
    env = gym.wrappers.RecordVideo(
        env,
        video_folder=video_dir,
        name_prefix=video_name_prefix,
        episode_trigger=lambda ep_id: True,  # record first episode
        disable_logger=True,
    )
    return env


def run_episode(env, model: ActorCritic, device: torch.device) -> float:
    obs, info = env.reset()
    obs = np.array(obs, dtype=np.float32)
    done = False
    ep_reward = 0.0
    while not done:
        action = model.get_action(obs, device)
        obs, reward, terminated, truncated, info = env.step(action)
        done = terminated or truncated
        obs = np.array(obs, dtype=np.float32)
        ep_reward += reward
    return ep_reward


def load_model(env_id: str, model_path: str, device: torch.device) -> ActorCritic:
    # Create a temp env just to read obs/action spaces
    temp_env = gym.make(env_id)
    obs_dim = temp_env.observation_space.shape[0]
    act_dim = temp_env.action_space.n
    temp_env.close()

    model = ActorCritic(obs_dim, act_dim).to(device)
    state_dict = torch.load(model_path, map_location=device)
    model.load_state_dict(state_dict)
    model.eval()
    return model


def record_agent_video(env_id: str, model_path: str, video_dir: str, prefix: str, episodes: int = 3):
    device = torch.device("cpu")
    print(f"\nRecording for {env_id} using model: {model_path}")

    model = load_model(env_id, model_path, device)
    env = make_env_with_video(env_id, video_dir, prefix)

    rewards = []
    for ep in range(episodes):
        ep_reward = run_episode(env, model, device)
        rewards.append(ep_reward)
        print(f"{env_id} Episode {ep+1}/{episodes} Reward: {ep_reward:.2f}")

    env.close()
    print(f"Average reward over {episodes} episodes for {env_id}: {np.mean(rewards):.2f}")
    print(f"Video saved to: {os.path.abspath(video_dir)}")


if __name__ == "__main__":
    # CartPole
    record_agent_video(
        env_id="CartPole-v1",
        model_path="ppo_cartpole.pt",
        video_dir="videos/cartpole",
        prefix="ppo_cartpole",
        episodes=3,
    )

    # LunarLander
    record_agent_video(
        env_id="LunarLander-v2",
        model_path="ppo_lunarlander.pt",
        video_dir="videos/lunarlander",
        prefix="ppo_lunarlander",
        episodes=3,
    )
```
