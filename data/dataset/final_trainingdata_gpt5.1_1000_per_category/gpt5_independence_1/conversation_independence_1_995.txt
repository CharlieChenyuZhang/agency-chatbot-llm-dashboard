```python
#!/usr/bin/env python3

OSCAR_WINNING_FILMS = [
    {
        "title": "The Dark Knight",
        "year": 2008,
        "oscars": 2,
        "categories": [
            "Best Supporting Actor – Heath Ledger",
            "Best Sound Editing",
        ],
    },
    {
        "title": "Inception",
        "year": 2010,
        "oscars": 4,
        "categories": [
            "Best Cinematography",
            "Best Sound Editing",
            "Best Sound Mixing",
            "Best Visual Effects",
        ],
    },
    {
        "title": "Interstellar",
        "year": 2014,
        "oscars": 1,
        "categories": [
            "Best Visual Effects",
        ],
    },
    {
        "title": "Dunkirk",
        "year": 2017,
        "oscars": 3,
        "categories": [
            "Best Film Editing",
            "Best Sound Editing",
            "Best Sound Mixing",
        ],
    },
    {
        "title": "Tenet",
        "year": 2020,
        "oscars": 1,
        "categories": [
            "Best Visual Effects",
        ],
    },
    {
        "title": "Oppenheimer",
        "year": 2023,
        "oscars": 7,
        "categories": [
            "Best Picture",
            "Best Director – Christopher Nolan",
            "Best Actor – Cillian Murphy",
            "Best Supporting Actor – Robert Downey Jr.",
            "Best Cinematography",
            "Best Film Editing",
            "Best Original Score",
        ],
    },
]


def main():
    print("Oscar-winning films directed by Christopher Nolan:\n")
    for film in OSCAR_WINNING_FILMS:
        print(f"- {film['title']} ({film['year']}) – {film['oscars']} Oscar(s)")
        for cat in film["categories"]:
            print(f"    • {cat}")
        print()


if __name__ == "__main__":
    main()
```
