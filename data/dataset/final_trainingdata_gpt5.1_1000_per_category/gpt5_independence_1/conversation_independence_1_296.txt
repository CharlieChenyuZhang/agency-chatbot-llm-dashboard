```js
function sortByCustomOrder(items, order) {
    // Map each value in 'order' to its index
    const position = new Map(order.map((value, index) => [value, index]));

    // Sort based on the position; unknown items go to the end in original order
    return items
        .map((value, originalIndex) => ({
            value,
            key: position.has(value) ? position.get(value) : order.length,
            originalIndex
        }))
        .sort((a, b) => {
            if (a.key !== b.key) return a.key - b.key;
            // Preserve original relative order for items with the same key
            return a.originalIndex - b.originalIndex;
        })
        .map(entry => entry.value);
}

// Example usage:

const desiredOrder = ["B", "D", "A", "C", "E"];

const sortedItems = sortByCustomOrder(items, desiredOrder);
console.log(sortedItems); // ["B", "D", "A", "C", "E"]
```

This will:
- Enforce your exact `order`.
