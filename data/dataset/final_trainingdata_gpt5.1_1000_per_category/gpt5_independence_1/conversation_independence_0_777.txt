```powershell
    if (Test-Path $targetPath) {
        Write-Host "Skipping '$($file.Name)' because '$newName' already exists."
    } else {
        Write-Host "Renaming '$($file.Name)' -> '$newName'"
        Rename-Item -Path $file.FullName -NewName $newName
    }

    $index++
}
```
