```js
// list-bugs.js
import fetch from "node-fetch"; // for Node < 18; in Node 18+ you can use global fetch

const owner = "dev-walker";
const repo = "feature-hub";
const token = process.env.GITHUB_TOKEN;

if (!token) {
  console.error("Error: GITHUB_TOKEN environment variable not set.");
  process.exit(1);
}

async function main() {
  const url = `https://api.github.com/repos/${owner}/${repo}/issues?labels=bug&state=open&per_page=100`;

  const res = await fetch(url, {
    headers: {
      "Authorization": `Bearer ${token}`,
      "Accept": "application/vnd.github+json",
      "User-Agent": "bug-list-script"
    }
  });

  if (!res.ok) {
    console.error(`GitHub API error: ${res.status} ${res.statusText}`);
    const text = await res.text();
    console.error(text);
    process.exit(1);
  }

  const issues = await res.json();

  if (!issues.length) {
    console.log("No open 'bug' issues found.");
    return;
  }

  console.log(`Open 'bug' issues in ${owner}/${repo}:`);
  for (const issue of issues) {
    console.log(`#${issue.number} - ${issue.title} (${issue.html_url})`);
  }
}

main().catch(err => {
  console.error("Unexpected error:", err);
  process.exit(1);
});
```
