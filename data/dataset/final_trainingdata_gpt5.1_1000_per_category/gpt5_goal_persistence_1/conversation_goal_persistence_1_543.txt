### Assistant: Let’s design a very minimal, abstract GameEnv that you can implement quickly. We’ll ignore graphics and keep everything scalar.

**Game Idea (Simplified):**
- Each episode is up to `T` time steps.
- At each step:
  - Some enemies appear, depending on a difficulty parameter `difficulty` (controlled by the director).
  - The player bot takes some expected damage and gains some score, both influenced by `difficulty`.
- Player dies if health ≤ 0 → episode ends early.

**State Variables:**
- `health` (float, e.g., 0–100).
- `difficulty` (integer or float, e.g., 0–10).
- `score`.
- We’ll also track recent damage and score rate via simple moving averages.

**Director’s Action Effect:**
- Action `0`: `difficulty -= 1`
- Action `1`: `difficulty += 0`
- Action `2`: `difficulty += 1`
- Clamp `difficulty` between 0 and 10.

**Heuristic Player Model:**
- Damage taken each step: random around `base_damage * difficulty`.
- Score gained each step: random around `base_score * difficulty` (harder can give more score).

**Reward for Director (per step):**
- Target health fraction around 0.5–0.7.
- Let `h_frac = health / 100`.
- Reward `r = 1 - abs(h_frac - 0.6)` (clipped ≥ 0).
- Extra penalty if player dies (`-1`).

Here’s a compact skeleton you can adapt:

```python
import numpy as np

class GameEnv:
    def __init__(self, max_steps=100):
        self.max_steps = max_steps
        self.base_damage = 1.0
        self.base_score = 1.0
        self.reset()

    def reset(self):
        self.health = 100.0
        self.difficulty = 5  # start medium
        self.score = 0.0
        self.step_count = 0
        self.recent_damage = 0.0
        self.recent_score_rate = 0.0
        return self.get_features()

    def apply_director_action(self, action):
        if action == 0:
            self.difficulty -= 1
        elif action == 2:
            self.difficulty += 1
        # action == 1: no change

        self.difficulty = np.clip(self.difficulty, 0, 10)

    def step(self):
        self.step_count += 1

        # Simulate damage and score based on difficulty
        dmg_mean = self.base_damage * (0.5 + 0.2 * self.difficulty)
        score_mean = self.base_score * (0.5 + 0.2 * self.difficulty)

        damage = max(0.0, np.random.normal(dmg_mean, 0.5))
        gain = max(0.0, np.random.normal(score_mean, 1.0))

        self.health -= damage
        self.score += gain

        # Update simple moving averages
        alpha = 0.1
        self.recent_damage = (1 - alpha) * self.recent_damage + alpha * damage
        self.recent_score_rate = (1 - alpha) * self.recent_score_rate + alpha * gain

        done = self.health <= 0 or self.step_count >= self.max_steps

        # Director reward
        h_frac = max(0.0, self.health / 100.0)
        r = 1.0 - abs(h_frac - 0.6)
        r = max(0.0, r)
        if self.health <= 0:
            r -= 1.0  # penalty for death

        return r, done

    def get_features(self):
        # These will be bucketed later
        health = self.health
        damage = self.recent_damage
        score_rate = self.recent_score_rate
        return health, damage, score_rate
```

```python
import numpy as np

# ===== Environment =====

class GameEnv:
    def __init__(self, max_steps=50):
        self.max_steps = max_steps
        self.base_damage = 1.0
        self.base_score = 1.0

    def reset(self):
        self.health = 100.0
        self.difficulty = 5
        self.score = 0.0
        self.step_count = 0
        self.recent_damage = 0.0
        self.recent_score_rate = 0.0
        return self.get_features()

    def apply_director_action(self, action):
        if action == 0:
            self.difficulty -= 1
        elif action == 2:
            self.difficulty += 1
        self.difficulty = int(np.clip(self.difficulty, 0, 10))

    def step(self):
        self.step_count += 1

        dmg_mean = self.base_damage * (0.5 + 0.2 * self.difficulty)
        score_mean = self.base_score * (0.5 + 0.2 * self.difficulty)

        damage = max(0.0, np.random.normal(dmg_mean, 0.5))
        gain = max(0.0, np.random.normal(score_mean, 1.0))

        self.health -= damage
        self.score += gain

        alpha = 0.1
        self.recent_damage = (1 - alpha) * self.recent_damage + alpha * damage
        self.recent_score_rate = (1 - alpha) * self.recent_score_rate + alpha * gain

        done = (self.health <= 0) or (self.step_count >= self.max_steps)

        h_frac = max(0.0, self.health / 100.0)
        r = 1.0 - abs(h_frac - 0.6)
        r = max(0.0, r)
        if self.health <= 0:
            r -= 1.0

        return r, done, self.get_features()

    def get_features(self):
        return self.health, self.recent_damage, self.recent_score_rate

# ===== Bucketing & State Encoding =====

def bucket_health(h):
    if h <= 0:
        return 0
    elif h < 30:
        return 1
    elif h < 60:
        return 2
    elif h < 90:
        return 3
    else:
        return 4

def bucket_damage(d):
    if d < 1.0:
        return 0
    elif d < 3.0:
        return 1
    else:
        return 2

def bucket_score_rate(s):
    if s < 1.0:
        return 0
    elif s < 3.0:
        return 1
    else:
        return 2

def encode_state(health, damage, score_rate):
    h = bucket_health(health)
    d = bucket_damage(damage)
    s = bucket_score_rate(score_rate)
    return h * 9 + d * 3 + s  # 5 * 3 * 3 = 45 states

# ===== Q-Learning =====

n_states = 45
n_actions = 3
Q = np.zeros((n_states, n_actions))

alpha = 0.1
gamma = 0.99
epsilon_start = 1.0
epsilon_end = 0.05
epsilon_decay = 0.995

def choose_action(state, epsilon):
    if np.random.rand() < epsilon:
        return np.random.randint(n_actions)
    return np.argmax(Q[state])

env = GameEnv()
num_episodes = 5000
epsilon = epsilon_start

for ep in range(num_episodes):
    health, dmg, sr = env.reset()
    s = encode_state(health, dmg, sr)
    done = False
    total_reward = 0.0

    while not done:
        a = choose_action(s, epsilon)
        env.apply_director_action(a)
        r, done, (health, dmg, sr) = env.step()
        s_next = encode_state(health, dmg, sr)

        best_next = np.max(Q[s_next])
        Q[s, a] += alpha * (r + gamma * best_next - Q[s, a])

        s = s_next
        total_reward += r

    epsilon = max(epsilon_end, epsilon * epsilon_decay)

    if (ep + 1) % 500 == 0:
        print(f"Episode {ep+1}, total_reward={total_reward:.2f}, epsilon={epsilon:.3f}")
```
