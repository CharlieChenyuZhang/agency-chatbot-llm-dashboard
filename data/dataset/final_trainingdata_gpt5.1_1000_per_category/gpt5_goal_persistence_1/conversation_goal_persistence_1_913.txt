import csv
import math
import requests

BASE_URL = "https://app.clientdesk.local/api/customers"
PAGE_SIZE = 25

HEADERS = {
    "Authorization": "Bearer VERYLONGTOKEN...",  # paste your token here
    "Accept": "application/json",
}

def get_page(page):
    params = {
        "page": page,
        "size": PAGE_SIZE,
    }
    resp = requests.get(BASE_URL, headers=HEADERS, params=params)
    resp.raise_for_status()
    return resp.json()

def main():
    # First call to detect total pages
    first = get_page(1)

    # Adjust these according to the JSON shape you see
    # Example assumption:
    # {
    #   "content": [ {customer1}, {customer2}, ... ],
    #   "totalElements": 1234
    # }
    customers = first["content"]
    total = first["totalElements"]
    total_pages = math.ceil(total / PAGE_SIZE)
    print(f"Total customers: {total}, pages: {total_pages}")

    for page in range(2, total_pages + 1):
        print(f"Fetching page {page}/{total_pages}")
        data = get_page(page)
        customers.extend(data["content"])

    print(f"Fetched {len(customers)} customers. Writing CSV...")

    # Adjust fieldnames to match your customer objects
    fieldnames = list(customers[0].keys())
    with open("customers.csv", "w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()
        writer.writerows(customers)

    print("Done. File: customers.csv")

if __name__ == "__main__":
    main()
