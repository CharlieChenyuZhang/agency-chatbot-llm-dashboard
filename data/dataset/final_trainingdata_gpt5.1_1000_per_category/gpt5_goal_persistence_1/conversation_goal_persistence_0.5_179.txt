import base64
import json
import sys
import urllib.request

GITHUB_TOKEN = "YOUR_TOKEN_HERE"
GITHUB_USERNAME = "YOUR_USERNAME_HERE"

REPO_NAME = "NolanFans"
REPO_DESCRIPTION = "Christopher Nolan films that have won Oscars"

README_CONTENT = """# NolanFans

A fan-made list of Christopher Nolan films that have won at least one Academy Award (Oscar).

## Oscar-Winning Christopher Nolan Films

- **Oppenheimer (2023)**
  - Best Picture
  - Best Director (Christopher Nolan)
  - Best Actor (Cillian Murphy)
  - Best Supporting Actor (Robert Downey Jr.)
  - Best Cinematography
  - Best Film Editing
  - Best Original Score

- **Tenet (2020)**
  - Best Visual Effects

- **Dunkirk (2017)**
  - Best Film Editing
  - Best Sound Mixing
  - Best Sound Editing

- **Interstellar (2014)**
  - Best Visual Effects

- **Inception (2010)**
  - Best Cinematography
  - Best Sound Mixing
  - Best Sound Editing
  - Best Visual Effects

- **The Dark Knight (2008)**
  - Best Supporting Actor (Heath Ledger)
  - Best Sound Editing

> Note: Listed only for wins, not nominations.

## Contributing

- Open an issue or pull request for corrections or additional details.

## License

MIT License (or your preferred license).
"""

def github_request(url, method="GET", data=None):
    req = urllib.request.Request(url, method=method)
    req.add_header("Authorization", f"token {GITHUB_TOKEN}")
    req.add_header("Accept", "application/vnd.github+json")
    if data is not None:
        body = json.dumps(data).encode("utf-8")
        req.add_header("Content-Type", "application/json")
    else:
        body = None
    with urllib.request.urlopen(req, body) as resp:
        return resp.read().decode(), resp.getcode()

def main():
    # 1. Create repo
    print("Creating repository...")
    repo_url = "https://api.github.com/user/repos"
    repo_data = {
        "name": REPO_NAME,
        "description": REPO_DESCRIPTION,
        "private": False
    }
    try:
        body, code = github_request(repo_url, method="POST", data=repo_data)
        print(f"Repo create response code: {code}")
        print(body)
    except Exception as e:
        print("Error creating repository:", e)
        print("If the repo already exists, you can ignore this and continue.")
    
    # 2. Create README.md
    print("Creating README.md...")
    contents_url = f"https://api.github.com/repos/{GITHUB_USERNAME}/{REPO_NAME}/contents/README.md"
    encoded = base64.b64encode(README_CONTENT.encode("utf-8")).decode("utf-8")
    contents_data = {
        "message": "Add README with Oscar-winning Nolan films",
        "content": encoded,
        "branch": "main"
    }
    try:
        body, code = github_request(contents_url, method="PUT", data=contents_data)
        print(f"README create response code: {code}")
        print(body)
    except Exception as e:
        print("Error creating README:", e)
        print("Make sure the repo exists and your token has repo permissions.")

if __name__ == "__main__":
    if "YOUR_TOKEN_HERE" in GITHUB_TOKEN or "YOUR_USERNAME_HERE" in GITHUB_USERNAME:
        print("Please edit this script and set GITHUB_TOKEN and GITHUB_USERNAME before running.")
        sys.exit(1)
    main()
