**Step 2: Define a minimal traffic environment**

Create `traffic_env.py`:

```python
import numpy as np
import gymnasium as gym
from gymnasium import spaces

class SimpleTrafficEnv(gym.Env):
    metadata = {"render_modes": ["human"]}

    def __init__(self):
        super().__init__()
        # State: [queue_NS, queue_EW]
        self.observation_space = spaces.Box(
            low=0, high=20, shape=(2,), dtype=np.float32
        )
        # Action: 0 = NS green, 1 = EW green
        self.action_space = spaces.Discrete(2)

        self.max_steps = 100
        self.reset(seed=None)

    def reset(self, seed=None, options=None):
        super().reset(seed=seed)
        self.step_count = 0
        self.queue_NS = 10
        self.queue_EW = 10
        obs = np.array([self.queue_NS, self.queue_EW], dtype=np.float32)
        return obs, {}

    def step(self, action):
        self.step_count += 1

        # Random arrivals
        self.queue_NS += np.random.poisson(1.0)
        self.queue_EW += np.random.poisson(1.0)

        # Service depending on action
        if action == 0:
            self.queue_NS = max(0, self.queue_NS - 3)
        else:
            self.queue_EW = max(0, self.queue_EW - 3)

        # Reward: negative total queue length
        reward = -float(self.queue_NS + self.queue_EW)

        terminated = False
        truncated = self.step_count >= self.max_steps

        obs = np.array([self.queue_NS, self.queue_EW], dtype=np.float32)
        info = {}
        return obs, reward, terminated, truncated, info

    def render(self):
        print(f"Step {self.step_count}: NS={self.queue_NS}, EW={self.queue_EW}")
```

        action, _ = model.predict(obs, deterministic=True)
