In the same PowerShell window, try this alternative hashing script:

```powershell
$extensions = @(".pdf",".doc",".docx",".ppt",".pptx",".txt",".rtf")

$files = Get-ChildItem -Path $roots -Recurse -File |
  Where-Object { $extensions -contains $_.Extension.ToLower() }

function Get-FileHashDotNet {
    param(
        [Parameter(Mandatory=$true)]
        [string]$Path
    )

    $sha256 = [System.Security.Cryptography.SHA256]::Create()
    $stream = [System.IO.File]::OpenRead($Path)
    try {
        $hashBytes = $sha256.ComputeHash($stream)
    } finally {
        $stream.Dispose()
        $sha256.Dispose()
    }
    -join ($hashBytes | ForEach-Object { $_.ToString("x2") })
}

$hashes = $files | ForEach-Object {
    try {
        $h = Get-FileHashDotNet -Path $_.FullName
        [PSCustomObject]@{
            FullName = $_.FullName
            Hash     = $h
            Length   = $_.Length
        }
    } catch {
        Write-Warning "Failed to hash $($_.FullName): $_"
    }
}

$hashLog = Join-Path $logDir "file_hashes.csv"
$hashes | Export-Csv -Path $hashLog -NoTypeInformation -Encoding UTF8
```

```powershell
        continue
```

```powershell
        continue
```
