### Assistant: We’ll keep the logic but change *how* and *when* it runs so the UI stays responsive. A few practical options you can combine:

1. Process in chunks with `setTimeout` to avoid blocking:

```js
function filterProductsAsync(products, chunkSize = 1000) {
  return new Promise(resolve => {
    const result = [];
    let index = 0;

    function processChunk() {
      const end = Math.min(index + chunkSize, products.length);

      for (let i = index; i < end; i++) {
        const p = products[i];

        const rawPrice = p.sale_price_usd ?? p.price_usd;
        if (rawPrice == null || Number.isNaN(Number(rawPrice))) continue;
        const effectivePrice = Number(rawPrice);

        let ratingNum = null;
        if (typeof p.avg_rating === 'string') {
          if (p.avg_rating.toUpperCase() === 'N/A') continue;
          const [val] = p.avg_rating.split('/');
          ratingNum = parseFloat(val);
        } else if (typeof p.avg_rating === 'number') {
          ratingNum = p.avg_rating;
        }
        if (ratingNum == null || Number.isNaN(ratingNum)) continue;

        if (ratingNum >= 4.0 && effectivePrice < 50) {
          result.push(p);
        }
      }

      index = end;
      if (index < products.length) {
        setTimeout(processChunk, 0); // yield to UI
      } else {
        resolve(result);
      }
    }

    processChunk();
  });
}
```
