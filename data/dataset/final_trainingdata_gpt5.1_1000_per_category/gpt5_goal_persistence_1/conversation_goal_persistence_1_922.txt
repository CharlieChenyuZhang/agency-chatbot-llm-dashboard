```python
@app.route('/mark', methods=['POST'])
def mark():
    count = int(request.form.get('count', 0))
    for i in range(count):
        phrase_text = request.form.get(f'phrase_{i}')
        difficulty = request.form.get(f'diff_{i}')
        if not phrase_text or not difficulty:
            continue
        # find the phrase in the list and update difficulty
        for p in phrases:
            if p['phrase'] == phrase_text:
                p['difficulty'] = difficulty
                break
    return redirect('/practice')
```
