### Human: Yes, please show me a minimal Python script example that auto-adds approvers to any MR into `main`.

### Assistant:
Here’s a minimal example using Python and the GitLab REST API. This assumes:

- You have a **personal access token** with API scope.
- You know your **project ID**.
- You run this script periodically (e.g., via cron or a CI job).

**What it does:**

- Finds all **open MRs** targeting `main`.
- For each MR, sets a list of **approvers** to `senior1`, `senior2`, and `qa1`.

```python
import requests

GITLAB_URL = "https://gitlab.example.com"  # no trailing slash
PROJECT_ID = 1234  # your project ID
PRIVATE_TOKEN = "YOUR_ACCESS_TOKEN"

REVIEWERS = ["senior1", "senior2", "qa1"]  # usernames to add as approvers


def get_user_ids(usernames):
    """Look up GitLab user IDs by username."""
    user_ids = []
    for username in usernames:
        resp = requests.get(
            f"{GITLAB_URL}/api/v4/users",
            headers={"PRIVATE-TOKEN": PRIVATE_TOKEN},
            params={"username": username},
            timeout=10,
        )
        resp.raise_for_status()
        data = resp.json()
        if not data:
            print(f"Warning: user '{username}' not found")
            continue
        user_ids.append(data[0]["id"])
    return user_ids


def get_open_mrs_targeting_main():
    """Return open MRs that target main."""
    mrs = []
    page = 1
    while True:
        resp = requests.get(
            f"{GITLAB_URL}/api/v4/projects/{PROJECT_ID}/merge_requests",
            headers={"PRIVATE-TOKEN": PRIVATE_TOKEN},
            params={
                "state": "opened",
                "target_branch": "main",
                "per_page": 50,
                "page": page,
            },
            timeout=10,
        )
        resp.raise_for_status()
        batch = resp.json()
        if not batch:
            break
        mrs.extend(batch)
        page += 1
    return mrs


def set_approvers_for_mr(mr_iid, user_ids):
    """Set approvers on a single MR (CE 15.x uses /approvers endpoint)."""
    resp = requests.post(
        f"{GITLAB_URL}/api/v4/projects/{PROJECT_ID}/merge_requests/{mr_iid}/approvers",
        headers={"PRIVATE-TOKEN": PRIVATE_TOKEN},
        json={"approver_ids": user_ids},
        timeout=10,
    )
    if resp.status_code not in (200, 201):
        print(f"Failed to set approvers for MR !{mr_iid}: {resp.status_code}, {resp.text}")
    else:
        print(f"Set approvers for MR !{mr_iid} to {user_ids}")


def main():
    user_ids = get_user_ids(REVIEWERS)
    if not user_ids:
        print("No valid reviewers found; aborting.")
        return

    mrs = get_open_mrs_targeting_main()
    if not mrs:
        print("No open MRs targeting main.")
        return

    for mr in mrs:
        mr_iid = mr["iid"]
        print(f"Processing MR !{mr_iid} - {mr['title']}")
        set_approvers_for_mr(mr_iid, user_ids)


if __name__ == "__main__":
    main()
```

Here’s an adjusted script that:

```python
import requests

GITLAB_URL = "https://gitlab.example.com"
PROJECT_ID = 1234
PRIVATE_TOKEN = "YOUR_ACCESS_TOKEN"

REVIEWERS = ["senior1", "senior2", "qa1"]  # usernames
REVIEWER_LABELS = ["reviewer:senior1", "reviewer:senior2", "reviewer:qa1"]


def get_open_mrs_targeting_main():
    mrs = []
    page = 1
    while True:
        resp = requests.get(
            f"{GITLAB_URL}/api/v4/projects/{PROJECT_ID}/merge_requests",
            headers={"PRIVATE-TOKEN": PRIVATE_TOKEN},
            params={
                "state": "opened",
                "target_branch": "main",
                "per_page": 50,
                "page": page,
            },
            timeout=10,
        )
        resp.raise_for_status()
        batch = resp.json()
        if not batch:
            break
        mrs.extend(batch)
        page += 1
    return mrs


def post_review_request_comment(mr_iid, reviewers):
    mentions = " ".join(f"@{u}" for u in reviewers)
    body = f"Auto-requesting review from: {mentions}"
    resp = requests.post(
        f"{GITLAB_URL}/api/v4/projects/{PROJECT_ID}/merge_requests/{mr_iid}/notes",
        headers={"PRIVATE-TOKEN": PRIVATE_TOKEN},
        json={"body": body},
        timeout=10,
    )
    resp.raise_for_status()
    print(f"Posted review request comment on MR !{mr_iid}")


def add_labels_to_mr(mr_iid, labels):
    resp = requests.put(
        f"{GITLAB_URL}/api/v4/projects/{PROJECT_ID}/merge_requests/{mr_iid}",
        headers={"PRIVATE-TOKEN": PRIVATE_TOKEN},
        json={"labels": ",".join(labels)},
        timeout=10,
    )
    resp.raise_for_status()
    print(f"Set labels on MR !{mr_iid} to {labels}")


def main():
    mrs = get_open_mrs_targeting_main()
    if not mrs:
        print("No open MRs targeting main.")
        return

    for mr in mrs:
        mr_iid = mr["iid"]
        print(f"Processing MR !{mr_iid} - {mr['title']}")
        post_review_request_comment(mr_iid, REVIEWERS)
        # Merge existing labels with reviewer labels if needed:
        existing_labels = mr.get("labels", [])
        combined_labels = list(set(existing_labels + REVIEWER_LABELS))
        add_labels_to_mr(mr_iid, combined_labels)


if __name__ == "__main__":
    main()
```

- Each relevant MR gets:
  - A clear comment tagging the reviewers (they’re notified).
  - Labels that identify who the “reviewers” are, visible in MR lists and filters.
