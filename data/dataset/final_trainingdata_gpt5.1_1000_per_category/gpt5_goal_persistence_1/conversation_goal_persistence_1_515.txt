### Human: Yeah, I'd like a single command or script to fetch and create branches for all assigned MRs automatically.
### Assistant: We can wrap everything into a small shell script that:

1. Calls the API
2. Extracts `<IID> <BRANCH>`
3. Runs `git fetch` to create local branches like `mr-<IID>-<BRANCH>`.

Create a file `checkout_my_mrs.sh` with this content:

```bash
#!/bin/sh
set -eu

GITLAB_URL="${GITLAB_URL:-https://gitlab.internal}"
GITLAB_TOKEN="${GITLAB_TOKEN:?Set GITLAB_TOKEN in your environment}"
PROJECT_ID="${PROJECT_ID:-315}"
USER_ID="${USER_ID:-87}"
REMOTE_NAME="${REMOTE_NAME:-origin}"

TMP_JSON="${TMP_JSON:-/tmp/my_mrs_$$.json}"

# 1) Get opened MRs assigned to USER_ID
curl --silent --fail \
  --header "PRIVATE-TOKEN: $GITLAB_TOKEN" \
  "$GITLAB_URL/api/v4/projects/$PROJECT_ID/merge_requests?assignee_id=$USER_ID&state=opened&per_page=100" \
  -o "$TMP_JSON"

# 2) Extract "<IID> <SOURCE_BRANCH>" lines
extract_iid_branch() {
  cat "$TMP_JSON" | \
    tr '{' '\n' | \
    grep '"source_branch"' | \
    sed -n 's/.*"source_branch":[ ]*"\([^"]\+\)".*"iid":[ ]*\([0-9]\+\).*/\2 \1/p'
}

# 3) For each MR, fetch and create a local branch
extract_iid_branch | while read IID BRANCH; do
  SAFE_BRANCH=$(echo "$BRANCH" | tr '/' '-')
  LOCAL_BRANCH="mr-${IID}-${SAFE_BRANCH}"

  echo "Processing MR !${IID} (remote branch: ${BRANCH}) -> local branch: ${LOCAL_BRANCH}"

  git fetch "$REMOTE_NAME" "${BRANCH}:${LOCAL_BRANCH}" || {
    echo "Failed to fetch branch '$BRANCH' for MR !${IID}, continuing..."
    continue
  }
done

rm -f "$TMP_JSON"
```

FIRST_BRANCH=""

  --header "PRIVATE-TOKEN: $GITLAB_TOKEN" \
  "$GITLAB_URL/api/v4/projects/$PROJECT_ID/merge_requests?assignee_id=$USER_ID&state=opened&per_page=100" \
  -o "$TMP_JSON"

extract_iid_branch() {
  cat "$TMP_JSON" | \
    tr '{' '\n' | \
    grep '"source_branch"' | \
    sed -n 's/.*"source_branch":[ ]*"\([^"]\+\)".*"iid":[ ]*\([0-9]\+\).*/\2 \1/p'

# 3) For each MR, fetch and create a local branch
extract_iid_branch | while read IID BRANCH; do
  SAFE_BRANCH=$(echo "$BRANCH" | tr '/' '-')
