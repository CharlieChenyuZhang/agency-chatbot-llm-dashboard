```bash

import os, re

SHOW_NAME = "Show Name"

patterns = [
    # 1x02
    re.compile(r'(?i)(\d+)[xX](\d+)'),
    # S01E02
    re.compile(r'(?i)S(\d+)[\s_.-]*E(\d+)'),
    # Season 5 Episode 06
    re.compile(r'(?i)Season[\s_]*([0-9]+)[\s_]*Episode[\s_]*([0-9]+)'),
    # 203 or 0203 style in the middle of the name
    re.compile(r'(?<!\d)(\d{3,4})(?!\d)')
]

def find_se_ep(name):
    for pat in patterns:
        m = pat.search(name)
        if not m:
            continue
        if pat.pattern == patterns[-1].pattern:
            num = m.group(1)
            if len(num) == 3:
                s = int(num[0])
                e = int(num[1:])
            else:
                s = int(num[:-2])
                e = int(num[-2:])
            return s, e, m.span()
        else:
            s = int(m.group(1))
            e = int(m.group(2))
            return s, e, m.span()
    return None, None, None

def clean_title(raw, se_span):
    base = raw
    if se_span:
        start, end = se_span
        base = base[:start] + " " + base[end:]
    base = re.sub(r'(?i)\b(1080p|720p|480p|x264|x265|webrip|bluray|hdtv)\b', ' ', base)
    base = re.sub(r'(?i)\b' + re.escape(SHOW_NAME) + r'\b', ' ', base)
    base = base.replace('.', ' ').replace('_', ' ')
    base = re.sub(r'\s+', ' ', base).strip()
    if not base:
        return "Unknown"
    return base

for fname in os.listdir('.'):
    if os.path.isdir(fname):
        continue
    root, ext = os.path.splitext(fname)
    s, e, span = find_se_ep(root)
    if s is None or e is None:
        print(f"SKIP (no S/E): {fname}")
        continue
    title = clean_title(root, span)
    new_name = f"{SHOW_NAME} - S{s:02d}E{e:02d} - {title}{ext}"
    if new_name == fname:
        continue
    if os.path.exists(new_name):
        print(f"CONFLICT: {new_name} already exists, skipping {fname}")
        continue
    print(f"{fname} -> {new_name}")
    os.rename(fname, new_name)
PYCODE
```
