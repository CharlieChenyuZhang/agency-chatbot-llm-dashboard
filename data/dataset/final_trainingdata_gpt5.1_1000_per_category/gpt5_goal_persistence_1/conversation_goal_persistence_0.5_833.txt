```python
import pdfplumber
import re
from pathlib import Path

PDF_PATH = Path(r"C:\Bills\BigBank\statement-latest.pdf")

def extract_statement_balance(pdf_path):
    with pdf_path.open("rb") as f, pdfplumber.open(f) as pdf:
        text = ""
        for page in pdf.pages:
            text += page.extract_text() + "\n"

    # Try a few patterns that credit cards commonly use
    patterns = [
        r"Statement Balance\s*\$([0-9,]+\.\d{2})",
        r"New Balance\s*\$([0-9,]+\.\d{2})",
        r"Total Balance\s*\$([0-9,]+\.\d{2})",
    ]

    for pattern in patterns:
        match = re.search(pattern, text, re.IGNORECASE)
        if match:
            return float(match.group(1).replace(",", ""))

    return None

if __name__ == "__main__":
    balance = extract_statement_balance(PDF_PATH)
    if balance is None:
        print("Could not find statement balance in PDF.")
    else:
        print(f"Statement balance: ${balance:,.2f}")
```

3) It prints the statement balance you should pay.
