import os
import re
from datetime import date

root = r"C:\Users\Me\Documents\Reports"  # On Linux: "/home/me/Reports"

today = date.today().strftime("%Y-%m-%d")
pattern = re.compile(rf"^report_{re.escape(today)}_(\d{{3}})\.pdf$", re.IGNORECASE)

def rename_in_subfolder(folder):
    print(f"\nProcessing folder: {folder}")
    try:
        entries = list(os.scandir(folder))
    except FileNotFoundError:
        print("  Folder not found, skipping.")
        return

    files = [e for e in entries if e.is_file() and e.name.lower().endswith(".pdf")]
    if not files:
        print("  (No PDFs here, skipping)")
        return

    used = set()

    # First pass: find used numbers
    for f in files:
        m = pattern.match(f.name)
        if m:
            used.add(int(m.group(1)))

    def next_number():
        n = 1
        while n in used:
            n += 1
        used.add(n)
        return n

    # Second pass: rename messy files
    for f in files:
        if pattern.match(f.name):
            print(f"  Skipping already-correct: {f.name}")
            continue

        seq_num = next_number()
        new_name = f"report_{today}_{seq_num:03d}.pdf"
        src = os.path.join(folder, f.name)
        dst = os.path.join(folder, new_name)

        print(f"  Renaming '{f.name}' -> '{new_name}'")
        os.rename(src, dst)

def main():
    global root
    # Optional: uncomment to ask interactively
    # root = input(f"Root folder [{root}]: ") or root

    if not os.path.isdir(root):
        print(f"Root folder does not exist: {root}")
        return

    # Only immediate subfolders; make recursive if needed
    for entry in os.scandir(root):
        if entry.is_dir():
            rename_in_subfolder(entry.path)

if __name__ == "__main__":
    main()
