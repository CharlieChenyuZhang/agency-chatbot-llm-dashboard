This approach handles simple worksheets (no merged cells, etc.). Here’s a script that:

- Unzips the `.xlsx`
- Reads the main sheet
- Writes a CSV

2. In PowerShell, run:

```powershell
$xlsxPath = "C:\Temp\source.xlsx"
$csvPath  = "C:\Temp\output.csv"

# Create temp folder
$tempDir = Join-Path ([System.IO.Path]::GetDirectoryName($xlsxPath)) "tmp_xlsx"
if (Test-Path $tempDir) { Remove-Item $tempDir -Recurse -Force }
New-Item -ItemType Directory -Path $tempDir | Out-Null

# Copy and unzip
$zipPath = Join-Path $tempDir "temp.zip"
Copy-Item $xlsxPath $zipPath
Add-Type -AssemblyName System.IO.Compression.FileSystem
[System.IO.Compression.ZipFile]::ExtractToDirectory($zipPath, $tempDir)

# Paths inside the xlsx structure
$sheetPath = Join-Path $tempDir "xl\worksheets\sheet1.xml"
$sharedPath = Join-Path $tempDir "xl\sharedStrings.xml"

# Load shared strings
$sharedStrings = @{}
if (Test-Path $sharedPath) {
    [xml]$sharedXml = Get-Content $sharedPath
    $i = 0
    foreach ($si in $sharedXml.sst.si) {
        $text = ""
        if ($si.t) { $text = $si.t.'#text' }
        elseif ($si.r) { $text = ($si.r.t.'#text' -join "") }
        $sharedStrings[$i] = $text
        $i++
    }
}

# Load sheet
[xml]$sheetXml = Get-Content $sheetPath

# Helper: convert cell reference like "C3" to column index
function Get-ColIndex($ref) {
    $letters = ($ref -replace '\d','')
    $n = 0
    foreach ($ch in $letters.ToCharArray()) {
        $n = $n * 26 + ([int][char]::ToUpper($ch) - [int][char]'A' + 1)
    }
    return $n
}

# Parse rows
$rows = @()
foreach ($row in $sheetXml.worksheet.sheetData.row) {
    $maxCol = 0
    $cells = @{}
    foreach ($c in $row.c) {
        $colIndex = Get-ColIndex $c.r
        if ($colIndex -gt $maxCol) { $maxCol = $colIndex }

        $val = $c.v
        if ($c.t -eq "s") {
            $val = $sharedStrings[[int]$val]
        }
        $cells[$colIndex] = $val
    }

    $fields = @()
    for ($i = 1; $i -le $maxCol; $i++) {
        $v = ""
        if ($cells.ContainsKey($i)) { $v = [string]$cells[$i] }
        # Escape for CSV
        $v = $v -replace '"','""'
        $fields += '"' + $v + '"'
    }
    $rows += ($fields -join ",")
}

# Write CSV
$rows | Set-Content -Path $csvPath -Encoding UTF8

# Cleanup temp
Remove-Item $tempDir -Recurse -Force
```
