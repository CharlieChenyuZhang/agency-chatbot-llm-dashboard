Here’s a minimal but complete example you can adapt:

```python
import random
import collections
import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
import gym

# Hyperparameters
GAMMA = 0.99
LR = 1e-3
BATCH_SIZE = 64
BUFFER_SIZE = 10000
MIN_REPLAY_SIZE = 1000
EPS_START = 1.0
EPS_END = 0.05
EPS_DECAY = 5000    # steps over which epsilon decays
TARGET_UPDATE_FREQ = 1000

env = gym.make("CartPole-v1")

obs_size = env.observation_space.shape[0]
n_actions = env.action_space.n

class QNetwork(nn.Module):
    def __init__(self, obs_size, n_actions):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(obs_size, 128),
            nn.ReLU(),
            nn.Linear(128, 128),
            nn.ReLU(),
            nn.Linear(128, n_actions)
        )

    def forward(self, x):
        return self.net(x)

online_net = QNetwork(obs_size, n_actions)
target_net = QNetwork(obs_size, n_actions)
target_net.load_state_dict(online_net.state_dict())

optimizer = optim.Adam(online_net.parameters(), lr=LR)
replay_buffer = collections.deque(maxlen=BUFFER_SIZE)

def get_epsilon(step):
    return EPS_END + (EPS_START - EPS_END) * np.exp(-step / EPS_DECAY)

obs = env.reset()[0] if isinstance(env.reset(), tuple) else env.reset()

# Fill replay buffer with random experience
for _ in range(MIN_REPLAY_SIZE):
    action = env.action_space.sample()
    next_obs, reward, done, truncated, info = env.step(action) if len(env.step(action)) == 5 else (*env.step(action), False, {})
    replay_buffer.append((obs, action, reward, next_obs, done or truncated))
    if done or truncated:
        obs = env.reset()[0] if isinstance(env.reset(), tuple) else env.reset()
    else:
        obs = next_obs

obs = env.reset()[0] if isinstance(env.reset(), tuple) else env.reset()
episode_reward = 0
step = 0

for episode in range(500):
    done = False
    truncated = False
    obs = env.reset()[0] if isinstance(env.reset(), tuple) else env.reset()
    episode_reward = 0

    while not (done or truncated):
        step += 1
        epsilon = get_epsilon(step)

        # Epsilon-greedy action selection
        if random.random() < epsilon:
            action = env.action_space.sample()
        else:
            obs_t = torch.tensor(obs, dtype=torch.float32).unsqueeze(0)
            q_values = online_net(obs_t)
            action = int(torch.argmax(q_values, dim=1).item())

        next_obs, reward, done, truncated, info = env.step(action) if len(env.step(action)) == 5 else (*env.step(action), False, {})
        replay_buffer.append((obs, action, reward, next_obs, done or truncated))

        obs = next_obs
        episode_reward += reward

        # Sample from replay buffer
        batch = random.sample(replay_buffer, BATCH_SIZE)
        obses, actions, rewards, next_obses, dones = zip(*batch)

        obses_t = torch.tensor(obses, dtype=torch.float32)
        actions_t = torch.tensor(actions, dtype=torch.int64).unsqueeze(-1)
        rewards_t = torch.tensor(rewards, dtype=torch.float32).unsqueeze(-1)
        next_obses_t = torch.tensor(next_obses, dtype=torch.float32)
        dones_t = torch.tensor(dones, dtype=torch.float32).unsqueeze(-1)

        # Compute current Q values
        q_values = online_net(obses_t).gather(1, actions_t)

        # Compute target Q values
        with torch.no_grad():
            next_q_values = target_net(next_obses_t).max(dim=1, keepdim=True)[0]
            target_q_values = rewards_t + GAMMA * (1 - dones_t) * next_q_values

        loss = nn.functional.mse_loss(q_values, target_q_values)

        optimizer.zero_grad()
        loss.backward()
        optimizer.step()

        # Update target network periodically
        if step % TARGET_UPDATE_FREQ == 0:
            target_net.load_state_dict(online_net.state_dict())

    print(f"Episode {episode}, reward: {episode_reward}")
```
