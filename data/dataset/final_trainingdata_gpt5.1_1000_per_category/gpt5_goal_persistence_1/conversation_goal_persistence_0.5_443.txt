```python
import csv
import requests
from pathlib import Path

API_BASE = "https://inventory-prod.acme.local/api/inventory"
API_KEY = "abc123-key"
CSV_FILE = Path("updates.csv")

HEADERS = {
    "X-API-KEY": API_KEY,
    "Content-Type": "application/json"
}

def update_inventory(sku: str, qty: int, warehouse_id: int):
    payload = {
        "sku": sku,
        "qty": qty,
        "warehouse_id": warehouse_id
    }
    resp = requests.put(API_BASE, json=payload, headers=HEADERS)
    resp.raise_for_status()
    return resp.json()

def main():
    if not CSV_FILE.exists():
        print(f"CSV file not found: {CSV_FILE}")
        return

    with CSV_FILE.open(newline="", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        for row in reader:
            sku = row.get("sku")
            qty = row.get("qty")
            warehouse_id = row.get("warehouse_id")

            if not (sku and qty and warehouse_id):
                print(f"Skipping row with missing data: {row}")
                continue

            try:
                qty_int = int(qty)
                wh_int = int(warehouse_id)
            except ValueError:
                print(f"Invalid qty/warehouse_id (must be integers): {row}")
                continue

            try:
                result = update_inventory(sku, qty_int, wh_int)
                print(f"Updated {sku} (warehouse {wh_int}) to {qty_int}: {result}")
            except requests.HTTPError as e:
                print(f"Error updating {sku} (warehouse {warehouse_id}): {e} - {e.response.text}")
                # You can choose to continue or break; for now, continue:
                continue

if __name__ == "__main__":
    main()
```
