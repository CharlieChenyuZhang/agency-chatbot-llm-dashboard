Requirements: Python 3, pandas Script (save as bw_convert.py): - - - import pandas as pd import sys def to_bitwarden_csv(df): # Ensure expected columns exist (case-insensitive) rename_map = {} for col in df.columns: lc = col.strip().lower() if lc == 'site': rename_map[col] = 'name' elif lc in ('url','login_uri','website'): rename_map[col] = 'login_uri' elif lc in ('username','user'): rename_map[col] = 'login_username' elif lc in ('password','pass'): rename_map[col] = 'login_password' elif lc in ('notes','note'): rename_map[col] = 'notes' df = df.rename(columns=rename_map) # Create missing columns for c in ['folder','favorite','type','fields','login_totp']: if c not in df.columns: df[c] = '' if 'name' not in df.columns: df['name'] = df.get('login_uri','(no title)') # Required Bitwarden columns in order cols = ['folder','favorite','type','name','notes','fields', 'login_uri','login_username','login_password','login_totp'] df_out = df.reindex(columns=cols, fill_value='') # Set item type to "login" df_out['type'] = 'login' return df_out def main(infile, outfile): if infile.lower().endswith(('.xlsx','.xls')): df = pd.read_excel(infile) else: df = pd.read_csv(infile) df_out = to_bitwarden_csv(df) df_out.to_csv(outfile, index=False) if __name__ == '__main__': if len(sys.argv) != 3: print("Usage: python bw_convert.py input.(xlsx|csv) output.csv") sys.exit(1) main(sys.argv[1], sys.argv[2]) - - - Usage: - python bw_convert.py your_sheet.xlsx bitwarden_import.csv - Then in Bitwarden Web: Tools > Import Data > Format: Bitwarden (csv) > choose your file > Import.
