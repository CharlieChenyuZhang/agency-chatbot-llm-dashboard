### Human: Could you also include code snippets for the training loop and policy network?
### Assistant: Sure—minimal, runnable example using Gymnasium + PyTorch. Policy network (models.py): ```python import torch import torch.nn as nn class PolicyNet(nn.Module): def __init__(self, obs_dim, act_dim): super().__init__() self.net = nn.Sequential( nn.Linear(obs_dim, 128), nn.ReLU(), nn.Linear(128, 64), nn.ReLU(), nn.Linear(64, act_dim) ) def forward(self, x): return self.net(x) # logits ``` Training (train.py): ```python import gymnasium as gym import numpy as np import torch import torch.nn as nn import torch.optim as optim from models import PolicyNet from collections import deque import random def set_seed(seed): random.seed(seed) np.random.seed(seed) torch.manual_seed(seed) torch.cuda.manual_seed_all(seed) def compute_returns(rewards, gamma): R = 0.0 returns = [] for r in reversed(rewards): R = r + gamma * R returns.append(R) returns.reverse() returns = torch.tensor(returns, dtype=torch.float32) # normalize for variance reduction returns = (returns - returns.mean()) / (returns.std() + 1e-8) return returns def train(seed=42, gamma=0.99, lr=3e-3, episodes=2000, render=False, solve_avg=475, eval_every=50): set_seed(seed) env = gym.make("CartPole-v1") obs, _ = env.reset(seed=seed) obs_dim = env.observation_space.shape[0] act_dim = env.action_space.n policy = PolicyNet(obs_dim, act_dim) optimizer = optim.Adam(policy.parameters(), lr=lr) moving = deque(maxlen=100) best_mavg = -float("inf") best_state = None for ep in range(1, episodes + 1): logps = [] rewards = [] terminated = truncated = False obs, _ = env.reset() ep_reward = 0.0 while not (terminated or truncated): obs_t = torch.tensor(obs, dtype=torch.float32).unsqueeze(0) logits = policy(obs_t) probs = torch.distributions.Categorical(logits=logits) action = probs.sample() logps.append(probs.log_prob(action)) obs, reward, terminated, truncated, _info = env.step(action.item()) rewards.append(reward) ep_reward += reward if render: env.render() returns = compute_returns(rewards, gamma) logps = torch.stack(logps) loss = -(logps * returns).sum() optimizer.zero_grad() loss.backward() optimizer.step() moving.append(ep_reward) mavg = np.mean(moving) if mavg > best_mavg and len(moving) == moving.maxlen: best_mavg = mavg best_state = policy.state_dict() if ep % 10 == 0: print(f"Ep {ep:4d} | Reward {ep_reward:6.1f} | 100-avg {mavg:6.1f}") if len(moving) == moving.maxlen and mavg >= solve_avg: print(f"Solved! 100-episode average {mavg:.1f} at episode {ep}") break if best_state is not None: torch.save(best_state, "experiments/checkpoints/policy_cartpole.pt") env.close() if __name__ == "__main__": train() ``` Evaluation (evaluate.py): ```python import gymnasium as gym import numpy as np import torch from models import PolicyNet def evaluate(ckpt_path="experiments/checkpoints/policy_cartpole.pt", episodes=10, seed=123): env = gym.make("CartPole-v1") obs, _ = env.reset(seed=seed) obs_dim = env.observation_space.shape[0] act_dim = env.action_space.n policy = PolicyNet(obs_dim, act_dim) policy.load_state_dict(torch.load(ckpt_path, map_location="cpu")) policy.eval() rewards = [] for _ in range(episodes): obs, _ = env.reset() terminated = truncated = False ep_r = 0.0 while not (terminated or truncated): obs_t = torch.tensor(obs, dtype=torch.float32).unsqueeze(0) logits = policy(obs_t) action = torch.argmax(logits, dim=-1).item() obs, r, terminated, truncated, _ = env.step(action) ep_r += r rewards.append(ep_r) print(f"Mean reward over {episodes} episodes: {np.mean(rewards):.1f} ± {np.std(rewards):.1f}") env.close() if __name__ == "__main__": evaluate() ``` Notes: - REINFORCE can be high-variance. If learning is unstable, add a value baseline (separate critic) or subtract a running mean of returns. - For quick comparison, train PPO from Stable-Baselines3 on CartPole and plot both curves.
