### Human: Could you show me a minimal DQN implementation to build on?
### Assistant: Here’s a compact PyTorch DQN for a discrete Gymnasium env like MinAtar or CartPole. It’s not highly optimized but clear enough to extend. Dependencies: torch, gymnasium, numpy Outline: - Replay buffer (capacity, push, sample) - Q-network (MLP or small conv for MinAtar) - Training loop with target network updates and ε-greedy Example (MLP for CartPole; switch to a tiny conv for MinAtar): import random, numpy as np, torch, torch.nn as nn, torch.optim as optim, collections, gymnasium as gym class ReplayBuffer: def __init__(self, cap): self.buf = collections.deque(maxlen=cap) def push(self, s, a, r, s2, d): self.buf.append((s, a, r, s2, d)) def sample(self, bs): batch = random.sample(self.buf, bs) s, a, r, s2, d = map(np.array, zip(*batch)) return s, a, r, s2, d def __len__(self): return len(self.buf) class QNet(nn.Module): def __init__(self, obs_dim, act_dim): super().__init__() self.net = nn.Sequential( nn.Linear(obs_dim, 128), nn.ReLU(), nn.Linear(128, 128), nn.ReLU(), nn.Linear(128, act_dim)) def forward(self, x): return self.net(x) def dqn_train(env_id="CartPole-v1", steps=50000, start_steps=1000, batch_size=64, gamma=0.99, lr=1e-3, eps_start=1.0, eps_end=0.05, eps_decay=20000, target_sync=1000, buffer_cap=50000, seed=0): torch.manual_seed(seed); np.random.seed(seed); random.seed(seed) env, _ = gym.make(env_id) obs, _ = env.reset(seed=seed) obs_dim = env.observation_space.shape[0] act_dim = env.action_space.n device = torch.device("cpu") q = QNet(obs_dim, act_dim).to(device) qt = QNet(obs_dim, act_dim).to(device) qt.load_state_dict(q.state_dict()) opt = optim.Adam(q.parameters(), lr=lr) buf = ReplayBuffer(buffer_cap) def epsilon(t): return eps_end + (eps_start - eps_end) * max(0, (eps_decay - t) / eps_decay) t = 0 ep_ret, ep_len = 0.0, 0 for step in range(steps): e = epsilon(step) if random.random() < e: a = env.action_space.sample() else: with torch.no_grad(): qs = q(torch.tensor(obs, dtype=torch.float32).unsqueeze(0)).squeeze(0) a = int(torch.argmax(qs).item()) next_obs, r, term, trunc, _ = env.step(a) done = term or trunc buf.push(obs, a, r, next_obs, done) obs = next_obs ep_ret += r; ep_len += 1 if done: obs, _ = env.reset() ep_ret, ep_len = 0.0, 0 if len(buf) >= start_steps: s, a_b, r_b, s2, d = buf.sample(batch_size) s = torch.tensor(s, dtype=torch.float32) a_b = torch.tensor(a_b, dtype=torch.long) r_b = torch.tensor(r_b, dtype=torch.float32) s2 = torch.tensor(s2, dtype=torch.float32) d = torch.tensor(d, dtype=torch.float32) q_sa = q(s).gather(1, a_b.unsqueeze(1)).squeeze(1) with torch.no_grad(): max_q_next = qt(s2).max(1).values target = r_b + gamma * (1 - d) * max_q_next loss = nn.functional.mse_loss(q_sa, target) opt.zero_grad() loss.backward() opt.step() if (step + 1) % target_sync == 0: qt.load_state_dict(q.state_dict()) env.close() return q Notes: - For MinAtar, replace QNet with a tiny conv net and adjust preprocessing. - Add evaluation episodes separate from training to track progress. - Consider Double DQN (use qt for argmax on next state) for stability. If you want the MinAtar conv variant, say the game and I’ll adapt it.
