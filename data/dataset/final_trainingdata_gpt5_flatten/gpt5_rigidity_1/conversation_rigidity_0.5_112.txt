### Human: Okay, show me the minimal files for a todo app and explain.
### Assistant: Here’s a minimal version without saving. Three files: index.html <!DOCTYPE html> <html> <head> <meta charset="utf-8"> <title>Todo</title> <link rel="stylesheet" href="style.css"> </head> <body> <h1>Todo</h1> <form id="todo-form"> <input id="todo-input" placeholder="Add a task" required /> <button>Add</button> </form> <ul id="todo-list"></ul> <script src="script.js"></script> </body> </html> style.css body { font-family: sans-serif; max-width: 600px; margin: 40px auto; } form { display: flex; gap: 8px; } #todo-input { flex: 1; padding: 8px; } #todo-list { list-style: none; padding: 0; } #todo-list li { display: flex; justify-content: space-between; padding: 6px 8px; border: 1px solid #ddd; margin-top: 6px; } #todo-list li.done { text-decoration: line-through; color: #777; } button.remove { background: none; border: none; color: #c00; cursor: pointer; } script.js const form = document.getElementById('todo-form'); const input = document.getElementById('todo-input'); const list = document.getElementById('todo-list'); form.addEventListener('submit', (e) => { e.preventDefault(); const text = input.value.trim(); if (!text) return; const li = document.createElement('li'); li.textContent = text; li.addEventListener('click', () => li.classList.toggle('done')); const btn = document.createElement('button'); btn.textContent = '×'; btn.className = 'remove'; btn.addEventListener('click', (ev) => { ev.stopPropagation(); li.remove(); }); li.appendChild(btn); list.appendChild(li); input.value = ''; }); How it works: - HTML defines structure (input, list). - CSS makes it readable and shows “done” with a strike-through. - JS listens for form submission, creates list items, toggles “done” on click, and removes items with the × button. Next steps we can add: - Save tasks to local storage so they persist on refresh. - Basic input validation and nicer styles.
### Human: Can you add saving to local storage?
### Assistant: Sure. Replace script.js with this version that saves and loads: script.js const form = document.getElementById('todo-form'); const input = document.getElementById('todo-input'); const list = document.getElementById('todo-list'); function renderItem(item) { const li = document.createElement('li'); li.textContent = item.text; if (item.done) li.classList.add('done'); li.addEventListener('click', () => { li.classList.toggle('done'); save(); }); const btn = document.createElement('button'); btn.textContent = '×'; btn.className = 'remove'; btn.addEventListener('click', (e) => { e.stopPropagation(); li.remove(); save(); }); li.appendChild(btn); list.appendChild(li); } function save() { const items = [...list.children].map(li => ({ text: li.firstChild.textContent, done: li.classList.contains('done') })); localStorage.setItem('todos', JSON.stringify(items)); } function load() { const items = JSON.parse(localStorage.getItem('todos') || '[]'); list.innerHTML = ''; items.forEach(renderItem); } form.addEventListener('submit', (e) => { e.preventDefault(); const text = input.value.trim(); if (!text) return; renderItem({ text, done: false }); input.value = ''; save(); }); load(); This stores tasks in the browser so they persist across refreshes. If you later deploy, this still works, but for sharing across devices you’d use a backend or a hosted database. For this week, localStorage is enough.
### Human: This is a lot. Can you just give me a cheat sheet to memorize?
### Assistant: I can give you a concise cheat sheet, but memorization alone isn’t ideal for coding. Best combo: quick cheats + tiny daily practice. Here’s a compact sheet: HTML - Basic: <!DOCTYPE html>, <html>, <head>, <body> - Common: <h1>-<h6>, <p>, <a href>, <img src>, <ul>/<ol>/<li>, <div>, <span>, <form>, <input>, <button>, <script>, <link> CSS - Selectors: tag p, class .btn, id #main, descendant .card p - Box model: margin, border, padding, width/height - Text: color, font-size, font-family - Layout: display (block, inline, flex), justify-content, align-items, gap - Pseudo-classes: :hover, :focus JavaScript - Variables: let x = 1; const y = 'hi'; - Types: number, string, boolean, null, undefined, object, array - Arrays: const a = [1,2]; a.push(3); a.map(x => x*2) - Objects: const o = {name: 'Ana'}; o.name - Functions: function f(x){ return x+1 } or const f = x => x+1 - Conditionals: if (...) { ... } else { ... } - Loops: for (let i=0;i<3;i++) { ... } or for (const v of arr) { ... } - DOM: document.querySelector('#id'), el.addEventListener('click', fn) - Local storage: localStorage.setItem('k', 'v'); localStorage.getItem('k') Practice tip: Rebuild the todo app from scratch 2–3 times this week using the cheat sheet.
