#!/usr/bin/env python3
import argparse, hashlib, json, csv, os, shutil, subprocess, sys, time
from pathlib import Path
from collections import defaultdict

# Optional deps for near-dup
try:
    from PIL import Image
    import imagehash
    from PyPDF2 import PdfReader
    import docx
    from sklearn.feature_extraction.text import TfidfVectorizer
    from sklearn.metrics.pairwise import cosine_similarity
except Exception:
    pass

DOC_EXTS = {".txt", ".md", ".rtf", ".docx", ".pages"}
PDF_EXTS = {".pdf"}
IMG_EXTS = {".jpg",".jpeg",".png",".tif",".tiff",".bmp",".gif",".heic"}
TYPE_MAP = {
    "Docs": DOC_EXTS | PDF_EXTS,  # PDFs treated separately below in placement
    "Images": IMG_EXTS,
}

TOPICS = ["Work","Personal","Finance","Reference","Archive"]
QUARANTINE = "Duplicates_Quarantine"
INBOX = "0_Inbox"

def sha256_file(p: Path, buf=1024*1024):
    h = hashlib.sha256()
    with p.open("rb") as f:
        while True:
            chunk = f.read(buf)
            if not chunk: break
            h.update(chunk)
    return h.hexdigest()

def canonical_choice(paths):
    # Prefer shortest path depth, then newest mtime
    paths = sorted(paths, key=lambda p: (len(p.parts), -p.stat().st_mtime))
    return paths[0]

def ensure_dir(p: Path):
    p.mkdir(parents=True, exist_ok=True)

def create_alias(src: Path, alias_path: Path):
    # Uses AppleScript to create Finder alias
    script = f'''
    tell application "Finder"
        set targetFile to POSIX file "{src.as_posix()}" as alias
        set aliasFile to make new alias file to targetFile at POSIX file "{alias_path.parent.as_posix()}"
        set name of aliasFile to "{alias_path.name}"
    end tell
    '''
    subprocess.run(["osascript","-e",script], check=False)

def categorize(p: Path):
    ext = p.suffix.lower()
    # Simple heuristic: keep Finance under Finance by keywords
    lname = p.name.lower()
    if any(k in lname for k in ["tax","statement","invoice","receipt","bank"]):
        topic = "Finance"
    elif any(k in lname for k in ["resume","cv","personal","family"]):
        topic = "Personal"
    elif any(k in lname for k in ["manual","guide","whitepaper","paper","research"]):
        topic = "Reference"
    elif any(k in lname for k in ["archive","old","backup"]):
        topic = "Archive"
    else:
        topic = "Work"
    # Type subfolder
    if ext in PDF_EXTS:
        t = "PDFs"
    elif ext in DOC_EXTS:
        t = "Docs"
    elif ext in IMG_EXTS:
        t = "Images"
    else:
        t = "Docs"
    return topic, t

def build_target(root: Path, p: Path):
    topic, t = categorize(p)
    # Keep originals’ base name; we’ll handle collisions
    return root / topic / t / p.name

def text_from_file(p: Path):
    ext = p.suffix.lower()
    try:
        if ext in PDF_EXTS:
            text = []
            reader = PdfReader(str(p))
            for page in reader.pages:
                try:
                    text.append(page.extract_text() or "")
                except Exception:
                    pass
            return "\n".join(text)
        if ext == ".docx":
            d = docx.Document(str(p))
            return "\n".join([para.text for para in d.paragraphs])
        if ext in {".txt",".md",".rtf"}:
            return p.read_text(errors="ignore")
    except Exception:
        return ""
    return ""

def image_phash(p: Path):
    try:
        with Image.open(p) as im:
            return str(imagehash.phash(im))
    except Exception:
        return None

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--root", required=True)
    ap.add_argument("--apply", type=lambda x: x.lower()=="true", default=False)
    ap.add_argument("--near", type=lambda x: x.lower()=="true", default=False)
    ap.add_argument("--near-threshold", type=float, default=0.9)
    args = ap.parse_args()

    root = Path(os.path.expanduser(args.root)).resolve()
    qdir = root / QUARANTINE
    ensure_dir(qdir)

    files = [p for p in root.rglob("*") if p.is_file() and QUARANTINE not in p.parts and ".DS_Store" not in p.name]
    # Phase 1: exact dedup
    by_hash = defaultdict(list)
    for p in files:
        try:
            h = sha256_file(p)
            by_hash[h].append(p)
        except Exception:
            pass

    manifest_rows = []
    moves = []

    for h, group in by_hash.items():
        if len(group) < 2: 
            continue
        canon = canonical_choice(group)
        for dup in group:
            if dup == canon: 
                continue
            # plan: move dup to quarantine, create alias in old location pointing to canon
            qpath = qdir / dup.relative_to(root)
            moves.append(("exact-dup", dup, qpath, canon))
            manifest_rows.append({"type":"exact","hash":h,"duplicate":str(dup),"canonical":str(canon),"quarantine":str(qpath)})

    # Phase 2: near-dup (lightweight, pairwise within buckets)
    near_pairs = []
    if args.near:
        text_candidates = [p for p in files if p.suffix.lower() in (DOC_EXTS | PDF_EXTS)]
        # Make small buckets by filename stems to avoid O(n^2) blowups
        buckets = defaultdict(list)
        for p in text_candidates:
            buckets[p.stem.lower()[:20]].append(p)
        for _, group in buckets.items():
            if len(group) < 2: 
                continue
            corpus = []
            idxmap = []
            for p in group:
                corpus.append(text_from_file(p))
                idxmap.append(p)
            try:
                vec = TfidfVectorizer(stop_words="english").fit_transform(corpus)
                sim = cosine_similarity(vec)
                n = len(group)
                for i in range(n):
                    for j in range(i+1,n):
                        if sim[i,j] >= args.near_threshold:
                            a, b = idxmap[i], idxmap[j]
                            # choose newer as canonical
                            canon = a if a.stat().st_mtime >= b.stat().st_mtime else b
                            dup = b if canon == a else a
                            qpath = qdir / dup.relative_to(root)
                            near_pairs.append((dup, qpath, canon, float(sim[i,j])))
                            manifest_rows.append({"type":"near","similarity":round(float(sim[i,j]),3),"duplicate":str(dup),"canonical":str(canon),"quarantine":str(qpath)})
            except Exception:
                continue
        # Images
        img_candidates = [p for p in files if p.suffix.lower() in IMG_EXTS]
        phashes = []
        for p in img_candidates:
            ph = image_phash(p)
            if ph: phashes.append((p, ph))
        # Compare within small name buckets
        ibuckets = defaultdict(list)
        for p, ph in phashes:
            ibuckets[p.stem.lower()[:20]].append((p, ph))
        for _, group in ibuckets.items():
            for i in range(len(group)):
                for j in range(i+1, len(group)):
                    p1, h1 = group[i]
                    p2, h2 = group[j]
                    # Hamming distance threshold; map to ~similarity
                    dist = imagehash.hex_to_hash(h1) - imagehash.hex_to_hash(h2)
                    # interpret <= 8 as similar; map to similarity
                    if dist <= 8:
                        sim = max(0.0, 1.0 - dist/16.0)
                        canon = p1 if p1.stat().st_mtime >= p2.stat().st_mtime else p2
                        dup = p2 if canon == p1 else p1
                        qpath = qdir / dup.relative_to(root)
                        near_pairs.append((dup, qpath, canon, sim))
                        manifest_rows.append({"type":"near-image","similarity":round(sim,3),"duplicate":str(dup),"canonical":str(canon),"quarantine":str(qpath)})

    # Apply or show plan
    print(f"Planned exact moves: {len(moves)}")
    print(f"Planned near moves: {len(near_pairs)}")
    manifest = root / "dedupe_manifest.csv"
    with manifest.open("w", newline="") as f:
        w = csv.DictWriter(f, fieldnames=["type","hash","similarity","duplicate","canonical","quarantine"])
        w.writeheader()
        for r in manifest_rows:
            w.writerow(r)

    if not args.apply:
        print("Dry-run complete. No changes made.")
        print(f"Manifest written to: {manifest}")
        sys.exit(0)

    # Apply changes
    for kind, dup, qpath, canon in moves:
        ensure_dir(qpath.parent)
        if dup.exists():
            shutil.move(str(dup), str(qpath))
            # Create alias where dup used to be
            create_alias(canon, dup.with_suffix(dup.suffix + ".alias"))

    for dup, qpath, canon, sim in near_pairs:
        ensure_dir(qpath.parent)
        if dup.exists():
            shutil.move(str(dup), str(qpath))
            # no alias for near-dup

    print(f"Applied moves. Quarantine at: {qdir}")
    print(f"Review manifest: {manifest}")
    print("Reminder: review and clean Quarantine within 30 days.")

if __name__ == "__main__":
    main()
