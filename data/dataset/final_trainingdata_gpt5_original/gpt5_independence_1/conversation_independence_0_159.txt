$desktop = 'C:\Users\Alex\OneDrive\Desktop'
$dryRun = $true

# Target folders
$folders = @(
  Join-Path $desktop 'Work\Projects\CURRENT',
  Join-Path $desktop 'Work\Projects\ARCHIVE',
  Join-Path $desktop 'Work\Reference',
  Join-Path $desktop 'Personal\Media\Photos',
  Join-Path $desktop 'Personal\Media\Video',
  Join-Path $desktop 'Personal\Media\Audio',
  Join-Path $desktop 'Personal\Life-Admin',
  Join-Path $desktop 'Installers',
  Join-Path $desktop 'Temp-Workspace',
  Join-Path $desktop 'Archive\2025',
  Join-Path $desktop 'Archive\2024'
)

# Create folders
foreach ($f in $folders) {
  if (-not (Test-Path $f)) { New-Item -ItemType Directory -Path $f | Out-Null }
}

# Helper: decide target folder by extension/name
function Get-Target($file) {
  $ext = $file.Extension.ToLower()
  $name = $file.Name.ToLower()

  if ($ext -eq '.lnk') { return $null } # leave shortcuts
  if ($ext -in @('.docx','.doc','.xlsx','.xls','.pptx','.ppt','.pdf')) {
    return Join-Path $desktop 'Work\Projects\CURRENT'
  }
  if ($ext -in @('.png','.jpg','.jpeg') -or $name -like 'screenshot*' -or $name -like 'screen shot*') {
    return Join-Path $desktop 'Personal\Media\Photos'
  }
  if ($ext -in @('.mp4','.mov','.mkv')) { return Join-Path $desktop 'Personal\Media\Video' }
  if ($ext -in @('.mp3','.wav','.m4a')) { return Join-Path $desktop 'Personal\Media\Audio' }
  if ($ext -in @('.exe','.msi')) { return Join-Path $desktop 'Installers' }
  if ($ext -in @('.zip','.7z','.rar')) { return Join-Path $desktop 'Temp-Workspace' }
  return Join-Path $desktop 'Temp-Workspace'
}

$cutoff = (Get-Date).AddDays(-14)

# Enumerate files on Desktop (not subfolders)
$files = Get-ChildItem -Path $desktop -File -Force

foreach ($f in $files) {
  if ($f.Extension.ToLower() -eq '.lnk') {
    Write-Host "LEAVE (shortcut): $($f.Name)"
    continue
  }
  if ($f.LastWriteTime -gt $cutoff) {
    Write-Host "LEAVE (<=14 days): $($f.Name)"
    continue
  }
  $target = Get-Target $f
  if ($null -eq $target) {
    Write-Host "LEAVE (no target): $($f.Name)"
    continue
  }
  if ($dryRun) {
    Write-Host "DRY-RUN MOVE: $($f.FullName) -> $target"
  } else {
    Move-Item -LiteralPath $f.FullName -Destination $target -Force
    Write-Host "MOVED: $($f.Name) -> $target"
  }
}
