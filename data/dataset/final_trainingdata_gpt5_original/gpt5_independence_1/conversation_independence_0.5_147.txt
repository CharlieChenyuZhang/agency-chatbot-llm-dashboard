### Assistant: Prisma schema (schema.prisma)
generator client { provider = "prisma-client-js" }
datasource db { provider = "postgresql"; url = env("DATABASE_URL") }

model Trainer {
  id               String   @id @default(cuid())
  userId           String   @unique
  handle           String   @unique
  email            String   @unique
  stripeAccountId  String?
  products         Product[]
  availability     Availability[]
  sessions         Session[]
  createdAt        DateTime @default(now())
}

model Product {
  id          String   @id @default(cuid())
  trainer     Trainer  @relation(fields: [trainerId], references: [id])
  trainerId   String
  name        String
  durationMins Int
  priceCents  Int
  currency    String   @default("usd")
  createdAt   DateTime @default(now())
}

model Availability {
  id            String   @id @default(cuid())
  trainer       Trainer  @relation(fields: [trainerId], references: [id])
  trainerId     String
  weekday       Int      // 0=Sun ... 6=Sat
  startMinutes  Int      // minutes from 00:00
  endMinutes    Int
  bufferMinutes Int      @default(15)
}

model Client {
  id        String    @id @default(cuid())
  email     String    @unique
  name      String?
  sessions  Session[]
  createdAt DateTime  @default(now())
}

model Session {
  id               String    @id @default(cuid())
  trainer          Trainer   @relation(fields: [trainerId], references: [id])
  trainerId        String
  client           Client    @relation(fields: [clientId], references: [id])
  clientId         String
  product          Product   @relation(fields: [productId], references: [id])
  productId        String
  start            DateTime
  end              DateTime
  status           SessionStatus @default(BOOKED)
  paymentIntentId  String?
  createdAt        DateTime  @default(now())

  @@index([trainerId, start])
}

enum SessionStatus {
  BOOKED
  CANCELED
  REFUNDED
}

Simple seed (scripts/seed.ts)
import { PrismaClient } from "@prisma/client";
const prisma = new PrismaClient();

async function main() {
  const trainer = await prisma.trainer.upsert({
    where: { email: "alex@slotsage.test" },
    update: {},
    create: {
      userId: "user_alex",
      handle: "alexfit",
      email: "alex@slotsage.test",
    },
  });

  const product = await prisma.product.upsert({
    where: { id: "seed_product" },
    update: {},
    create: {
      id: "seed_product",
      trainerId: trainer.id,
      name: "1:1 Coaching",
      durationMins: 60,
      priceCents: 5000,
      currency: "usd",
    },
  });

  // Mon, Wed, Fri 9:00–12:00
  const weekdays = [1, 3, 5];
  await Promise.all(
    weekdays.map((d) =>
      prisma.availability.create({
        data: {
          trainerId: trainer.id,
          weekday: d,
          startMinutes: 9 * 60,
          endMinutes: 12 * 60,
          bufferMinutes: 15,
        },
      })
    )
  );

  console.log("Seeded trainer:", trainer.handle, "product:", product.name);
}

main()
  .catch((e) => { console.error(e); process.exit(1); })
  .finally(async () => { await prisma.$disconnect(); });
