import argparse, json, os, shutil, sys, time, hashlib
from pathlib import Path

EXCLUDES = ['.git', 'node_modules']
PACKAGE_EXTS = ['.photoslibrary']

DOC_EXTS = {'.pdf','.doc','.docx','.txt','.rtf','.md','.xls','.xlsx','.ppt','.pptx'}
MEDIA_EXTS = {'.jpg','.jpeg','.png','.gif','.heic','.mov','.mp4','.mp3','.wav','.aiff'}
CODE_EXTS = {'.py','.js','.ts','.java','.c','.cpp','.rb','.go','.rs','.sh'}

def is_excluded(p: Path) -> bool:
    if any(part in EXCLUDES for part in p.parts):
        return True
    if any(p.name.endswith(ext) for ext in PACKAGE_EXTS):
        return True
    return False

def file_hash(p: Path, block=1024*1024):
    h = hashlib.sha256()
    with p.open('rb') as f:
        while True:
            chunk = f.read(block)
            if not chunk: break
            h.update(chunk)
    return h.hexdigest()

def classify_target(root: Path, p: Path) -> Path:
    # Keep files in existing Projects if path includes a recognized project folder name under Workspace
    # Otherwise classify by type
    ext = p.suffix.lower()
    if any(part.lower()=='projects' for part in p.parts):
        # Preserve project structure under Workspace/Projects
        # Keep relative segments after 'Projects'
        try:
            idx = [s.lower() for s in p.parts].index('projects')
            return root / Path(*p.parts[idx:])
        except ValueError:
            pass
    if ext in DOC_EXTS:
        return root / 'Documents' / p.name
    if ext in MEDIA_EXTS:
        if ext in {'.jpg','.jpeg','.png','.gif','.heic'}:
            return root / 'Media' / 'Photos' / p.name
        if ext in {'.mov','.mp4'}:
            return root / 'Media' / 'Videos' / p.name
        return root / 'Media' / p.name
    if ext in CODE_EXTS:
        return root / 'Projects' / p.stem / p.name
    return root / 'Documents' / p.name

def is_conflicted_name(name: str) -> bool:
    lowered = name.lower()
    return ('(conflicted copy' in lowered) or name.endswith(' (1)'+Path(name).suffix)

def newer(a: Path, b: Path) -> Path:
    return a if a.stat().st_mtime >= b.stat().st_mtime else b

def safe_rel(src: Path, base: Path) -> Path:
    try:
        return src.relative_to(base)
    except Exception:
        return Path(src.name)

def collect_files(sources):
    for src in sources:
        for root, dirs, files in os.walk(src):
            root_p = Path(root)
            # modify dirs in-place to skip excluded directories
            dirs[:] = [d for d in dirs if not is_excluded(root_p / d)]
            for f in files:
                p = root_p / f
                if is_excluded(p): continue
                if p.is_symlink(): continue
                yield p

def build_plan(workspace: Path, sources):
    plan = {'moves':[], 'symlinks':[], 'archive':[], 'skip':[], 'report_only':[], 'duplicates':{}, 'canonical':{}}
    hash_index = {}
    all_files = list(collect_files(sources))
    for p in all_files:
        # Skip anything inside a git repo by reporting only
        if any('.git' in part for part in p.parts):
            plan['report_only'].append(str(p))
            continue
        try:
            h = file_hash(p)
        except Exception as e:
            plan['skip'].append({'path':str(p),'reason':str(e)})
            continue
        info = {'path':str(p), 'mtime':p.stat().st_mtime}
        if h not in hash_index:
            hash_index[h] = [info]
        else:
            hash_index[h].append(info)

    # Choose canonical files and mark duplicates
    for h, items in hash_index.items():
        items_sorted = sorted(items, key=lambda x: x['mtime'])
        canonical = Path(items_sorted[-1]['path'])  # newest
        plan['canonical'][h] = str(canonical)
        dups = [Path(i['path']) for i in items_sorted if i['path'] != str(canonical)]
        plan['duplicates'][h] = [str(p) for p in dups]

    # For every file (canonical or not), decide target location
    for h, canon_path in plan['canonical'].items():
        cpath = Path(canon_path)
        target = classify_target(workspace, cpath)
        if target != cpath:
            plan['moves'].append({'src':str(cpath), 'dst':str(target)})
        for dp in plan['duplicates'][h]:
            dpp = Path(dp)
            # replace with symlink to canonical target location
            plan['symlinks'].append({'replace':str(dpp), 'to':str(target)})

    # Name-conflicted copies: we already treat by hash; we can nudge archive for obviously conflicted names if not dup by hash
    for p in all_files:
        if is_conflicted_name(p.name):
            # If not already in duplicates, suggest archive move
            is_dup = any(str(p) in dlist for dlist in plan['duplicates'].values())
            if not is_dup:
                arch = workspace / 'Archive' / 'Old-versions' / p.name
                plan['archive'].append({'src':str(p), 'dst':str(arch)})

    return plan

def apply_plan(plan, dry_run=True):
    created = []
    for move in plan['moves']:
        dst = Path(move['dst']); src = Path(move['src'])
        if dry_run: continue
        dst.parent.mkdir(parents=True, exist_ok=True)
        if dst.exists():
            # If same file by hash, skip; else rename
            if file_hash(dst)==file_hash(src):
                pass
            else:
                dst = dst.with_name(dst.stem + ' (merged)' + dst.suffix)
        shutil.move(str(src), str(dst))
    for link in plan['symlinks']:
        rep = Path(link['replace']); to = Path(link['to'])
        if dry_run: continue
        # remove original duplicate and place a symlink
        try:
            if rep.exists():
                rep.unlink()
        except: pass
        rep.parent.mkdir(parents=True, exist_ok=True)
        os.symlink(str(to), str(rep))
    for arch in plan['archive']:
        src = Path(arch['src']); dst = Path(arch['dst'])
        if dry_run: continue
        dst.parent.mkdir(parents=True, exist_ok=True)
        if src.exists():
            shutil.move(str(src), str(dst))
    return True

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--workspace', required=True)
    ap.add_argument('sources', nargs='+')
    ap.add_argument('--commit', action='store_true', help='Apply changes; default is dry-run')
    ap.add_argument('--plan', default='plan.json')
    args = ap.parse_args()

    workspace = Path(os.path.expanduser(args.workspace))
    sources = [Path(os.path.expanduser(s)) for s in args.sources]

    plan = build_plan(workspace, sources)
    with open(args.plan,'w') as f:
        json.dump(plan, f, indent=2)
    print(f'Wrote plan to {args.plan}. Moves: {len(plan["moves"])}, symlinks: {len(plan["symlinks"])}, archive: {len(plan["archive"])}, report-only: {len(plan["report_only"])}')

    if args.commit:
        confirm = input('This will apply the plan. Type APPLY to continue: ')
        if confirm.strip()!='APPLY':
            print('Aborted.')
            sys.exit(1)
        apply_plan(plan, dry_run=False)
        print('Applied.')
    else:
        print('Dry-run only. No changes applied.')

if __name__ == '__main__':
    main()
