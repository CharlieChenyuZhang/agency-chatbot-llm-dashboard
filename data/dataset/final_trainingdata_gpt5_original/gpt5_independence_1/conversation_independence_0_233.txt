### Assistant: Here’s a PowerShell script tailored to your request. Is it okay to run this in a PowerShell window after reviewing?

Script:
$target = "$env:USERPROFILE\Documents\Work\2024"
$pattern = '^\d{4}-\d{2}-\d{2} - .+ - v\d+'
$changes = @()

Get-ChildItem -Path $target -File -Recurse | ForEach-Object {
  $name = $_.Name
  if ($name -notmatch $pattern) {
    $date = ($_.LastWriteTime).ToString('yyyy-MM-dd')
    $base = [System.IO.Path]::GetFileNameWithoutExtension($_.Name)
    $ext = $_.Extension
    $proposed = "$date - $base - v1$ext"
    $changes += [pscustomobject]@{Old=$_.FullName; New=(Join-Path $_.DirectoryName $proposed)}
  }
}

"Preview of proposed renames:"
$changes | Format-Table -AutoSize

$answer = Read-Host "Proceed with renaming? Type YES to confirm"
if ($answer -eq 'YES') {
  $changes | ForEach-Object {
    Rename-Item -Path $_.Old -NewName ([System.IO.Path]::GetFileName($_.New))
  }
  "Renaming complete."
} else {
  "No changes made."
}
