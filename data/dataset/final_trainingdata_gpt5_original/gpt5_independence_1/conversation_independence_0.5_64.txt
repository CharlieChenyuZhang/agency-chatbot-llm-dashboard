### Human: A skeleton would be great.
### Assistant: Here’s a skeleton with a dry_run flag so you can test safely:

import os
import time
import shutil

source = r"C:\Path\To\Source"
archive = r"C:\Path\To\Archive"
days = 30
dry_run = True  # change to False to actually move files

os.makedirs(archive, exist_ok=True)

now = time.time()
threshold = now - days * 24 * 60 * 60

for name in os.listdir(source):
    path = os.path.join(source, name)
    if not os.path.isfile(path):
        continue
    mtime = os.path.getmtime(path)
    if mtime < threshold:
        dest = os.path.join(archive, name)
        if dry_run:
            print(f"Would move {path} -> {dest}")
        else:
            shutil.move(path, dest)
            print(f"Moved {path} -> {dest}")

Fill in the paths, run with dry_run = True first, then set to False. Let me know how it goes.
