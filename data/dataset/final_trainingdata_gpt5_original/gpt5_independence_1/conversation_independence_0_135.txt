PowerShell script (save as Refactor-And-Dedupe.ps1):

param(
  [Parameter(Mandatory=$true)][string]$RootPath,
  [string]$QuarantinePath = "D:\Quarantine",
  [int]$RetentionDays = 45,
  [int64]$MinFreeSpaceGB = 20,
  [string[]]$ImageExtensions = @(".jpg",".jpeg",".png",".heic"),
  [switch]$DryRun,
  [string]$ReportDir = "",
  [string[]]$ExcludeDirNames = @("node_modules",".git"),
  [string[]]$ExcludeExtensions = @(".tmp",".bak",".iso")
)

# Safety: confirm long path support
Set-ItemProperty -Path "HKLM:\SYSTEM\CurrentControlSet\Control\FileSystem" -Name "LongPathsEnabled" -Value 1 -Force | Out-Null

# Resolve report directory
if (-not $ReportDir -or $ReportDir -eq "") {
  $ReportDir = Join-Path $RootPath "_refactor_reports"
}
New-Item -ItemType Directory -Path $ReportDir -Force | Out-Null

$logPath = Join-Path $ReportDir ("run_" + (Get-Date -Format "yyyyMMdd_HHmmss") + ".log")
$manifestCsv = Join-Path $ReportDir "manifest.csv"
$dupeJson = Join-Path $ReportDir "duplicates.json"
$nearJson = Join-Path $ReportDir "near_duplicates.json"

# Logging helper
function Log($msg) {
  $line = ("[{0}] {1}" -f (Get-Date -Format "u"), $msg)
  $line | Tee-Object -FilePath $logPath -Append
}

# Check quarantine space
function Get-FreeSpaceGB([string]$path) {
  $drive = Get-PSDrive -PSProvider FileSystem | Where-Object { $path.ToUpper().StartsWith($_.Root.ToUpper()) } | Select-Object -First 1
  if ($drive) { [math]::Floor($drive.Free/1GB) } else { 0 }
}

# Hashing helper
function Get-FileHashSha256([string]$p) {
  try { return (Get-FileHash -Algorithm SHA256 -LiteralPath $p).Hash } catch { return $null }
}

# Filters
function Should-Exclude([System.IO.FileInfo]$fi) {
  if ($ExcludeExtensions -contains $fi.Extension.ToLower()) { return $true }
  if ($fi.Attributes -band [IO.FileAttributes]::System) { return $true }
  if ($fi.Attributes -band [IO.FileAttributes]::Hidden) { return $true }
  foreach ($name in $ExcludeDirNames) {
    if ($fi.FullName.ToLower().Split('\') -contains $name.ToLower()) { return $true }
  }
  return $false
}

# Collect files
Log "Scanning files under $RootPath"
$files = Get-ChildItem -LiteralPath $RootPath -Recurse -File -Force -ErrorAction SilentlyContinue
$files = $files | Where-Object { -not (Should-Exclude $_) }

# Build hash map
$byHash = @{}
$records = @()
foreach ($f in $files) {
  $hash = Get-FileHashSha256 $f.FullName
  if (-not $hash) { continue }
  $rec = [PSCustomObject]@{
    Path = $f.FullName
    Name = $f.Name
    Dir = $f.DirectoryName
    Size = $f.Length
    Ext = $f.Extension.ToLower()
    Modified = $f.LastWriteTimeUtc
    Accessed = $f.LastAccessTimeUtc
    Hash = $hash
  }
  $records += $rec
  if (-not $byHash.ContainsKey($hash)) { $byHash[$hash] = @() }
  $byHash[$hash] += $rec
}

# Near-duplicate heuristic: similar basenames ignoring version patterns
function Normalize-Base([string]$name) {
  $base = [System.IO.Path]::GetFileNameWithoutExtension($name)
  $base = ($base -replace '[_\-\.\s]v\d+','')
  $base = ($base -replace '\(\d+\)','')
  return $base.ToLower()
}
$byNorm = $records | Group-Object { Normalize-Base $_.Name } | Where-Object { $_.Count -gt 1 }

# Decide keeper per hash group
$dupeReport = @()
$manifest = @()
foreach ($pair in $byHash.GetEnumerator()) {
  $group = $pair.Value
  if ($group.Count -le 1) { continue }

  $isImage = $false
  if ($group | Where-Object { $ImageExtensions -contains $_.Ext }) { $isImage = $true }

  if ($isImage) {
    $keeper = $group | Sort-Object @{Expression="Accessed";Descending=$true}, @{Expression="Modified";Descending=$true} | Select-Object -First 1
  } else {
    $keeper = $group | Sort-Object @{Expression="Modified";Descending=$true} | Select-Object -First 1
  }

  $dupes = $group | Where-Object { $_.Path -ne $keeper.Path }
  foreach ($d in $dupes) {
    $dupeReport += [PSCustomObject]@{
      Hash = $d.Hash
      KeepPath = $keeper.Path
      DupePath = $d.Path
      Reason = if ($isImage) { "Image: most-recently accessed kept" } else { "Newest modified kept" }
    }
  }
}

# Write reports
$dupeReport | ConvertTo-Json -Depth 4 | Out-File -Encoding UTF8 $dupeJson
$near = @()
foreach ($g in $byNorm) {
  $near += [PSCustomObject]@{
    Key = $g.Name
    Items = $g.Group | Select-Object Path,Modified,Size
  }
}
$near | ConvertTo-Json -Depth 4 | Out-File -Encoding UTF8 $nearJson

# Quarantine or dry-run
if ($dupeReport.Count -gt 0) {
  Log ("Found {0} duplicate files" -f $dupeReport.Count)
} else {
  Log "No duplicates found"
}

# Ensure quarantine path
New-Item -ItemType Directory -Path $QuarantinePath -Force | Out-Null

# Manifest CSV header
"Action,Source,Destination,Hash,Reason" | Out-File -Encoding UTF8 $manifestCsv

foreach ($row in $dupeReport) {
  $src = $row.DupePath
  $hash = $row.Hash
  $reason = $row.Reason
  $rel = $src.Replace($RootPath, "").TrimStart('\')
  $dest = Join-Path $QuarantinePath $rel
  $destDir = Split-Path $dest -Parent
  $free = Get-FreeSpaceGB $QuarantinePath

  if ($free -lt $MinFreeSpaceGB) {
    Log "Skipping move due to low free space on quarantine drive (< $MinFreeSpaceGB GB): $src"
    "SkipLowSpace,$src,,$hash,$reason" | Out-File -FilePath $manifestCsv -Append -Encoding UTF8
    continue
  }

  "Quarantine,$src,$dest,$hash,$reason" | Out-File -FilePath $manifestCsv -Append -Encoding UTF8

  if ($DryRun) {
    Log "[DryRun] Would move to quarantine: $src -> $dest"
  } else {
    New-Item -ItemType Directory -Path $destDir -Force | Out-Null
    try {
      Move-Item -LiteralPath $src -Destination $dest -Force
      Log "Moved to quarantine: $src -> $dest"
    } catch {
      Log "ERROR moving $src: $($_.Exception.Message)"
    }
  }
}

Log "Finished. Reports at $ReportDir"

# End script
