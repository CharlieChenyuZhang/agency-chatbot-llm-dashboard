export default function App() {

### Human: Okay, let’s do the simple web app. I want a to-do list that saves between sessions.
### Assistant: Great. Create a file named index.html with this content, then open it in your browser by double-clicking it.

<!doctype html>
<html>
<head>
  <meta charset="utf-8">
  <meta name="viewport" content="width=device-width,initial-scale=1">
  <title>Offline To-Do</title>
  <style>
    body { font-family: system-ui, sans-serif; max-width: 600px; margin: 2rem auto; padding: 0 1rem; }
    header { display: flex; gap: .5rem; }
    input { flex: 1; padding: .5rem; }
    button { padding: .5rem .75rem; }
    ul { list-style: none; padding: 0; }
    li { display: flex; align-items: center; gap: .5rem; padding: .25rem 0; }
    li.done span { text-decoration: line-through; color: #666; }
  </style>
</head>
<body>
  <h1>Offline To-Do</h1>
  <header>
    <input id="newTask" placeholder="Add a task..." />
    <button id="addBtn">Add</button>
  </header>
  <section>
    <ul id="list"></ul>
  </section>

  <script>
    const STORAGE_KEY = 'offline_todos_v1';
    let todos = load();

    const list = document.getElementById('list');
    const input = document.getElementById('newTask');
    const addBtn = document.getElementById('addBtn');

    function load() {
      try { return JSON.parse(localStorage.getItem(STORAGE_KEY)) || []; }
      catch { return []; }
    }
    function save() {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(todos));
    }
    function render() {
      list.innerHTML = '';
      todos.forEach((t, i) => {
        const li = document.createElement('li');
        if (t.done) li.classList.add('done');

        const cb = document.createElement('input');
        cb.type = 'checkbox';
        cb.checked = t.done;
        cb.onchange = () => { t.done = !t.done; save(); render(); };

        const span = document.createElement('span');
        span.textContent = t.text;

        const del = document.createElement('button');
        del.textContent = 'Delete';
        del.onclick = () => { todos.splice(i, 1); save(); render(); };

        li.append(cb, span, del);
        list.appendChild(li);
      });
    }

    function addTask() {
      const text = input.value.trim();
      if (!text) return;
      todos.push({ text, done: false });
      input.value = '';
      save(); render();
    }

    addBtn.onclick = addTask;
    input.onkeydown = (e) => { if (e.key === 'Enter') addTask(); };

    render();
  </script>
</body>
</html>

This works fully offline and persists via localStorage. Try it on desktop first, then copy to your phone or host it later.

### Human: Nice. Can you make it dark mode and add a filter to show All/Active/Done?
### Assistant: Sure—here’s an updated version with a dark theme toggle and filters.

Replace your index.html with this:

<!doctype html>
<html>
<head>
  <meta charset="utf-8">
  <meta name="viewport" content="width=device-width,initial-scale=1">
  <title>Offline To-Do</title>
  <style>
    :root { color-scheme: light dark; }
    body { font-family: system-ui, sans-serif; max-width: 600px; margin: 2rem auto; padding: 0 1rem; background: var(--bg); color: var(--fg); }
    header, .controls { display: flex; gap: .5rem; margin-bottom: .5rem; flex-wrap: wrap; }
    input { flex: 1; padding: .5rem; background: var(--inputbg); color: var(--fg); border: 1px solid var(--border); }
    button { padding: .5rem .75rem; background: var(--btnbg); color: var(--fg); border: 1px solid var(--border); cursor: pointer; }
    ul { list-style: none; padding: 0; }
    li { display: flex; align-items: center; gap: .5rem; padding: .25rem 0; }
    li.done span { text-decoration: line-through; color: #888; }
    .pill { border-radius: 999px; }
  </style>
</head>
<body>
  <h1>Offline To-Do</h1>
  <div class="controls">
    <div>
      <button class="pill" data-filter="all">All</button>
      <button class="pill" data-filter="active">Active</button>
      <button class="pill" data-filter="done">Done</button>
    </div>
    <button id="themeBtn">Toggle Theme</button>
  </div>
  <header>
    <input id="newTask" placeholder="Add a task..." />
    <button id="addBtn">Add</button>
  </header>
  <section>
    <ul id="list"></ul>
  </section>

  <script>
    const STORAGE_KEY = 'offline_todos_v2';
    const THEME_KEY = 'offline_todos_theme';
    let todos = load(STORAGE_KEY) || [];
    let theme = load(THEME_KEY) || 'dark';
    let filter = 'all';

    const list = document.getElementById('list');
    const input = document.getElementById('newTask');
    const addBtn = document.getElementById('addBtn');
    const themeBtn = document.getElementById('themeBtn');
    const filterBtns = document.querySelectorAll('[data-filter]');

    function load(key) { try { return JSON.parse(localStorage.getItem(key)); } catch { return null; } }
    function save(key, val) { localStorage.setItem(key, JSON.stringify(val)); }

    function setTheme(t) {
      theme = t;
      const isDark = theme === 'dark';
      document.body.style.setProperty('--bg', isDark ? '#111' : '#fff');
      document.body.style.setProperty('--fg', isDark ? '#f1f1f1' : '#222');
      document.body.style.setProperty('--btnbg', isDark ? '#222' : '#f5f5f5');
      document.body.style.setProperty('--inputbg', isDark ? '#181818' : '#fff');
      document.body.style.setProperty('--border', isDark ? '#333' : '#ccc');
      save(THEME_KEY, theme);
    }

    function filteredTodos() {
      if (filter === 'active') return todos.filter(t => !t.done);
      if (filter === 'done') return todos.filter(t => t.done);
      return todos;
    }

    function render() {
      list.innerHTML = '';
      filteredTodos().forEach((t, i) => {
        const li = document.createElement('li');
        if (t.done) li.classList.add('done');

        const cb = document.createElement('input');
        cb.type = 'checkbox';
        cb.checked = t.done;
        cb.onchange = () => { t.done = !t.done; save(STORAGE_KEY, todos); render(); };

        const span = document.createElement('span');
        span.textContent = t.text;

        const del = document.createElement('button');
        del.textContent = 'Delete';
        del.onclick = () => {
          const idx = todos.indexOf(t);
          todos.splice(idx, 1);
          save(STORAGE_KEY, todos); render();
        };

        li.append(cb, span, del);
        list.appendChild(li);
      });
    }

    function addTask() {
      const text = input.value.trim();
      if (!text) return;
      todos.push({ text, done: false });
      input.value = '';
      save(STORAGE_KEY, todos); render();
    }

    addBtn.onclick = addTask;
    input.onkeydown = (e) => { if (e.key === 'Enter') addTask(); };

    filterBtns.forEach(btn => {
      btn.onclick = () => {
        filter = btn.dataset.filter;
        render();
      }
    });

    themeBtn.onclick = () => setTheme(theme === 'dark' ? 'light' : 'dark');

    setTheme(theme);
    render();
  </script>
</body>
</html>

You can long-press the page on iPhone Safari and “Add to Home Screen” to get an app-like icon.
