### Human: Use 2025, all four quarters, base path /mnt/c/Projects.
### Assistant:
#!/usr/bin/env bash
set -euo pipefail
BASE="/mnt/c/Projects"
YEAR="2025"
quarters=(Q1 Q2 Q3 Q4)
declare -A sub
sub[Marketing]="01_Planning 02_Assets 03_Campaigns 04_Reports"
sub[Finance]="01_Budget 02_Invoices 03_Expenses 04_Reports"
sub[Product]="01_Research 02_Design 03_Development 04_Testing 05_Release"
for q in "${quarters[@]}"; do
  for dept in Marketing Finance Product; do
    for f in ${sub[$dept]}; do
      mkdir -p "$BASE/$YEAR/$q/$dept/$f"
    done
  done
done

### Human: Now make a PowerShell version too.
### Assistant:
$Base = 'C:\Projects'
$Year = '2025'
$Quarters = 'Q1','Q2','Q3','Q4'
$Subs = @{
  'Marketing' = @('01_Planning','02_Assets','03_Campaigns','04_Reports')
  'Finance'   = @('01_Budget','02_Invoices','03_Expenses','04_Reports')
  'Product'   = @('01_Research','02_Design','03_Development','04_Testing','05_Release')
}
foreach ($q in $Quarters) {
  foreach ($dept in $Subs.Keys) {
    foreach ($f in $Subs[$dept]) {
      New-Item -ItemType Directory -Force -Path (Join-Path $Base "$Year\$q\$dept\$f") | Out-Null
    }
  }
}
