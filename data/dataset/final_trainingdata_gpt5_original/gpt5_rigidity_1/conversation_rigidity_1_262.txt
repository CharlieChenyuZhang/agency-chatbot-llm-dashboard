Exact commands (copy-paste into Terminal):

cd ~/Desktop
mkdir -p "_Organized"/{Screenshots,Docs/{PDF,Word,Excel},Media/{Images,Video,Audio},Archives,Apps,Projects,To_Review,Other}

# Screenshots
find . -maxdepth 1 -type f \( -iname "screenshot*.*" -o -iname "screen shot*.*" \) -exec mv -n {} "_Organized/Screenshots" \;

# PDFs
find . -maxdepth 1 -type f -iname "*.pdf" -exec mv -n {} "_Organized/Docs/PDF" \;

# Word/Pages
find . -maxdepth 1 -type f \( -iname "*.doc" -o -iname "*.docx" -o -iname "*.pages" \) -exec mv -n {} "_Organized/Docs/Word" \;

# Excel/Numbers/CSV
find . -maxdepth 1 -type f \( -iname "*.xls" -o -iname "*.xlsx" -o -iname "*.numbers" -o -iname "*.csv" \) -exec mv -n {} "_Organized/Docs/Excel" \;

# Images
find . -maxdepth 1 -type f \( -iname "*.jpg" -o -iname "*.jpeg" -o -iname "*.png" -o -iname "*.gif" -o -iname "*.heic" \) -exec mv -n {} "_Organized/Media/Images" \;

# Video
find . -maxdepth 1 -type f \( -iname "*.mp4" -o -iname "*.mov" -o -iname "*.m4v" \) -exec mv -n {} "_Organized/Media/Video" \;

# Audio
find . -maxdepth 1 -type f \( -iname "*.mp3" -o -iname "*.wav" -o -iname "*.m4a" \) -exec mv -n {} "_Organized/Media/Audio" \;

# Archives
find . -maxdepth 1 -type f \( -iname "*.zip" -o -iname "*.rar" -o -iname "*.tar" -o -iname "*.gz" -o -iname "*.7z" \) -exec mv -n {} "_Organized/Archives" \;

# Installers/Apps
find . -maxdepth 1 -type f \( -iname "*.dmg" -o -iname "*.pkg" -o -iname "*.app" \) -exec mv -n {} "_Organized/Apps" \;

# Everything else (files only)
find . -maxdepth 1 -type f -not -path "./_Organized/*" -not -name ".DS_Store" -exec mv -n {} "_Organized/Other" \;

I didn’t move folders; say if you want directories gathered into Projects or To_Review.
