"""
import time, requests, sys

WIKI_API = "https://en.wikipedia.org/w/api.php"
NOMINATIM_API = "https://nominatim.openstreetmap.org/search"
UA = "PghMuseumsHelper/1.0 (contact: your-email@example.com)"

def get_wiki_titles():
    titles = []
    cmcontinue = None
    while True:
        params = {
            "action": "query",
            "list": "categorymembers",
            "cmtitle": "Category:Museums in Pittsburgh",
            "cmlimit": "500",
            "format": "json",
        }
        if cmcontinue:
            params["cmcontinue"] = cmcontinue
        r = requests.get(WIKI_API, params=params, headers={"User-Agent": UA}, timeout=30)
        r.raise_for_status()
        data = r.json()
        for item in data["query"]["categorymembers"]:
            titles.append(item["title"])
        cmcontinue = data.get("continue", {}).get("cmcontinue")
        if not cmcontinue:
            break
    return titles

def geocode(name):
    q = f"{name}, Pittsburgh, PA"
    params = {"q": q, "format": "json", "addressdetails": 1, "limit": 1}
    r = requests.get(NOMINATIM_API, params=params, headers={"User-Agent": UA}, timeout=30)
    r.raise_for_status()
    results = r.json()
    if not results:
        return None
    hit = results[0]
    addr = hit.get("address", {})
    return {
        "lat": hit.get("lat"),
        "lon": hit.get("lon"),
        "display_name": hit.get("display_name"),
        "addr": addr,
    }

def wiki_url(title):
    return "https://en.wikipedia.org/wiki/" + title.replace(" ", "_")

def main():
    titles = get_wiki_titles()
    out_lines = []
    out_lines.append("## Museums in Pittsburgh")
    out_lines.append("")
    for t in titles:
        time.sleep(1)  # be nice to Nominatim
        g = geocode(t)
        if not g:
            line = f"- {t} — location not found — {wiki_url(t)}"
        else:
            city = g["addr"].get("city") or g["addr"].get("town") or g["addr"].get("municipality")
            state = g["addr"].get("state")
            postcode = g["addr"].get("postcode", "")
            road = g["addr"].get("road", "")
            house = g["addr"].get("house_number", "")
            lat, lon = g["lat"], g["lon"]
            address = ", ".join([part for part in [f\"{house} {road}\".strip(), city, state, postcode] if part])
            map_link = f"https://www.openstreetmap.org/?mlat={lat}&mlon={lon}#map=17/{lat}/{lon}"
            line = f"- [{t}]({wiki_url(t)}) — {address} — [Map]({map_link})"
        out_lines.append(line)
    md = "\n".join(out_lines) + "\n"
    with open("museums_pgh.md", "w", encoding="utf-8") as f:
        f.write(md)
    print("Wrote museums_pgh.md")

if __name__ == "__main__":
    main()
