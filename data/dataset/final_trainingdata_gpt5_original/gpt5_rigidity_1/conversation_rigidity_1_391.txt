2) BigQuery SQL (monthly 30-day churn by signup cohort, last 6 months):

DECLARE months_back INT64 DEFAULT 6;

WITH month_range AS (
  SELECT
    DATE_TRUNC(DATE_SUB(CURRENT_DATE(), INTERVAL months_back - 1 MONTH), MONTH) AS start_month,
    DATE_TRUNC(CURRENT_DATE(), MONTH) AS end_month
),
months AS (
  SELECT month AS month_start
  FROM month_range, UNNEST(GENERATE_DATE_ARRAY(start_month, end_month, INTERVAL 1 MONTH)) AS month
),
cohorts AS (
  SELECT
    user_id,
    DATE_TRUNC(signup_date, MONTH) AS cohort_month
  FROM `analytics.users`
),
activity AS (
  SELECT
    user_id,
    DATE_TRUNC(DATE(event_timestamp), MONTH) AS activity_month
  FROM `analytics.events`
  WHERE event_name = 'session_start'
  GROUP BY user_id, activity_month
),
base AS (
  SELECT
    a.user_id,
    a.activity_month,
    c.cohort_month
  FROM activity a
  JOIN cohorts c USING (user_id)
  WHERE a.activity_month IN (SELECT month_start FROM months)
),
churn_eval AS (
  SELECT
    b.user_id,
    b.activity_month,
    b.cohort_month,
    -- 30-day window after the end of the activity month
    NOT EXISTS (
      SELECT 1
      FROM `analytics.events` e2
      WHERE e2.user_id = b.user_id
        AND e2.event_name = 'session_start'
        AND DATE(e2.event_timestamp)
            BETWEEN DATE_ADD(DATE_ADD(b.activity_month, INTERVAL 1 MONTH), INTERVAL 0 DAY)
                AND DATE_ADD(DATE_ADD(b.activity_month, INTERVAL 1 MONTH), INTERVAL 30 DAY)
    ) AS churned_30d
  FROM base b
)
SELECT
  activity_month AS month,
  cohort_month,
  COUNT(*) AS active_users,
  SUM(CASE WHEN churned_30d THEN 1 ELSE 0 END) AS churned_users_30d,
  SAFE_DIVIDE(SUM(CASE WHEN churned_30d THEN 1 ELSE 0 END), COUNT(*)) AS churn_rate_30d
FROM churn_eval
GROUP BY month, cohort_month
ORDER BY month, cohort_month;
