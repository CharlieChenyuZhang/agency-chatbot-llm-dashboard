### Assistant:
class Stack:
    """LIFO stack using Python list."""
    def __init__(self):
        """Initialize empty stack."""
        self._d = []
    def push(self, x):
        """Push item x onto the stack."""
        self._d.append(x)
    def pop(self):
        """Pop and return top item; raise IndexError if empty."""
        if not self._d: raise IndexError("pop from empty stack")
        return self._d.pop()
    def peek(self):
        """Return top item without removing; raise IndexError if empty."""
        if not self._d: raise IndexError("peek from empty stack")
        return self._d[-1]
    def is_empty(self):
        """Return True if stack is empty."""
        return not self._d
    def size(self):
        """Return number of items in stack."""
        return len(self._d)

#include <bits/stdc++.h>
using namespace std;
int main(){
    int n,m,s; cin>>n>>m>>s;
    vector<vector<pair<int,int>>> g(n);
    while(m--){int u,v,w;cin>>u>>v>>w;g[u].push_back({v,w});}
    const long long INF=1e18;
    vector<long long> d(n,INF); d[s]=0;
    priority_queue<pair<long long,int>,vector<pair<long long,int>>,greater<pair<long long,int>>> pq;
    pq.push({0,s});
    while(!pq.empty()){
        auto [du,u]=pq.top(); pq.pop();
        if(du!=d[u]) continue;
        for(auto [v,w]:g[u]) if(d[v]>du+w){ d[v]=du+w; pq.push({d[v],v}); }
    }
    for(int i=0;i<n;i++) cout<<(d[i]==INF?-1:d[i])<<(i+1==n?'\n':' ');
}

### Human: Now contrast greedy vs dynamic programming with 2 sentences total.
### Assistant: Greedy makes the locally optimal choice at each step hoping it leads to a global optimum. Dynamic programming systematically explores subproblems with optimal substructure and uses memoization/tabulation to guarantee optimality.
