### Human: Okay, let's stick with Telegram for now. Can I see a more detailed code example?
### Assistant: Here’s a minimal Python example using python-telegram-bot, SQLite, and APScheduler. It supports /start, setting a habit and check-in time, sending a daily check-in, and recording responses.

Requirements:
- python-telegram-bot==20.7
- APScheduler
- pytz

Environment:
- Set TELEGRAM_TOKEN env var.

Code (single file demo):

import os
import sqlite3
import pytz
from datetime import datetime, time as dtime
from apscheduler.schedulers.background import BackgroundScheduler
from apscheduler.triggers.cron import CronTrigger
from telegram import Update, InlineKeyboardMarkup, InlineKeyboardButton
from telegram.ext import ApplicationBuilder, CommandHandler, CallbackQueryHandler, ContextTypes

DB = "app.db"

def init_db():
    conn = sqlite3.connect(DB)
    cur = conn.cursor()
    cur.execute("""CREATE TABLE IF NOT EXISTS users (
        id INTEGER PRIMARY KEY,
        telegram_id INTEGER UNIQUE,
        tz TEXT
    )""")
    cur.execute("""CREATE TABLE IF NOT EXISTS habits (
        id INTEGER PRIMARY KEY,
        user_id INTEGER,
        name TEXT,
        hour INTEGER,
        FOREIGN KEY(user_id) REFERENCES users(id)
    )""")
    cur.execute("""CREATE TABLE IF NOT EXISTS checkins (
        id INTEGER PRIMARY KEY,
        habit_id INTEGER,
        date TEXT,
        response TEXT,
        created_at TEXT,
        FOREIGN KEY(habit_id) REFERENCES habits(id)
    )""")
    conn.commit()
    conn.close()

def get_conn():
    return sqlite3.connect(DB)

async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    chat_id = update.effective_user.id
    # simple tz default; in production, ask user for tz or detect via location
    tz = "UTC"
    conn = get_conn(); cur = conn.cursor()
    cur.execute("INSERT OR IGNORE INTO users (telegram_id, tz) VALUES (?,?)", (chat_id, tz))
    conn.commit(); conn.close()
    await update.message.reply_text(
        "Welcome! Let's set one habit to track.\nSend: /sethabit Workout 20 min\nThen set a daily check-in hour with /sethour 19 (24h format)."
    )

async def sethabit(update: Update, context: ContextTypes.DEFAULT_TYPE):
    chat_id = update.effective_user.id
    name = " ".join(context.args) if context.args else None
    if not name:
        await update.message.reply_text("Usage: /sethabit Habit name")
        return
    conn = get_conn(); cur = conn.cursor()
    cur.execute("SELECT id FROM users WHERE telegram_id=?", (chat_id,))
    row = cur.fetchone()
    if not row:
        await update.message.reply_text("Please /start first.")
        return
    user_id = row[0]
    cur.execute("INSERT INTO habits (user_id, name, hour) VALUES (?,?,?)", (user_id, name, 19))
    conn.commit(); conn.close()
    await update.message.reply_text(f"Got it. Tracking: {name}. Now set a check-in hour with /sethour 19")

async def sethour(update: Update, context: ContextTypes.DEFAULT_TYPE):
    chat_id = update.effective_user.id
    if not context.args:
        await update.message.reply_text("Usage: /sethour 19")
        return
    try:
        hour = int(context.args[0])
        assert 0 <= hour <= 23
    except:
        await update.message.reply_text("Please provide an hour 0–23.")
        return
    conn = get_conn(); cur = conn.cursor()
    cur.execute("SELECT id FROM users WHERE telegram_id=?", (chat_id,))
    row = cur.fetchone()
    if not row:
        await update.message.reply_text("Please /start first.")
        return
    user_id = row[0]
    cur.execute("UPDATE habits SET hour=? WHERE user_id=? ORDER BY id DESC LIMIT 1", (hour, user_id))
    conn.commit(); conn.close()
    await update.message.reply_text(f"Daily check-in set to {hour}:00. I’ll ping you then.")

async def button(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()
    data = query.data  # e.g., "checkin:123:yes"
    _, habit_id, resp = data.split(":")
    habit_id = int(habit_id)
    today = datetime.utcnow().date().isoformat()
    conn = get_conn(); cur = conn.cursor()
    cur.execute("INSERT INTO checkins (habit_id, date, response, created_at) VALUES (?,?,?,?)",
                (habit_id, today, resp, datetime.utcnow().isoformat()))
    conn.commit(); conn.close()
    msg = "Nice! Keep it up." if resp == "yes" else ("No worries, tomorrow’s a new chance." if resp == "no" else "Noted.")
    await query.edit_message_text(msg)

def send_checkin(app):
    # Runs hourly; sends check-in to users whose hour matches local setting.
    conn = get_conn(); cur = conn.cursor()
    cur.execute("""SELECT habits.id, users.telegram_id, users.tz, habits.name, habits.hour
                   FROM habits JOIN users ON habits.user_id = users.id""")
    rows = cur.fetchall()
    conn.close()
    now_utc = datetime.utcnow()
    for habit_id, tg_id, tzname, habit_name, hour in rows:
        try:
            tz = pytz.timezone(tzname)
        except:
            tz = pytz.UTC
        local_now = now_utc.astimezone(tz)
        if local_now.hour == hour:
            kb = InlineKeyboardMarkup([[
                InlineKeyboardButton("Yes", callback_data=f"checkin:{habit_id}:yes"),
                InlineKeyboardButton("No", callback_data=f"checkin:{habit_id}:no"),
                InlineKeyboardButton("Skip", callback_data=f"checkin:{habit_id}:skip"),
            ]])
            app.create_task(app.bot.send_message(chat_id=tg_id, text=f"Did you complete: {habit_name} today?", reply_markup=kb))

async def stats(update: Update, context: ContextTypes.DEFAULT_TYPE):
    chat_id = update.effective_user.id
    conn = get_conn(); cur = conn.cursor()
    cur.execute("""SELECT habits.id, habits.name FROM habits
                   JOIN users ON users.id = habits.user_id
                   WHERE users.telegram_id=?""", (chat_id,))
    habits = cur.fetchall()
    if not habits:
        await update.message.reply_text("No habits yet. Use /sethabit.")
        return
    lines = []
    for hid, name in habits:
        cur.execute("SELECT response FROM checkins WHERE habit_id=? ORDER BY id DESC LIMIT 7", (hid,))
        recent = [r[0] for r in cur.fetchall()]
        yes_count = sum(1 for r in recent if r == "yes")
        lines.append(f"{name}: {yes_count}/7 last 7 days")
    conn.close()
    await update.message.reply_text("\n".join(lines))

async def main():
    init_db()
    token = os.environ["TELEGRAM_TOKEN"]
    app = ApplicationBuilder().token(token).build()

    app.add_handler(CommandHandler("start", start))
    app.add_handler(CommandHandler("sethabit", sethabit))
    app.add_handler(CommandHandler("sethour", sethour))
    app.add_handler(CommandHandler("stats", stats))
    app.add_handler(CallbackQueryHandler(button, pattern="^checkin:"))

    scheduler = BackgroundScheduler()
    scheduler.add_job(lambda: send_checkin(app), CronTrigger(minute="0"))  # hourly on the hour
    scheduler.start()

    await app.initialize()
    await app.start()
    await app.updater.start_polling()
    await app.updater.idle()

if __name__ == "__main__":
    import asyncio
    asyncio.run(main())
