PowerShell script (InventoryAndHash.ps1):

- Produces inventory.csv in the root.

```
param(
  [string]$Root = "C:\Users\me\Stuff",
  [string]$OutputCsv = "C:\Users\me\Stuff\inventory.csv"
)

Add-Type -AssemblyName System.Drawing | Out-Null

function Get-DateTaken {
  param([string]$Path)
  try {
    $ext = [System.IO.Path]::GetExtension($Path).ToLowerInvariant()
    if (@(".jpg",".jpeg",".png",".tif",".tiff",".gif",".bmp") -notcontains $ext) {
      return $null
    }
    $img = [System.Drawing.Image]::FromFile($Path)
    try {
      # EXIF tag 36867 = DateTimeOriginal, 306 = DateTime
      $prop = $img.PropertyItems | Where-Object { $_.Id -in 36867, 306 } | Select-Object -First 1
      if ($prop -ne $null) {
        $str = [System.Text.Encoding]::ASCII.GetString($prop.Value).Trim([char]0)
        # EXIF date format: "YYYY:MM:DD HH:MM:SS"
        if ($str -match '^\d{4}:\d{2}:\d{2} \d{2}:\d{2}:\d{2}$') {
          $normalized = $str -replace ':','-',1 -replace ':','-',1
          return [datetime]::ParseExact($normalized, "yyyy-MM-dd HH:mm:ss", $null)
        }
      }
    } finally {
      $img.Dispose()
    }
  } catch {
    # ignore failures; return null
  }
  return $null
}

Write-Host "Scanning files under $Root ..."
$files = Get-ChildItem -LiteralPath $Root -Recurse -File -ErrorAction SilentlyContinue

# Optional exclusions
$excludeTop = @("Misc","Personal")
$rootFull = (Resolve-Path $Root).Path

$rows = foreach ($f in $files) {
  $relative = [System.IO.Path]::GetRelativePath($rootFull, $f.FullName)
  # Skip excluded top-levels
  $parts = $relative.Split([IO.Path]::DirectorySeparatorChar, [System.StringSplitOptions]::None)
  if ($parts.Length -gt 0 -and $excludeTop -contains $parts[0]) { continue }

  $hash = (Get-FileHash -LiteralPath $f.FullName -Algorithm SHA256).Hash
  $dateTaken = Get-DateTaken -Path $f.FullName
  [pscustomobject]@{
    FullPath        = $f.FullName
    RelativePath    = $relative
    TopLevel        = ($parts | Select-Object -First 1)
    Name            = $f.Name
    Extension       = $f.Extension.ToLowerInvariant()
    Length          = $f.Length
    LastWriteTime   = $f.LastWriteTimeUtc
    DateTaken       = if ($dateTaken) { $dateTaken.ToUniversalTime() } else { $null }
    SHA256          = $hash
  }
}

$rows | Export-Csv -NoTypeInformation -Encoding UTF8 -Path $OutputCsv
Write-Host "Wrote inventory to $OutputCsv"
```

PowerShell (DedupeAndRefactor.ps1):
```
param(
  [string]$Root = "C:\Users\me\Stuff",
  [string]$InventoryCsv = "C:\Users\me\Stuff\inventory.csv",
  [switch]$WhatIf  # run as -WhatIf for dry-run
)

$ErrorActionPreference = "Stop"

function Get-YearFromRow($row) {
  if ($row.DateTaken) {
    return ([datetime]$row.DateTaken).ToLocalTime().Year
  }
  return ([datetime]$row.LastWriteTime).ToLocalTime().Year
}

function Ensure-Directory([string]$dir, [bool]$simulate) {
  if (-not (Test-Path -LiteralPath $dir)) {
    if ($simulate) { Write-Host "[DRY] New-Item -ItemType Directory $dir" }
    else { New-Item -ItemType Directory -Path $dir | Out-Null }
  }
}

function Move-File([string]$src, [string]$dst, [bool]$simulate) {
  $dstDir = [System.IO.Path]::GetDirectoryName($dst)
  Ensure-Directory $dstDir $simulate
  if ($simulate) {
    Write-Host "[DRY] Move-Item `"$src`" `"$dst`""
  } else {
    Move-Item -LiteralPath $src -Destination $dst
  }
}

function New-HardLink([string]$linkPath, [string]$targetPath, [bool]$simulate) {
  $dir = [System.IO.Path]::GetDirectoryName($linkPath)
  Ensure-Directory $dir $simulate
  if ($simulate) {
    Write-Host "[DRY] New-Item -ItemType HardLink -Path `"$linkPath`" -Target `"$targetPath`""
  } else {
    if (Test-Path -LiteralPath $linkPath) {
      Remove-Item -LiteralPath $linkPath -Force
    }
    New-Item -ItemType HardLink -Path $linkPath -Target $targetPath | Out-Null
  }
}

# Load inventory
$rows = Import-Csv -LiteralPath $InventoryCsv

# Build groups by hash
$groups = $rows | Group-Object -Property SHA256

$plan = @()
$canonicalRoot = Join-Path $Root "By_Project"

# Validate drive
$rootDrive = ([System.IO.Path]::GetPathRoot($Root)).ToLowerInvariant()

foreach ($g in $groups) {
  $files = $g.Group
  if ($files.Count -eq 0) { continue }

  # Choose keeper: newest by DateTaken, else LastWriteTime
  $keeper = $files | Sort-Object `
    @{Expression={ if($_.DateTaken){[datetime]$_.DateTaken} else {[datetime]$_.LastWriteTime} }}, `
    @{Expression={$_.FullPath}} `
    -Descending | Select-Object -First 1

  # Determine Project and Year for keeper
  $proj = $keeper.TopLevel
  if ($proj -in @("Misc","Personal")) {
    # If excluded, place under "_Unassigned"
    $proj = "_Unassigned"
  }
  $year = Get-YearFromRow $keeper

  # Compute canonical keeper path
  $keeperName = $keeper.Name
  $canonicalDir = Join-Path (Join-Path $canonicalRoot $proj) $year
  $canonicalPath = Join-Path $canonicalDir $keeperName

  # If keeper not already at canonical path, plan to move
  $needsMove = ($keeper.FullPath -ne $canonicalPath)

  $plan += [pscustomobject]@{
    Action      = if ($needsMove) { "MoveKeeper" } else { "KeepAsIs" }
    Source      = $keeper.FullPath
    Destination = $canonicalPath
    Note        = "Keeper for hash $($g.Name)"
  }

  # For each duplicate (non-keeper), plan hardlink replacement at original location
  $dupes = $files | Where-Object { $_.FullPath -ne $keeper.FullPath }
  foreach ($d in $dupes) {
    $plan += [pscustomobject]@{
      Action      = "ReplaceWithHardLink"
      Source      = $d.FullPath
      Destination = $canonicalPath
      Note        = "Duplicate of keeper; link points to canonical"
    }
  }
}

# Write plan
$planCsv = Join-Path $Root "dedupe_refactor_plan.csv"
$plan | Export-Csv -NoTypeInformation -Encoding UTF8 -Path $planCsv
Write-Host "Wrote plan to $planCsv"

# Execute plan
$simulate = $WhatIf.IsPresent

# Ensure canonical root
Ensure-Directory $canonicalRoot $simulate

# First move all keepers (so the canonical targets exist before linking)
$keepers = $plan | Where-Object { $_.Action -in "MoveKeeper","KeepAsIs" }
foreach ($item in $keepers) {
  if ($item.Action -eq "MoveKeeper") {
    # Confirm same drive
    $srcDrive = ([System.IO.Path]::GetPathRoot($item.Source)).ToLowerInvariant()
    $dstDrive = ([System.IO.Path]::GetPathRoot($item.Destination)).ToLowerInvariant()
    if ($srcDrive -ne $rootDrive -or $dstDrive -ne $rootDrive) {
      throw "Move would cross drives; hardlinks won’t work across volumes. Adjust paths to stay on $rootDrive."
    }
    Move-File -src $item.Source -dst $item.Destination -simulate $simulate
  }
}

# Then replace duplicates with hardlinks pointing to canonical keeper
$links = $plan | Where-Object { $_.Action -eq "ReplaceWithHardLink" }
foreach ($ln in $links) {
  # Confirm target exists (or will exist)
  if (-not $simulate -and -not (Test-Path -LiteralPath $ln.Destination)) {
    Write-Warning "Canonical target missing: $($ln.Destination). Skipping link for $($ln.Source)."
    continue
  }

  # Create link at original duplicate path, pointing to canonical keeper
  New-HardLink -linkPath $ln.Source -targetPath $ln.Destination -simulate $simulate
}

Write-Host ("Completed " + ($(if ($simulate) { "DRY-RUN" } else { "EXECUTION" })))
```

```
function Get-UniquePath([string]$desiredPath) {
  if (-not (Test-Path -LiteralPath $desiredPath)) { return $desiredPath }
  $dir = [System.IO.Path]::GetDirectoryName($desiredPath)
  $name = [System.IO.Path]::GetFileNameWithoutExtension($desiredPath)
  $ext = [System.IO.Path]::GetExtension($desiredPath)
  $i = 1
  do {
    $candidate = Join-Path $dir ("{0} ({1}){2}" -f $name,$i,$ext)
    $i++
  } while (Test-Path -LiteralPath $candidate)
  return $candidate
}

$logPath = Join-Path $Root "dedupe_refactor_log.txt"
function Write-Log($msg) {
  $timestamp = (Get-Date).ToString("s")
  "$timestamp `t $msg" | Out-File -FilePath $logPath -Append -Encoding UTF8
}

function Move-File([string]$src, [string]$dst, [bool]$simulate) {
  $dstDir = [System.IO.Path]::GetDirectoryName($dst)
  Ensure-Directory $dstDir $simulate
  $finalDst = $dst
  if (Test-Path -LiteralPath $dst) {
    $finalDst = Get-UniquePath $dst
    Write-Log "Collision: $dst exists. Using $finalDst"
  }
  if ($simulate) {
    Write-Host "[DRY] Move-Item `"$src`" `"$finalDst`""
    Write-Log "[DRY] Move $src -> $finalDst"
  } else {
    Move-Item -LiteralPath $src -Destination $finalDst
    Write-Log "Moved $src -> $finalDst"
  }
}
```
