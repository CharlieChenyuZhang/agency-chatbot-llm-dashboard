script.js:
```
const form = document.getElementById('orderForm');
const list = document.getElementById('orderList');
const exportBtn = document.getElementById('export');
const clearBtn  = document.getElementById('clear');

let orders = JSON.parse(localStorage.getItem('orders') || '[]');
render();

form.addEventListener('submit', (e) => {
  e.preventDefault();
  const order = {
    drink: document.getElementById('drink').value.trim(),
    size: document.getElementById('size').value,
    notes: document.getElementById('notes').value.trim(),
    time: new Date().toISOString()
  };
  orders.push(order);
  localStorage.setItem('orders', JSON.stringify(orders));
  form.reset();
  render();
});

function render() {
  list.innerHTML = '';
  orders.forEach((o, i) => {
    const li = document.createElement('li');
    li.className = 'order';
    li.innerHTML = `
      <div>
        <strong>${o.drink} (${o.size})</strong><br/>
        <small>${o.notes || 'No notes'} • ${new Date(o.time).toLocaleTimeString()}</small>
      </div>
      <button data-i="${i}" class="del">✕</button>
    `;
    list.appendChild(li);
  });
  list.querySelectorAll('.del').forEach(btn => {
    btn.addEventListener('click', () => {
      const i = +btn.dataset.i;
      orders.splice(i, 1);
      localStorage.setItem('orders', JSON.stringify(orders));
      render();
    });
  });
}

exportBtn.addEventListener('click', () => {
  const header = ['drink','size','notes','time'];
  const rows = orders.map(o => [o.drink,o.size,o.notes,o.time]);
  const csv = [header, ...rows].map(r => r.map(field => `"${(field||'').replace(/"/g,'""')}"`).join(',')).join('\n');
  const blob = new Blob([csv], { type: 'text/csv' });
  const url = URL.createObjectURL(blob);
  const a = Object.assign(document.createElement('a'), { href: url, download: 'orders.csv' });
  a.click();
  URL.revokeObjectURL(url);
});

clearBtn.addEventListener('click', () => {
  if (confirm('Clear all orders?')) {
    orders = [];
    localStorage.setItem('orders', '[]');
    render();
  }
});
```

function render() {

    btn.addEventListener('click', () => {

      localStorage.setItem('orders', JSON.stringify(orders));
      render();
    });
  });
}
