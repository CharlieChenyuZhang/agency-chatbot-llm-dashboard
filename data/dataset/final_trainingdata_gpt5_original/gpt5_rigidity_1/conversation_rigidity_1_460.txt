### Assistant: import discord

intents = discord.Intents.default()
intents.members = True

client = discord.Client(intents=intents)

WELCOME_MESSAGE = """Welcome—you’re in good company. You’ll find peers who build, test, and share. Start by saying hello in #lounge, then post a current project in #showcase. When you want constructive input, open a thread in #feedback with your goal and a build link. You’re encouraged to ask small questions, offer thoughtful critiques, and set simple weekly goals. If anything feels off, ping the mods. We’re glad you’re here and excited to see what you create."""

CHANNEL_ID = 123456789012345678

@client.event
async def on_member_join(member):
    role = discord.utils.get(member.guild.roles, name="Member")
    if role:
        await member.add_roles(role)
    channel = client.get_channel(CHANNEL_ID)
    if channel:
        await channel.send(f"{member.mention} {WELCOME_MESSAGE}")

client.run("YOUR_BOT_TOKEN")
