Code: generate synthetic panel + run DiD and event study (paste into Colab)

# Install packages
!pip install linearmodels statsmodels pandas numpy matplotlib seaborn

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from linearmodels.panel import PanelOLS
import statsmodels.api as sm

# 1) Generate synthetic panel data
np.random.seed(42)
n_units = 50
n_periods = 24
units = [f"store_{i}" for i in range(n_units)]
periods = pd.date_range("2022-01-01", periods=n_periods, freq="MS")

df = pd.MultiIndex.from_product([units, periods], names=["unit","time"]).to_frame(index=False)
df["month"] = df["time"].dt.month
df["trend"] = np.arange(n_periods).repeat(n_units)

# Staggered treatment start by unit
start_times = np.random.choice(range(8, 18), size=n_units, replace=True)
start_map = dict(zip(units, start_times))
df["treat_start"] = df["unit"].map(start_map)
df["treated"] = (df["trend"] >= df["treat_start"]).astype(int)

# Baseline outcomes with unit and time effects
unit_fe = pd.Series(np.random.normal(0, 2, n_units), index=units)
time_fe = pd.Series(np.sin(np.linspace(0, 3*np.pi, n_periods))*2, index=periods)
df["unit_fe"] = df["unit"].map(unit_fe)
df["time_fe"] = df["time"].map(time_fe)

# Covariate (price) and noise
df["price"] = np.clip(np.random.normal(10, 2, len(df)) + 0.2*df["month"], 5, None)
true_tau = 1.5  # true treatment effect
df["epsilon"] = np.random.normal(0, 2, len(df))

# Outcome: sales
df["sales"] = 20 + df["unit_fe"] + df["time_fe"] - 0.8*df["price"] + true_tau*df["treated"] + df["epsilon"]

# 2) Event-time for event study
df["event_time"] = df["trend"] - df["treat_start"]

# 3) Two-way fixed effects DiD
# Prepare panel data with entity and time effects
df = df.set_index(["unit","time"])
df["treated"] = df["treated"].astype(int)
exog = sm.add_constant(df[["treated","price"]])
mod = PanelOLS(df["sales"], exog, entity_effects=True, time_effects=True)
res = mod.fit(cov_type="clustered", cluster_entity=True)
print(res.summary)

# 4) Event study (bin extremes)
df = df.reset_index()
df["event_time_binned"] = df["event_time"].clip(-6, 6)
# Create dummies excluding -1 as reference
dummies = pd.get_dummies(df["event_time_binned"], prefix="et")
for col in dummies.columns:
    if col == "et_-1":  # reference period
        continue
    df[col] = dummies[col]

# Build formula-like regression with unit/time FE
y = df["sales"]
X_cols = [c for c in df.columns if c.startswith("et_") and c != "et_-1"] + ["price"]
X = sm.add_constant(df[X_cols])
# Absorb unit and time fixed effects via dummies (ok for small synthetic data)
X = pd.concat([X, pd.get_dummies(df["unit"], drop_first=True), pd.get_dummies(df["time"], drop_first=True)], axis=1)
res_es = sm.OLS(y, X).fit(cov_type="HC1")
print(res_es.summary().tables[0])

# 5) Plot event study coefficients
coef = res_es.params[[c for c in X_cols if c.startswith("et_") and c != "et_-1"]]
se = res_es.bse[coef.index]
event_vals = [int(c.split("_")[1]) for c in coef.index]
plot_df = pd.DataFrame({"event_time": event_vals, "coef": coef.values, "se": se.values})
plot_df = plot_df.sort_values("event_time")
plt.figure(figsize=(8,5))
plt.axhline(0, color="grey", linestyle="--", linewidth=1)
plt.errorbar(plot_df["event_time"], plot_df["coef"], yerr=1.96*plot_df["se"], fmt="o-")
plt.title("Event Study: Treatment Effect Over Time (ref = -1)")
plt.xlabel("Event Time")
plt.ylabel("Estimated Effect")
plt.show()
