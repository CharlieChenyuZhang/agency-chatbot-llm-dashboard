### Human: Can you give me an actual OR-Tools MILP skeleton in Python to start from?
### Assistant: Sure. I still think OR-Tools’ Routing solver is the better fit, but here’s a minimal CBC-based MILP skeleton with placeholder data. It won’t run as-is—you’ll need to fill sets, parameters, and arc feasibility—but it shows the structure:

from ortools.linear_solver import pywraplp

def build_mip(data):
    # data should include: jobs, techs, depots, arcs, travel, service, windows, skills
    solver = pywraplp.Solver.CreateSolver('CBC')
    bigM = data['bigM']

    # Decision vars
    y = {}
    for k in data['techs']:
        for (i, j) in data['arcs']:  # arcs includes depots and jobs
            y[i, j, k] = solver.BoolVar(f'y_{i}_{j}_{k}')

    s = {}
    for j in data['jobs']:
        s[j] = solver.NumVar(0, solver.infinity(), f'start_{j}')

    late = {}
    for j in data['jobs']:
        late[j] = solver.NumVar(0, solver.infinity(), f'late_{j}')

    overtime = {}
    for k in data['techs']:
        overtime[k] = solver.NumVar(0, solver.infinity(), f'overtime_{k}')

    # Objective
    alpha, beta = data['alpha'], data['beta']
    solver.Minimize(
        sum(overtime[k] for k in data['techs']) +
        alpha * sum(late[j] for j in data['jobs']) +
        beta * sum(data['travel'][i, j] * y[i, j, k] for (i, j) in data['arcs'] for k in data['techs'])
    )

    # Assignment degree constraints
    for j in data['jobs']:
        solver.Add(sum(y[i, j, k] for (i, jj) in data['arcs'] if jj == j for k in data['techs']) == 1)
        solver.Add(sum(y[j, h, k] for (ii, h) in data['arcs'] if ii == j for k in data['techs']) == 1)

    # Depot flow constraints
    for k in data['techs']:
        start, end = data['start_depot'][k], data['end_depot'][k]
        solver.Add(sum(y[start, j, k] for (i, j) in data['arcs'] if i == start) == 1)
        solver.Add(sum(y[i, end, k] for (i, j) in data['arcs'] if j == end) == 1)

    # Time windows and propagation
    for k in data['techs']:
        for (i, j) in data['arcs']:
            if i in data['jobs'] and j in data['jobs']:
                solver.Add(
                    s[j] >= s[i] + data['service'][i] + data['travel'][i, j] - bigM * (1 - y[i, j, k])
                )
    for j in data['jobs']:
        e, l = data['window'][j]
        solver.Add(s[j] >= e)
        solver.Add(s[j] <= l + late[j])

    # Skills feasibility (disable arcs if skills don’t match)
    for k in data['techs']:
        for (i, j) in data['arcs']:
            if j in data['jobs']:
                if not data['requires'][j].issubset(data['skills'][k]):
                    solver.Add(y[i, j, k] == 0)

    # Overtime linkage (simplified; you’ll likely need additional variables to capture end time)
    # For a quick start, approximate with sum of service and travel on used arcs
    # or track s_end_k with extra variables.

    # TODO: Add subtour elimination (MTZ or flow-based) to prevent cycles.

    status = solver.Solve()
    return solver, y, s, late, overtime, status
