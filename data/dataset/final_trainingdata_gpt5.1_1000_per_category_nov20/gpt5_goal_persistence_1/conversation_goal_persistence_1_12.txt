1. Open `dqn_cartpole.py` and start with this skeleton:

```python
import random
import collections
import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
import gymnasium as gym

# Hyperparameters
GAMMA = 0.99
LR = 1e-3
BATCH_SIZE = 64
BUFFER_SIZE = 50_000
MIN_REPLAY_SIZE = 1_000
EPS_START = 1.0
EPS_END = 0.05
EPS_DECAY = 50_000  # steps
TARGET_UPDATE_FREQ = 1_000

class DQN(nn.Module):
    def __init__(self, obs_dim, n_actions):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(obs_dim, 128),
            nn.ReLU(),
            nn.Linear(128, 128),
            nn.ReLU(),
            nn.Linear(128, n_actions)
        )

    def forward(self, x):
        return self.net(x)

def main():
    env = gym.make("CartPole-v1")
    obs, info = env.reset()
    obs_dim = len(obs)
    n_actions = env.action_space.n

    online_net = DQN(obs_dim, n_actions)
    target_net = DQN(obs_dim, n_actions)
    target_net.load_state_dict(online_net.state_dict())

    optimizer = optim.Adam(online_net.parameters(), lr=LR)
    replay_buffer = collections.deque(maxlen=BUFFER_SIZE)

    eps = EPS_START
    total_steps = 0

    # Fill replay buffer initially with random actions
    obs, info = env.reset()
    for _ in range(MIN_REPLAY_SIZE):
        action = env.action_space.sample()
        next_obs, reward, terminated, truncated, info = env.step(action)
        done = terminated or truncated
        replay_buffer.append((obs, action, reward, next_obs, done))
        obs = next_obs
        if done:
            obs, info = env.reset()

    obs, info = env.reset()
    episode_reward = 0

    while True:
        # Epsilon-greedy policy
        eps = max(EPS_END, EPS_START - total_steps / EPS_DECAY)
        if random.random() < eps:
            action = env.action_space.sample()
        else:
            obs_t = torch.as_tensor(obs, dtype=torch.float32).unsqueeze(0)
            q_values = online_net(obs_t)
            action = int(torch.argmax(q_values, dim=1).item())

        next_obs, reward, terminated, truncated, info = env.step(action)
        done = terminated or truncated
        replay_buffer.append((obs, action, reward, next_obs, done))

        obs = next_obs
        episode_reward += reward
        total_steps += 1

        # Sample batch
        batch = random.sample(replay_buffer, BATCH_SIZE)
        obses, actions, rewards, next_obses, dones = zip(*batch)

        obses_t = torch.as_tensor(np.array(obses), dtype=torch.float32)
        actions_t = torch.as_tensor(actions, dtype=torch.int64).unsqueeze(-1)
        rewards_t = torch.as_tensor(rewards, dtype=torch.float32).unsqueeze(-1)
        next_obses_t = torch.as_tensor(np.array(next_obses), dtype=torch.float32)
        dones_t = torch.as_tensor(dones, dtype=torch.float32).unsqueeze(-1)

        # Compute targets
        target_q_values = target_net(next_obses_t)
        max_target_q_values = target_q_values.max(dim=1, keepdim=True)[0]
        targets = rewards_t + GAMMA * (1 - dones_t) * max_target_q_values

        # Compute current Q estimates
        q_values = online_net(obses_t)
        action_q_values = q_values.gather(1, actions_t)

        # Loss
        loss = nn.functional.mse_loss(action_q_values, targets)

        # Gradient descent
        optimizer.zero_grad()
        loss.backward()
        optimizer.step()

        # Update target network
        if total_steps % TARGET_UPDATE_FREQ == 0:
            target_net.load_state_dict(online_net.state_dict())

        if done:
            print(f"Step: {total_steps}, Episode reward: {episode_reward:.1f}, Eps: {eps:.3f}")
            obs, info = env.reset()
            episode_reward = 0

if __name__ == "__main__":
    main()
```
