```javascript
function cleanPricesAndRatings() {
  const sheet = SpreadsheetApp.getActiveSheet();
  const lastRow = sheet.getLastRow();

  if (lastRow < 2) return; // no data

  // Read original data (B=price, C=rating)
  const prices = sheet.getRange(2, 2, lastRow - 1, 1).getValues(); // col B
  const ratings = sheet.getRange(2, 3, lastRow - 1, 1).getValues(); // col C

  const cleanPrices = [];
  const cleanRatings = [];

  for (let i = 0; i < prices.length; i++) {
    const rawPrice = prices[i][0];
    const rawRating = ratings[i][0];

    // Clean price
    let p = "";
    if (rawPrice !== "" && rawPrice !== "N/A") {
      const s = String(rawPrice).replace("$", "").trim();
      const num = parseFloat(s);
      p = isNaN(num) ? "" : num;
    }
    cleanPrices.push([p]);

    // Clean rating
    let r = "";
    if (rawRating !== "" && rawRating !== "N/A") {
      const s = String(rawRating).trim();
      if (s.includes("/")) {
        // format like 4.5/5
        const parts = s.split("/");
        const num = parseFloat(parts[0]);
        const den = parseFloat(parts[1]);
        if (!isNaN(num) && !isNaN(den) && den !== 0) {
          r = (num / den) * 100;
        } else {
          r = "";
        }
      } else {
        // format like 85 or 85%
        const noPercent = s.replace("%", "");
        const num = parseFloat(noPercent);
        r = isNaN(num) ? "" : num;
      }
    }
    cleanRatings.push([r]);
  }

  // Write cleaned values to E=price_numeric, F=rating_numeric
  sheet.getRange(2, 5, cleanPrices.length, 1).setValues(cleanPrices); // col E
  sheet.getRange(2, 6, cleanRatings.length, 1).setValues(cleanRatings); // col F
}
```
