**Detection plan (v1):**

5. **Scoring**
   - Give each potential subscription a score based on:
     - Number of occurrences (≥3 is stronger).
     - Regularity of intervals.
     - Stability of amounts.
   - Only show subscriptions over a threshold score.

**Output per subscription:**
- Merchant name
- Average amount
- Count of occurrences
- Approximate frequency (monthly, weekly, etc.)
- Estimated yearly cost

```python
def group_by_merchant(transactions: list[Transaction]) -> dict[str, list[Transaction]]:
    groups = defaultdict(list)
    for tx in transactions:
        norm = normalize_merchant(tx.merchant_raw)
        groups[norm].append(tx)
    # sort each group by date
    for norm in groups:
        groups[norm].sort(key=lambda t: t.date)
    return groups

def detect_subscriptions_for_group(txs: list[Transaction]):
    if len(txs) < 3:
        return None

    # intervals in days
    intervals = [
        (txs[i].date - txs[i-1].date).days
        for i in range(1, len(txs))
    ]
    if not intervals:
        return None

    median_interval = median(intervals)
    # classify frequency
    if 25 <= median_interval <= 35:
        freq = "monthly"
    elif 5 <= median_interval <= 9:
        freq = "weekly"
    elif 330 <= median_interval <= 395:
        freq = "yearly"
    else:
        return None  # not regular enough (v1 simplification)

    # check amount stability
    amounts = [abs(t.amount) for t in txs]
    med_amount = median(amounts)
    # % deviation from median
    deviations = [abs(a - med_amount) / med_amount for a in amounts]
    if max(deviations) > 0.2:  # >20% variation, skip for v1
        return None

    est_yearly = {
        "weekly": med_amount * 52,
        "monthly": med_amount * 12,
        "yearly": med_amount,
    }[freq]

    return {
        "merchant": txs[0].merchant_raw,
        "normalized": normalize_merchant(txs[0].merchant_raw),
        "frequency": freq,
        "avg_amount": med_amount,
        "occurrences": len(txs),
        "estimated_yearly": est_yearly,
    }

def detect_subscriptions(transactions: list[Transaction]):
    groups = group_by_merchant(transactions)
    results = []
    for norm, txs in groups.items():
        sub = detect_subscriptions_for_group(txs)
        if sub:
            results.append(sub)
    return results
```
