Sample script (save as `find_similar_docs.py`):

```python
import os
import sys
import hashlib
from rapidfuzz import fuzz

try:
    import docx
except ImportError:
    docx = None

try:
    from pptx import Presentation
except ImportError:
    Presentation = None

try:
    import pdfplumber
except ImportError:
    pdfplumber = None

SUPPORTED_EXT = {'.docx', '.pptx', '.pdf'}

def get_text_from_docx(path):
    if docx is None:
        return ""
    try:
        d = docx.Document(path)
        return "\n".join(p.text for p in d.paragraphs)
    except Exception:
        return ""

def get_text_from_pptx(path):
    if Presentation is None:
        return ""
    try:
        prs = Presentation(path)
        texts = []
        for slide in prs.slides:
            for shape in slide.shapes:
                if hasattr(shape, "text"):
                    texts.append(shape.text)
        return "\n".join(texts)
    except Exception:
        return ""

def get_text_from_pdf(path):
    if pdfplumber is None:
        return ""
    try:
        texts = []
        with pdfplumber.open(path) as pdf:
            for page in pdf.pages:
                texts.append(page.extract_text() or "")
        return "\n".join(texts)
    except Exception:
        return ""

def extract_text(path):
    ext = os.path.splitext(path)[1].lower()
    if ext == '.docx':
        return get_text_from_docx(path)
    elif ext == '.pptx':
        return get_text_from_pptx(path)
    elif ext == '.pdf':
        return get_text_from_pdf(path)
    else:
        return ""

def main(folder, threshold=90, recurse=False):
    files = []
    for root, dirs, fs in os.walk(folder):
        for f in fs:
            ext = os.path.splitext(f)[1].lower()
            if ext in SUPPORTED_EXT:
                files.append(os.path.join(root, f))
        if not recurse:
            break

    print(f"Found {len(files)} supported files.")
    docs = []
    for path in files:
        text = extract_text(path)
        if not text.strip():
            continue
        docs.append((path, text))

    print(f"Got text for {len(docs)} files. Comparing...")

    n = len(docs)
    visited = set()
    groups = []

    for i in range(n):
        if i in visited:
            continue
        base_path, base_text = docs[i]
        group = [base_path]
        visited.add(i)
        for j in range(i + 1, n):
            if j in visited:
                continue
            other_path, other_text = docs[j]
            # Use a simple, relatively fast similarity metric
            score = fuzz.token_set_ratio(base_text, other_text)
            if score >= threshold:
                group.append(other_path)
                visited.add(j)
        if len(group) > 1:
            groups.append(group)

    print(f"\nFound {len(groups)} groups of similar documents (threshold={threshold}).\n")

    for idx, group in enumerate(groups, start=1):
        print(f"Group {idx}:")
        for p in group:
            print(" ", p)
        print("")

if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Usage: python find_similar_docs.py <folder> [threshold] [--recurse]")
        sys.exit(1)
    folder = sys.argv[1]
    threshold = int(sys.argv[2]) if len(sys.argv) >= 3 else 90
    recurse = '--recurse' in sys.argv[3:]
    main(folder, threshold, recurse)
```

How to run (example):
- From Command Prompt:

```bash
cd "C:\Path\To\A\ProjectFolder"
python C:\Path\To\find_similar_docs.py . 90 --recurse
```

This will:
- Look for .docx, .pptx, .pdf files under that folder.
- Attempt to extract text and compare each pair.
- Print groups of files whose content similarity is ≥ 90 (you can adjust that).

Limitations:
- It can be slow on very large sets of big documents.
- PDF extraction is imperfect; some documents may appear unrelated or be skipped.
- It *only* prints, never modifies files.

```powershell
# Suggest target subfolder based on extension and identify version groups

$files = Get-ChildItem -File -Recurse

function Get-BaseName($name) {
    $base = [System.IO.Path]::GetFileNameWithoutExtension($name)
    $base = $base -replace '[_\-\s]*(v\d+|v\d\d+)$',''
    $base = $base -replace '[_\-\s]*(final\d*|FINAL\d*)$',''
    return $base.ToLower()
}

function Get-TypeFolder($ext) {
    switch ($ext.ToLower()) {
        '.doc' { 'Docs' }
        '.docx' { 'Docs' }
        '.ppt' { 'Slides' }
        '.pptx' { 'Slides' }
        '.pdf' { 'PDFs' }
        default { '' }
    }
}

$info = @()

$grouped = $files | Group-Object { Get-BaseName $_.Name }

foreach ($group in $grouped) {
    # Sort newest first
    $sorted = $group.Group | Sort-Object LastWriteTime -Descending
    for ($i = 0; $i -lt $sorted.Count; $i++) {
        $file = $sorted[$i]
        $ext = [System.IO.Path]::GetExtension($file.Name)
        $suggestedFolder = Get-TypeFolder($ext)
        $role = if ($sorted.Count -eq 1) { 'single' } elseif ($i -eq 0) { 'keep_latest' } else { 'older_version' }

        $info += [PSCustomObject]@{
            FullName         = $file.FullName
            Name             = $file.Name
            LastWriteTime    = $file.LastWriteTime
            LengthBytes      = $file.Length
            VersionGroupBase = $group.Name
            SuggestedRole    = $role
            SuggestedFolder  = $suggestedFolder
        }
    }
}

$info | Export-Csv .\RefactorSuggestions.csv -NoTypeInformation
Write-Host "Suggestions written to RefactorSuggestions.csv (no files moved)."
```

What this does:
- Groups files by a “base name” without v1/v2/final suffixes.
- For each group:
  - Labels the newest as `keep_latest`.
  - Labels others as `older_version`.
- Suggests a target subfolder (`Docs`, `Slides`, `PDFs`) based on extension.
- Writes everything to `RefactorSuggestions.csv` in the project folder.
- Does not move, rename, or delete anything.
