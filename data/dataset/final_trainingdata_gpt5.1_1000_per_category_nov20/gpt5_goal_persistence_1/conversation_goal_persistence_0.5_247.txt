```python
import pandas as pd

df = pd.read_csv("products.csv")

df = df.rename(columns={
    "Price ": "price",
    "Rating %": "rating_percent"
})

df["price"] = pd.to_numeric(df["price"], errors="coerce")

# Extract numeric rating before "/5"
df["rating_percent"] = (
    df["rating_percent"]
    .astype(str)
    .str.split("/", n=1)
    .str[0]                # take the part before "/"
)

df["rating_percent"] = pd.to_numeric(df["rating_percent"], errors="coerce") * 20

# Filter: at least 80% and under $50
filtered = df[(df["rating_percent"] >= 80) & (df["price"] < 50)]

print(filtered)
```
