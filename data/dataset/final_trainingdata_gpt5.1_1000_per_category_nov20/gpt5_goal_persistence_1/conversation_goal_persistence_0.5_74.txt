```python
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.common.keys import Keys
from selenium.webdriver.chrome.options import Options
import csv
import time

# --- Configuration ---
LOGIN_URL = "https://your-clienttrack-url.com/login"
CUSTOMER_LIST_URL = "https://your-clienttrack-url.com/customers"
USERNAME = "your_username"
PASSWORD = "your_password"
OUTPUT_CSV = "customers.csv"

# --- Setup ---
options = Options()
options.add_argument("--start-maximized")
driver = webdriver.Chrome(options=options)

def login():
    driver.get(LOGIN_URL)
    time.sleep(3)

    # Update these selectors:
    user_input = driver.find_element(By.ID, "username")
    pass_input = driver.find_element(By.ID, "password")

    user_input.send_keys(USERNAME)
    pass_input.send_keys(PASSWORD)
    pass_input.send_keys(Keys.RETURN)

    time.sleep(5)

def go_to_customer_list():
    driver.get(CUSTOMER_LIST_URL)
    time.sleep(5)

def extract_page_rows():
    rows_data = []
    # Update selector for the table rows:
    rows = driver.find_elements(By.CSS_SELECTOR, "table.customers tbody tr")
    for row in rows:
        cells = row.find_elements(By.TAG_NAME, "td")
        # Adjust indices according to your table columns:
        name = cells[0].text.strip()
        email = cells[1].text.strip()
        phone = cells[2].text.strip()
        rows_data.append([name, email, phone])
    return rows_data

def click_next_page():
    try:
        # Update selector for "Next" button:
        next_btn = driver.find_element(By.CSS_SELECTOR, "a.next-page")
        if "disabled" in next_btn.get_attribute("class"):
            return False
        next_btn.click()
        time.sleep(3)
        return True
    except:
        return False

def main():
    login()
    go_to_customer_list()

    with open(OUTPUT_CSV, "w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        writer.writerow(["Name", "Email", "Phone"])  # header row

        while True:
            rows = extract_page_rows()
            if not rows:
                break
            writer.writerows(rows)

            if not click_next_page():
                break

    driver.quit()

if __name__ == "__main__":
    main()
```
