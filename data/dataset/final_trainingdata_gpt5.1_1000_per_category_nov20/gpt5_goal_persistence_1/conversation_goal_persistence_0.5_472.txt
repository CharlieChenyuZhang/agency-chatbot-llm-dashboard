use gloo_net::http::Request;
use wasm_bindgen_futures::spawn_local;
use yew::prelude::*;

#[function_component(App)]
pub fn app() -> Html {
    let health = use_state(|| "checking...".to_string());

    {
        let health = health.clone();
        use_effect_with((), move |_| {
            spawn_local(async move {
                let result = Request::get("http://localhost:4000/health")
                    .send()
                    .await;

                match result {
                    Ok(resp) => {
                        if let Ok(json) = resp.json::<serde_json::Value>().await {
                            let status = json
                                .get("status")
                                .and_then(|s| s.as_str())
                                .unwrap_or("unknown")
                                .to_string();
                            health.set(status);
                        } else {
                            health.set("parse error".to_string());
                        }
                    }
                    Err(_) => health.set("error".to_string()),
                }
            });
            || ()
        });
    }

    html! {
        <main>
            <h1>{ "My Rust WASM App" }</h1>
            <p>{ format!("Backend health: {}", *health) }</p>
        </main>
    }
}

`backend/src/routes.rs`:

```rust
use actix_web::{get, web, HttpResponse, Responder};
use serde::Serialize;
use sqlx::PgPool;

pub fn config(cfg: &mut web::ServiceConfig) {
    cfg.service(health).service(list_users);
}

#[get("/health")]
async fn health() -> impl Responder {
    HttpResponse::Ok().json(serde_json::json!({ "status": "ok" }))
}

#[derive(Serialize)]
struct UserDto {
    id: String,
    email: String,
}

#[get("/users")]
async fn list_users(pool: web::Data<PgPool>) -> impl Responder {
    let rows = sqlx::query!(
        r#"
        SELECT id, email
        FROM users
        ORDER BY email
        "#
    )
    .fetch_all(pool.get_ref())
    .await;

    match rows {
        Ok(rows) => {
            let users: Vec<UserDto> = rows
                .into_iter()
                .map(|row| UserDto {
                    id: row.id.to_string(),
                    email: row.email,
                })
                .collect();
            HttpResponse::Ok().json(users)
        }
        Err(e) => {
            eprintln!("DB error: {e}");
            HttpResponse::InternalServerError().finish()
        }
    }
}
```
