$seen = @{}

# Helper: try to extract client name from path or file name
function Get-ClientName($filePath) {

function Get-ClientName($filePath) {
    # Remove drive and split into segments
    $relative = $filePath -replace "^[A-Za-z]:\\",""
    $segments = $relative -split "[\\/]"

    if ($segments.Count -ge 2) {
        # Last segment is filename, second-to-last is the immediate parent folder
        $parentFolder = $segments[$segments.Count - 2]

    $nameTokens = $fileNameNoExt -split "[ _\-().]+"

        }
    }

    return "UnknownClient"
}
