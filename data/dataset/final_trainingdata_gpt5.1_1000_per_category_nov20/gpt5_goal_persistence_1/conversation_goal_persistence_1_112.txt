import pandas as pd
from typing import Optional, Tuple

def load_sales_data(csv_path: str) -> pd.DataFrame:
    """Load the sales CSV into a DataFrame, with basic cleaning."""
    df = pd.read_csv(csv_path)
    
    # Standardize column names (strip, lowercase)
    df.columns = [c.strip().lower() for c in df.columns]
    
    # Ensure required columns exist
    required_cols = {"product_id", "region", "channel", "units", "revenue", "year"}
    missing_cols = required_cols - set(df.columns)
    if missing_cols:
        raise ValueError(f"Missing required columns in CSV: {missing_cols}")
    
    # Coerce numeric fields; invalid entries become NaN
    df["revenue"] = pd.to_numeric(df["revenue"], errors="coerce")
    df["units"] = pd.to_numeric(df["units"], errors="coerce")
    df["year"] = pd.to_numeric(df["year"], errors="coerce").astype("Int64")
    
    # Optional: normalize region strings (strip, upper)
    df["region"] = df["region"].astype(str).str.strip()
    
    return df

def top_product_by_revenue(
    df: pd.DataFrame,
    year: int,
    region: str,
    min_revenue: float = 0.0
) -> Optional[Tuple[str, float]]:
    """
    Find the top-1 product by total revenue for a given year and region.
    
    Robustness features:
    - Ignores rows with missing or non-positive revenue.
    - Works even if some rows are missing region or year.
    - Returns None if no valid data remains after filtering.
    - In case of ties on revenue, picks the product_id with the smallest
      lexicographic order (deterministic tie-breaking).
    """
    
    # Filter by year and region, handling potential missing values in those columns
    mask = (
        (df["year"] == year) &
        (df["region"].str.lower() == region.lower())
    )
    subset = df[mask].copy()
    
    if subset.empty:
        return None  # No data for that year/region
    
    # Drop rows with invalid or missing revenue
    subset = subset.dropna(subset=["revenue"])
    
    # Filter out negative or below-threshold revenue
    subset = subset[subset["revenue"] >= min_revenue]
    
    if subset.empty:
        return None  # No usable revenue data
    
    # Aggregate revenue per product_id
    agg = subset.groupby("product_id", as_index=False)["revenue"].sum()
    
    if agg.empty:
        return None
    
    # Sort by revenue desc, then product_id asc for deterministic ties
    agg_sorted = agg.sort_values(by=["revenue", "product_id"],
                                 ascending=[False, True])
    
    top_row = agg_sorted.iloc[0]
    top_product_id = str(top_row["product_id"])
    top_revenue = float(top_row["revenue"])
    
    return top_product_id, top_revenue

if __name__ == "__main__":
    # Example usage:
    csv_path = "sales_data.csv"
    df = load_sales_data(csv_path)
    
    year = 2022
    region = "United States"

    else:
        product_id, revenue = result
        print(f"Top product in {region}, {year} by revenue: {product_id} (revenue={revenue:.2f})")

```python
import pandas as pd
from typing import Optional, Tuple

def load_sales_data(csv_path: str) -> pd.DataFrame:
    df = pd.read_csv(csv_path)
    df.columns = [c.strip().lower() for c in df.columns]
    
    required_cols = {"product_id", "region", "channel", "units", "revenue", "year"}
    missing_cols = required_cols - set(df.columns)
    if missing_cols:
        raise ValueError(f"Missing required columns in CSV: {missing_cols}")
    
    df["revenue"] = pd.to_numeric(df["revenue"], errors="coerce")
    df["units"] = pd.to_numeric(df["units"], errors="coerce")
    df["year"] = pd.to_numeric(df["year"], errors="coerce").astype("Int64")
    df["region"] = df["region"].astype(str).str.strip()
    
    return df

def estimate_missing_revenue(df: pd.DataFrame) -> pd.DataFrame:
    """
    Estimate missing revenue using average revenue-per-unit per product_id.
    Does not modify the input DataFrame in-place; returns a copy.
    """
    df = df.copy()
    
    # Compute avg price per unit for each product where both units and revenue are valid
    valid_mask = df["revenue"].notna() & df["units"].notna() & (df["units"] > 0)
    valid = df[valid_mask]
    
    if valid.empty:
        # No basis for estimation; return df unchanged
        return df
    
    price_per_unit = (valid["revenue"] / valid["units"]).groupby(valid["product_id"]).mean()
    price_per_unit = price_per_unit.rename("avg_price_per_unit")
    
    # Merge avg price back to original df
    df = df.merge(price_per_unit, on="product_id", how="left")
    
    # Identify rows with missing revenue but known units and known avg price
    missing_rev_mask = df["revenue"].isna() & df["units"].notna() & (df["units"] > 0) & df["avg_price_per_unit"].notna()
    
    # Estimate revenue
    df.loc[missing_rev_mask, "revenue"] = (
        df.loc[missing_rev_mask, "units"] * df.loc[missing_rev_mask, "avg_price_per_unit"]
    )
    
    # Drop helper column
    df = df.drop(columns=["avg_price_per_unit"])
    
    return df

def top_product_by_revenue(
    df: pd.DataFrame,
    year: int,
    region: str,
    min_revenue: float = 0.0,
    allow_estimation: bool = True
) -> Optional[Tuple[str, float]]:
    
    # Optionally estimate missing revenue first
    if allow_estimation:
        df = estimate_missing_revenue(df)
    
    mask = (
        (df["year"] == year) &
        (df["region"].str.lower() == region.lower())
    )
    subset = df[mask].copy()
    
    if subset.empty:
        return None
    
    subset = subset.dropna(subset=["revenue"])
    subset = subset[subset["revenue"] >= min_revenue]
    
    if subset.empty:
        return None
    
    agg = subset.groupby("product_id", as_index=False)["revenue"].sum()
    
    if agg.empty:
        return None
    
    agg_sorted = agg.sort_values(by=["revenue", "product_id"],
                                 ascending=[False, True])
    
    top_row = agg_sorted.iloc[0]
    top_product_id = str(top_row["product_id"])
    top_revenue = float(top_row["revenue"])
    
    return top_product_id, top_revenue

if __name__ == "__main__":
    csv_path = "sales_data.csv"
    df = load_sales_data(csv_path)
    
    year = 2022
    region = "United States"
    
    result = top_product_by_revenue(df, year, region, allow_estimation=True)
    
    if result is None:
        print(f"No valid or estimable revenue data for year={year}, region='{region}'.")
    else:
        product_id, revenue = result
        print(f"Top product in {region}, {year} by (actual+estimated) revenue: "
              f"{product_id} (revenue={revenue:.2f})")
```
