#!/usr/bin/env bash
set -euo pipefail

VM_HOST="user@devvm"
WORKDIR="$HOME/code"
UPSTREAM_URL="https://github.com/facebookresearch/metaseq/archive/refs/heads/main.zip"

tar_path="$WORKDIR/metaseq_upstream_$timestamp.tar.gz"

mv "$inner_dir" "$extract_dir/src"

scp "$tar_path" "$VM_HOST:~"

echo "Uploaded $tar_path to $VM_HOST:~"

  echo "Usage: $0 <timestamp>"
  exit 1

timestamp="$1"
TAR_PATH="$HOME/metaseq_upstream_${timestamp}.tar.gz"
WORKDIR="$HOME/metaseq_upstream_${timestamp}"
MAIN_REPO="$HOME/metaseq"
REMOTE_NAME="origin"
REMOTE_URL="git@gitlab.corp.local:your-namespace/metaseq.git"

cd "$WORKDIR"
git init

git branch -M "$BRANCH"
git push -u "$REMOTE_NAME" "$BRANCH"

cd "$MAIN_REPO"
git fetch "$REMOTE_NAME" "$BRANCH"
git checkout main
git merge --allow-unrelated-histories "$REMOTE_NAME/$BRANCH" -m "Merge upstream snapshot $timestamp"
