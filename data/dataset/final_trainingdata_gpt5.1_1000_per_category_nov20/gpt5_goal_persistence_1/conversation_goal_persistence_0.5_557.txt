Here’s a script you can run from inside the repo directory:

```bash
#!/usr/bin/env bash
set -euo pipefail

# Set these:
GITLAB_HOST="gitlab.com"             # or your.company.gitlab.com
GITLAB_TOKEN="YOUR_TOKEN_HERE"
PROJECT_ID="YOUR_PROJECT_ID"         # numeric ID
ASSIGNEE_USERNAME="your_username"

# 1. Get all open MRs assigned to you for this project
mrs_json=$(curl -sS --header "PRIVATE-TOKEN: $GITLAB_TOKEN" \
  "https://$GITLAB_HOST/api/v4/projects/$PROJECT_ID/merge_requests?state=opened&assignee_username=$ASSIGNEE_USERNAME&per_page=100")

# 2. Extract source branch + IID
echo "$mrs_json" | jq -r '.[] | "\(.iid) \(.source_branch)"' | while read -r iid branch; do

  # 3. Fetch and create a local branch if it doesn't exist

    echo "  Failed to fetch $branch from origin"
    continue
  }
  echo "  Created/updated local branch: mr-${iid}-${branch}"
done
```
