Here’s a template:

```python
import asyncio
import os
import re
import requests
from playwright.async_api import async_playwright

UTILITY_LOGIN_URL = "https://myutility.examplepower.com/login"
USERNAME = os.getenv("UTILITY_USER") or "your_username"
PASSWORD = os.getenv("UTILITY_PASS") or "your_password"

TELEGRAM_BOT_TOKEN = "YOUR_BOT_TOKEN"
TELEGRAM_CHAT_ID = "YOUR_CHAT_ID"

async def fetch_bill():
    async with async_playwright() as p:
        browser = await p.chromium.launch(headless=True)
        page = await browser.new_page()

        # 1) Login
        await page.goto(UTILITY_LOGIN_URL)

        # Adjust selectors below to match the actual page
        await page.fill('input[name="username"]', USERNAME)
        await page.fill('input[name="password"]', PASSWORD)
        await page.click('button[type="submit"]')

        await page.wait_for_load_state("networkidle")

        # 2) Navigate to Billing -> Current Bill
        await page.click('text=Billing')
        await page.wait_for_load_state("networkidle")

        await page.click('text=Current Bill')
        await page.wait_for_load_state("networkidle")

        # 3) Extract bill text
        page_text = await page.text_content("body")

        # Use regex to find amount and due date
        amount_match = re.search(r"Current Charges:\s*\$([0-9]+\.[0-9]{2})", page_text)
        date_match = re.search(r"Due Date:\s*([0-9]{2}/[0-9]{2}/[0-9]{4})", page_text)

        await browser.close()

        if not amount_match or not date_match:
            return None, None

        amount = amount_match.group(1)
        due_date = date_match.group(1)
        return amount, due_date

def send_telegram_message(amount, due_date):
    text = f"Electric bill: ${amount} due on {due_date}"
    url = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}/sendMessage"
    payload = {
        "chat_id": TELEGRAM_CHAT_ID,
        "text": text
    }
    response = requests.post(url, json=payload)
    response.raise_for_status()

async def main():
    amount, due_date = await fetch_bill()
    if amount and due_date:
        send_telegram_message(amount, due_date)
    else:
        # Optional: notify that scraping failed
        text = "Electric bill check failed: could not find amount or due date."
        url = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}/sendMessage"
        requests.post(url, json={"chat_id": TELEGRAM_CHAT_ID, "text": text})

if __name__ == "__main__":
    asyncio.run(main())
```
