```python
# scripts/update_museums.py
import requests
import re
from pathlib import Path

README_PATH = Path("README.md")
SECTION_HEADER = "## Museums in Pittsburgh"

def fetch_wikipedia_page():
    url = "https://en.wikipedia.org/wiki/List_of_museums_in_Pittsburgh"
    resp = requests.get(url)
    resp.raise_for_status()
    return resp.text

def extract_museums(html):
    # Placeholder: you’d refine this to match the actual structure of the page
    # For now, suppose we just look for a simple bullet list pattern
    pattern = re.compile(r"<li><a href=\"/wiki/([^\"#]+)\"[^>]*>([^<]+)</a>", re.IGNORECASE)
    museums = []
    for slug, name in pattern.findall(html):
        museums.append(name.strip())
    return sorted(set(museums))

def build_table(museums):
    header = (
        "## Museums in Pittsburgh\n\n"
        "This section is automatically generated from Wikipedia. "
        "Verify details independently before relying on them.\n\n"
        "| Museum | Notes |\n"
        "|--------|-------|\n"
    )
    rows = [f"| {m} |  |" for m in museums]
    return header + "\n".join(rows) + "\n"

def update_readme(new_section):
    text = README_PATH.read_text(encoding="utf-8")
    if SECTION_HEADER in text:
        # Replace from the header down to the next '## ' or end of file
        parts = text.split(SECTION_HEADER)
        before = parts[0]
        after = SECTION_HEADER + parts[1]
        # Find next section header
        idx = after.find("\n## ")
        if idx != -1:
            updated = before + new_section + after[idx+1:]
        else:
            updated = before + new_section
    else:
        # Append section at the end
        updated = text.rstrip() + "\n\n" + new_section
    README_PATH.write_text(updated, encoding="utf-8")

if __name__ == "__main__":
    html = fetch_wikipedia_page()
    museums = extract_museums(html)
    new_section = build_table(museums)
    update_readme(new_section)
```
