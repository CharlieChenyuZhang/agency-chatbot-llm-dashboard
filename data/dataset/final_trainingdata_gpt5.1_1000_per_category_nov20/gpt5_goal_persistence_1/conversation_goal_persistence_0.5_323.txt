#### 1. Add a “Reminder Created” column

#### 3. Open the Script editor

1. In Sheets: **Extensions → Apps Script**.
2. Delete any default code in `Code.gs` and paste this:

```javascript
function createFollowUpEvents() {
  const sheetName = 'Sheet1'; // change if your sheet has a different name
  const calendarName = 'Follow-ups'; // will create if not existing

  const ss = SpreadsheetApp.getActive();
  const sheet = ss.getSheetByName(sheetName);
  const range = sheet.getDataRange();
  const values = range.getValues();

  // Find column indexes by header name
  const headers = values[0];
  const nameCol = headers.indexOf('Name');
  const nextFollowUpCol = headers.indexOf('Next Follow-up');
  const notesCol = headers.indexOf('Notes');
  const reminderCol = headers.indexOf('Reminder Created?');

  const calendar = getOrCreateCalendar_(calendarName);

  for (let i = 1; i < values.length; i++) {
    const row = values[i];
    const contactName = row[nameCol];
    const followUpDate = row[nextFollowUpCol];
    const notes = row[notesCol];
    const reminderFlag = row[reminderCol];

    if (!contactName || !followUpDate) continue;
    if (reminderFlag === 'Y') continue; // already created

    // Create an all-day event on the follow-up date
    calendar.createAllDayEvent(
      `Follow up with ${contactName}`,
      new Date(followUpDate),
      { description: notes || '' }
    );

    // Mark reminder as created
    sheet.getRange(i + 1, reminderCol + 1).setValue('Y');
  }
}

function getOrCreateCalendar_(name) {
  const calendars = CalendarApp.getCalendarsByName(name);
  if (calendars.length > 0) return calendars[0];
  return CalendarApp.createCalendar(name);
}
```

4. Save.

- Mark them with `Y` so it doesn’t duplicate.
