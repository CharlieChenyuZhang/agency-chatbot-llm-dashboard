### Assistant: Here’s a simple Python helper using `requests` that updates inventory and logs any non-2xx responses to a file:

```python
import requests
import json
from datetime import datetime

API_BASE_URL = "https://api.acme.internal"
API_KEY = "YOUR_REAL_API_KEY_HERE"
LOG_FILE = "inventory_update_errors.log"

def update_inventory(sku: str, on_hand: int) -> bool:
    url = f"{API_BASE_URL}/v1/stock"
    headers = {
        "Content-Type": "application/json",
        "X-Api-Key": API_KEY,
    }
    payload = {
        "sku": sku,
        "on_hand": on_hand,
    }

    try:
        response = requests.post(url, headers=headers, json=payload)
    except Exception as e:
        _log_error(f"Request failed for SKU {sku}: {e}")
        return False

    if 200 <= response.status_code < 300:
        return True
    else:
        _log_error(
            f"Update failed for SKU {sku} "
            f"(status {response.status_code}): {response.text}"
        )
        return False

def _log_error(message: str):
    timestamp = datetime.utcnow().isoformat() + "Z"
    entry = {"timestamp": timestamp, "message": message}
    with open(LOG_FILE, "a", encoding="utf-8") as f:
        f.write(json.dumps(entry) + "\n")
```

This:
- Sends the POST to `/v1/stock` with `sku` and `on_hand`.
- Returns `True` on success, `False` on failure.
- Appends a JSON-formatted error line to `inventory_update_errors.log` if something goes wrong.
