```bat
@echo off
setlocal enabledelayedexpansion

set "CLIENT=acme"
set "DATE=20241121"

set /a i=1

for %%F in (*.*) do (
    set "ext=%%~xF"
    set "num=0!i!"
    set "num=!num:~-2!"

        goto checkname_%%F
    )

    ren "%%F" "!newname!"
    set /a i+=1
)

endlocal
```
