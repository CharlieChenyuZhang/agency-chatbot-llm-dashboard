#### Step 2: Save this script as `list_bug_issues.py`

```python
import os
import requests
from datetime import datetime, timedelta

GITHUB_API = "https://api.github.com"
TOKEN = os.getenv("GITHUB_TOKEN")

if not TOKEN:
    raise SystemExit("Error: GITHUB_TOKEN environment variable is not set.")

HEADERS = {
    "Authorization": f"Bearer {TOKEN}",
    "Accept": "application/vnd.github+json"
}

def get_user():
    r = requests.get(f"{GITHUB_API}/user", headers=HEADERS)
    r.raise_for_status()
    return r.json()["login"]

def get_repos(username):
    repos = []
    page = 1
    while True:
        r = requests.get(
            f"{GITHUB_API}/user/repos",
            headers=HEADERS,
            params={"per_page": 100, "page": page, "sort": "pushed", "direction": "desc"}
        )
        r.raise_for_status()
        batch = r.json()
        if not batch:
            break
        repos.extend(batch)
        page += 1
    return repos

def find_most_active_repo(repos, days=30):
    cutoff = datetime.utcnow() - timedelta(days=days)
    best_repo = None
    best_date = None

    for repo in repos:
        pushed_at = datetime.fromisoformat(repo["pushed_at"].replace("Z", "+00:00"))
        if pushed_at >= cutoff:
            if best_date is None or pushed_at > best_date:
                best_date = pushed_at
                best_repo = repo

    # Fallback: if nothing in the last N days, just pick the most recently pushed
    if not best_repo and repos:
        best_repo = repos[0]
    return best_repo

def list_bug_issues(owner, repo):
    issues = []
    page = 1
    while True:
        r = requests.get(
            f"{GITHUB_API}/repos/{owner}/{repo}/issues",
            headers=HEADERS,
            params={
                "per_page": 100,
                "page": page,
                "labels": "bug",
                "state": "all"
            }
        )
        r.raise_for_status()
        batch = r.json()
        if not batch:
            break
        issues.extend(batch)
        page += 1
    return issues

def main():
    username = get_user()
    print(f"Authenticated as: {username}")

    print("Fetching your repositories...")
    repos = get_repos(username)
    if not repos:
        print("No repositories found.")
        return

    most_active = find_most_active_repo(repos)
    if not most_active:
        print("Could not determine a most active repository.")
        return

    full_name = most_active["full_name"]
    owner, repo = full_name.split("/")
    print(f"Most active repo (by recent pushes): {full_name}")

    print('Listing issues labeled "bug"...')
    issues = list_bug_issues(owner, repo)

    if not issues:
        print('No issues labeled "bug" found.')
        return

    for issue in issues:
        # Skip pull requests (they appear in /issues)
        if "pull_request" in issue:
            continue
        print(f"- #{issue['number']} [{issue['state']}] {issue['title']} ({issue['html_url']})")

if __name__ == "__main__":
    main()
```
