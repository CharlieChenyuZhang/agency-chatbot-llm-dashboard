#### 3. Create the script
1. On your Desktop, right-click → **New → Text Document**.
2. Rename it to `organize_photos.py` (make sure it ends with `.py`, not `.txt`).
3. Right-click it → **Open with → Notepad**.
4. Paste this code:

```python
import os
import shutil
from datetime import datetime
from pathlib import Path

from PIL import Image
from PIL.ExifTags import TAGS

SOURCE_DIR = r"E:\Photos_Messy"      # change if needed
TARGET_DIR = r"E:\Photos_Organized"  # change if needed

def get_exif_date(path):
    try:
        img = Image.open(path)
        exif = img._getexif()
        if not exif:
            return None
        for tag_id, value in exif.items():
            tag = TAGS.get(tag_id, tag_id)
            if tag == 'DateTimeOriginal':
                # value like "2020:05:12 14:23:45"
                return datetime.strptime(value, "%Y:%m:%d %H:%M:%S")
    except Exception:
        return None
    return None

def get_file_date(path):
    # fallback: use file modification time
    ts = os.path.getmtime(path)
    return datetime.fromtimestamp(ts)

def organize():
    src = Path(SOURCE_DIR)
    dst = Path(TARGET_DIR)
    dst.mkdir(parents=True, exist_ok=True)

    for root, _, files in os.walk(src):
        for name in files:
            full_path = Path(root) / name
            if not name.lower().endswith((".jpg", ".jpeg", ".png", ".heic")):
                continue

            date = get_exif_date(full_path) or get_file_date(full_path)
            year = date.strftime("%Y")
            month = date.strftime("%Y-%m")

            target_folder = dst / year / month
            target_folder.mkdir(parents=True, exist_ok=True)

            target_path = target_folder / name

            counter = 1
            while target_path.exists():
                stem = target_path.stem
                suffix = target_path.suffix
                target_path = target_folder / f"{stem}_{counter}{suffix}"
                counter += 1

            print(f"Moving {full_path} -> {target_path}")
            shutil.copy2(full_path, target_path)

if __name__ == "__main__":
    organize()
```
