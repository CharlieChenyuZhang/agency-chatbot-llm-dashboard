import csv

}

def fetch_page(page):
    params = {"page": page, "per_page": PER_PAGE}
    r = requests.get(f"{BASE_URL}/customers", headers=headers, params=params)
    r.raise_for_status()
    return r.json()

all_customers = []
page = 1

while True:
    data = fetch_page(page)
    customers = data.get("data") or data  # adjust based on response structure
    if not customers:

    all_customers.extend(customers)
    print(f"Fetched page {page}, got {len(customers)} customers")
    # Stop if there's an obvious last-page indicator
    if len(customers) < PER_PAGE:
        break
    page += 1

if not all_customers:
    print("No customers found.")
    exit()

# Determine CSV columns from keys of first record
fieldnames = sorted(all_customers[0].keys())

with open("customers_export.csv", "w", newline="", encoding="utf-8") as f:
    writer = csv.DictWriter(f, fieldnames=fieldnames)
    writer.writeheader()
    writer.writerows(all_customers)
