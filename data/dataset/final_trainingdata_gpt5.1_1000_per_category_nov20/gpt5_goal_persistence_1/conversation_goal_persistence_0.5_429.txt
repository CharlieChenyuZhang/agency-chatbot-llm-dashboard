Here’s a minimal example script skeleton (you’ll need to fill in your credentials and maybe refine parsing):

```python
from __future__ import print_function
import base64
import os
import re
from email import message_from_bytes
from datetime import datetime

from google.oauth2 import service_account
from googleapiclient.discovery import build
from twilio.rest import Client as TwilioClient

# --- CONFIG ---
GMAIL_USER = 'your_email@gmail.com'
GMAIL_LABEL = 'Bills'
SHEET_ID = 'your_google_sheet_id'
TWILIO_ACCOUNT_SID = 'your_twilio_sid'
TWILIO_AUTH_TOKEN = 'your_twilio_auth_token'
TWILIO_FROM = '+1xxxxxxxxxx'
ALERT_TO = '+1yyyyyyyyyy'

# --- AUTH HELPERS ---

def get_gmail_service():
    # Uses OAuth client_secret.json (installed app credentials)
    from google_auth_oauthlib.flow import InstalledAppFlow
    from google.auth.transport.requests import Request
    import pickle

    SCOPES = ['https://www.googleapis.com/auth/gmail.modify']
    creds = None
    if os.path.exists('token_gmail.pickle'):
        with open('token_gmail.pickle', 'rb') as token:
            creds = pickle.load(token)
    if not creds or not creds.valid:
        if creds and creds.expired and creds.refresh_token:
            creds.refresh(Request())
        else:
            flow = InstalledAppFlow.from_client_secrets_file(
                'credentials_gmail.json', SCOPES)
            creds = flow.run_local_server(port=0)
        with open('token_gmail.pickle', 'wb') as token:
            pickle.dump(creds, token)
    return build('gmail', 'v1', credentials=creds)


def get_sheets_service():
    SCOPES = ['https://www.googleapis.com/auth/spreadsheets']
    creds = service_account.Credentials.from_service_account_file(
        'credentials_sheets.json', scopes=SCOPES)
    return build('sheets', 'v4', credentials=creds)

# --- CORE LOGIC ---

def get_label_id(service, label_name):
    results = service.users().labels().list(userId='me').execute()
    for lbl in results.get('labels', []):
        if lbl['name'] == label_name:
            return lbl['id']
    return None

def fetch_unread_bill_emails(service):
    label_id = get_label_id(service, GMAIL_LABEL)
    if not label_id:
        print(f'Label "{GMAIL_LABEL}" not found.')
        return []

    resp = service.users().messages().list(
        userId='me',
        labelIds=[label_id],
        q='is:unread'
    ).execute()
    messages = resp.get('messages', [])
    return messages

def get_message_body(service, msg_id):
    msg = service.users().messages().get(userId='me', id=msg_id, format='raw').execute()
    msg_bytes = base64.urlsafe_b64decode(msg['raw'].encode('UTF-8'))
    email_msg = message_from_bytes(msg_bytes)

    # try plain text part
    if email_msg.is_multipart():
        for part in email_msg.walk():
            if part.get_content_type() == 'text/plain':
                return part.get_payload(decode=True).decode(errors='ignore')
    else:
        return email_msg.get_payload(decode=True).decode(errors='ignore')

    return ""

def parse_bill_info(text):
    # Very rough examples – you’ll want to tailor to your bills
    amount_match = re.search(r'\$([0-9,]+\.\d{2})', text)
    date_match = re.search(r'(?:due|Due)\s+on\s+([A-Za-z]{3,9}\s+\d{1,2},\s+\d{4})', text)

    amount = float(amount_match.group(1).replace(',', '')) if amount_match else None
    due_str = date_match.group(1) if date_match else None

    due_date = None
    if due_str:
        try:
            due_date = datetime.strptime(due_str, '%B %d, %Y').date()
        except ValueError:
            try:
                due_date = datetime.strptime(due_str, '%b %d, %Y').date()
            except ValueError:
                pass

    return amount, due_date

def append_to_sheet(sheets_service, rows):
    body = {'values': rows}
    sheets_service.spreadsheets().values().append(
        spreadsheetId=SHEET_ID,
        range='A1',
        valueInputOption='USER_ENTERED',
        body=body
    ).execute()

def send_sms_alert(twilio_client, msg):
    twilio_client.messages.create(
        body=msg,
        from_=TWILIO_FROM,
        to=ALERT_TO
    )

def main():
    gmail_service = get_gmail_service()
    sheets_service = get_sheets_service()
    twilio_client = TwilioClient(TWILIO_ACCOUNT_SID, TWILIO_AUTH_TOKEN)

    msgs = fetch_unread_bill_emails(gmail_service)
    if not msgs:
        print("No unread bill emails.")
        return

    rows_to_add = []

    for m in msgs:
        msg_id = m['id']
        full = gmail_service.users().messages().get(
            userId='me', id=msg_id', format='metadata', metadataHeaders=['Subject', 'From']
        ).execute()
        headers = {h['name']: h['value'] for h in full['payload']['headers']}
        subject = headers.get('Subject', '(no subject)')
        sender = headers.get('From', '')

        body = get_message_body(gmail_service, msg_id)
        amount, due_date = parse_bill_info(body)

        now = datetime.now().isoformat(timespec='seconds')
        rows_to_add.append([
            now,
            sender,
            str(due_date) if due_date else '',
            amount if amount is not None else '',
            subject,
            msg_id
        ])

        alert_text = f"Bill: {subject} | From: {sender} | Amount: {amount} | Due: {due_date}"
        send_sms_alert(twilio_client, alert_text)

        # Mark as read
        gmail_service.users().messages().modify(
            userId='me',
            id=msg_id,
            body={'removeLabelIds': ['UNREAD']}
        ).execute()

    if rows_to_add:
        append_to_sheet(sheets_service, rows_to_add)
        print(f"Logged {len(rows_to_add)} bills and sent alerts.")

if __name__ == '__main__':
    main()
```
