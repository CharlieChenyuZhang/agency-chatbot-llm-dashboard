Here’s a very minimal `App.tsx` skeleton with just enough TypeScript to guide you without being strict everywhere:

```tsx
import { useState } from "react";

interface WeatherData {
  city: string;
  temperature: number;
  description: string;
}

function App() {
  const [city, setCity] = useState("");
  const [weather, setWeather] = useState<WeatherData | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const handleSearch = async () => {
    if (!city.trim()) return;

    setLoading(true);
    setError(null);

    try {
      // TODO: replace with real API call
      // Example placeholder:
      const fakeData: WeatherData = {
        city,
        temperature: 22,
        description: "Clear sky",
      };
      await new Promise((res) => setTimeout(res, 800)); // simulate delay
      setWeather(fakeData);
    } catch (err) {
      setError("Failed to fetch weather. Please try again.");
      setWeather(null);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div style={{ minHeight: "100vh", display: "flex", justifyContent: "center", alignItems: "center", padding: "1rem" }}>
      <div style={{ maxWidth: 400, width: "100%" }}>
        <h1>Weather App</h1>

        <div style={{ display: "flex", gap: "0.5rem", marginBottom: "1rem" }}>
          <input
            type="text"
            placeholder="Enter city"
            value={city}
            onChange={(e) => setCity(e.target.value)}
            style={{ flex: 1 }}
          />
          <button onClick={handleSearch} disabled={loading}>
            {loading ? "Loading..." : "Search"}
          </button>
        </div>

        {error && <p style={{ color: "red" }}>{error}</p>}

        {weather && !loading && (
          <div>
            <h2>{weather.city}</h2>
            <p>{weather.temperature}°C</p>
            <p>{weather.description}</p>
          </div>
        )}
      </div>
    </div>
  );
}

export default App;
```
