**Step 3: Use this basic script**

Replace `<YourUser>` in the path with your actual username, then paste this into PowerShell:

```powershell
$base = "C:\Users\<YourUser>\Organized"

# Create folders if they don't exist
$paths = @(
    "$base\Work",
    "$base\Personal",
    "$base\Media\Screenshots",
    "$base\Media\Photos",
    "$base\Software_Installers",
    "$base\_Archive",
    "$base\_Temporary"
)
foreach ($p in $paths) {
    if (-not (Test-Path $p)) {
        New-Item -ItemType Directory -Path $p | Out-Null
    }
}

# Get all files on the Desktop (not touching folders for now)
$files = Get-ChildItem -File

foreach ($f in $files) {
    $ext = $f.Extension.ToLower()

    # Images
    if ($ext -in ".png", ".jpg", ".jpeg", ".gif", ".bmp", ".webp") {
        Move-Item $f.FullName "$base\Media\Screenshots" -Force
        continue
    }

    # Documents
    if ($ext -in ".pdf", ".doc", ".docx", ".xls", ".xlsx", ".ppt", ".pptx", ".txt", ".rtf") {
        # Default: Personal; you can move work items later inside Organized
        Move-Item $f.FullName "$base\Personal" -Force
        continue
    }

    # Installers / archives
    if ($ext -in ".exe", ".msi", ".zip", ".rar", ".7z") {
        Move-Item $f.FullName "$base\Software_Installers" -Force
        continue
    }

    # Everything else → Temporary
    Move-Item $f.FullName "$base\_Temporary" -Force
}
```

Then press Enter to run it.
