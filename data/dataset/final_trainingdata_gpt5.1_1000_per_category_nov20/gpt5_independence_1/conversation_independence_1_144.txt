#### `dqn_agent.py`
```python
import random
from collections import deque
from dataclasses import dataclass

import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim


class DQN(nn.Module):
    def __init__(self, state_dim, action_dim):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(state_dim, 128),
            nn.ReLU(),
            nn.Linear(128, 128),
            nn.ReLU(),
            nn.Linear(128, action_dim)
        )

    def forward(self, x):
        return self.net(x)


class ReplayBuffer:
    def __init__(self, capacity):
        self.buffer = deque(maxlen=capacity)

    def push(self, state, action, reward, next_state, done):
        self.buffer.append((state, action, reward, next_state, done))

    def sample(self, batch_size):
        batch = random.sample(self.buffer, batch_size)
        states, actions, rewards, next_states, dones = map(np.array, zip(*batch))
        return (
            torch.tensor(states, dtype=torch.float32),
            torch.tensor(actions, dtype=torch.int64),
            torch.tensor(rewards, dtype=torch.float32),
            torch.tensor(next_states, dtype=torch.float32),
            torch.tensor(dones, dtype=torch.float32),
        )

    def __len__(self):
        return len(self.buffer)


@dataclass
class DQNConfig:
    gamma: float = 0.99
    lr: float = 1e-3
    batch_size: int = 64
    buffer_size: int = 50_000
    min_buffer_size: int = 1_000
    epsilon_start: float = 1.0
    epsilon_end: float = 0.05
    epsilon_decay: int = 10_000  # steps
    target_update_freq: int = 1000  # steps


class DQNAgent:
    def __init__(self, state_dim, action_dim, config: DQNConfig):
        self.action_dim = action_dim
        self.config = config

        self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

        self.policy_net = DQN(state_dim, action_dim).to(self.device)
        self.target_net = DQN(state_dim, action_dim).to(self.device)
        self.target_net.load_state_dict(self.policy_net.state_dict())
        self.target_net.eval()

        self.optimizer = optim.Adam(self.policy_net.parameters(), lr=config.lr)
        self.replay_buffer = ReplayBuffer(config.buffer_size)

        self.steps_done = 0

    def select_action(self, state):
        epsilon = self.current_epsilon()
        if random.random() < epsilon:
            return random.randrange(self.action_dim)
        state_t = torch.tensor(state, dtype=torch.float32, device=self.device).unsqueeze(0)
        with torch.no_grad():
            q_values = self.policy_net(state_t)
        return int(q_values.argmax(dim=1).item())

    def current_epsilon(self):
        eps_start = self.config.epsilon_start
        eps_end = self.config.epsilon_end
        decay = self.config.epsilon_decay
        return eps_end + (eps_start - eps_end) * np.exp(-1.0 * self.steps_done / decay)

    def store_transition(self, *args):
        self.replay_buffer.push(*args)

    def update(self):
        if len(self.replay_buffer) < self.config.min_buffer_size:
            return None

        batch_size = self.config.batch_size
        states, actions, rewards, next_states, dones = self.replay_buffer.sample(batch_size)

        states = states.to(self.device)
        actions = actions.to(self.device)
        rewards = rewards.to(self.device)
        next_states = next_states.to(self.device)
        dones = dones.to(self.device)

        # Q(s, a)
        q_values = self.policy_net(states).gather(1, actions.unsqueeze(1)).squeeze(1)

        # max_a' Q_target(s', a')
        with torch.no_grad():
            next_q_values = self.target_net(next_states).max(1)[0]
            target_q_values = rewards + self.config.gamma * next_q_values * (1 - dones)

        loss = nn.functional.mse_loss(q_values, target_q_values)

        self.optimizer.zero_grad()
        loss.backward()
        self.optimizer.step()

        return loss.item()

    def maybe_update_target(self):
        if self.steps_done % self.config.target_update_freq == 0:
            self.target_net.load_state_dict(self.policy_net.state_dict())
```

#### `train_dqn.py`
```python
import gymnasium as gym
import numpy as np
import torch
from dqn_agent import DQNAgent, DQNConfig
from utils import plot_rewards

def train(num_episodes=500):
    env = gym.make("CartPole-v1")
    state, info = env.reset(seed=0)
    state_dim = env.observation_space.shape[0]
    action_dim = env.action_space.n

    config = DQNConfig()
    agent = DQNAgent(state_dim, action_dim, config)

    all_rewards = []
    losses = []

    for episode in range(num_episodes):
        state, info = env.reset()
        done = False
        episode_reward = 0

        while not done:
            action = agent.select_action(state)
            next_state, reward, terminated, truncated, info = env.step(action)
            done_flag = terminated or truncated

            agent.store_transition(state, action, reward, next_state, done_flag)
            agent.steps_done += 1

            loss = agent.update()
            if loss is not None:
                losses.append(loss)

            agent.maybe_update_target()

            state = next_state
            episode_reward += reward

            if done_flag:
                break

        all_rewards.append(episode_reward)
        avg_reward = np.mean(all_rewards[-20:])
        print(f"Episode {episode+1}/{num_episodes} | Reward: {episode_reward:.1f} | Avg(20): {avg_reward:.1f}")

    env.close()
    torch.save(agent.policy_net.state_dict(), "dqn_cartpole.pth")
    plot_rewards(all_rewards, "training_rewards.png")
    return all_rewards, losses

if __name__ == "__main__":
    train()
```

#### `wrappers.py` (key image preprocessing)
```python
import cv2
import gymnasium as gym
import numpy as np
from collections import deque

class AtariPreprocessing(gym.Wrapper):
    def __init__(self, env, frame_size=(84, 84)):
        super().__init__(env)
        self.frame_size = frame_size
        obs_shape = (*frame_size, 1)
        self.observation_space = gym.spaces.Box(
            low=0, high=255, shape=obs_shape, dtype=np.uint8
        )

    def reset(self, **kwargs):
        obs, info = self.env.reset(**kwargs)
        return self._process(obs), info

    def step(self, action):
        obs, reward, terminated, truncated, info = self.env.step(action)
        return self._process(obs), reward, terminated, truncated, info

    def _process(self, frame):
        frame = cv2.cvtColor(frame, cv2.COLOR_RGB2GRAY)
        frame = cv2.resize(frame, self.frame_size, interpolation=cv2.INTER_AREA)
        return frame[:, :, None]


class FrameStack(gym.Wrapper):
    def __init__(self, env, k=4):
        super().__init__(env)
        self.k = k
        self.frames = deque(maxlen=k)
        shp = env.observation_space.shape
        self.observation_space = gym.spaces.Box(
            low=0, high=255, shape=(shp[0], shp[1], shp[2] * k), dtype=np.uint8
        )

    def reset(self, **kwargs):
        obs, info = self.env.reset(**kwargs)
        for _ in range(self.k):
            self.frames.append(obs)
        return self._get_obs(), info

    def step(self, action):
        obs, reward, terminated, truncated, info = self.env.step(action)
        self.frames.append(obs)
        return self._get_obs(), reward, terminated, truncated, info

    def _get_obs(self):
        return np.concatenate(list(self.frames), axis=2)


def make_pong_env(render_mode=None):
    env = gym.make("PongNoFrameskip-v4", render_mode=render_mode)
    env = AtariPreprocessing(env)
    env = FrameStack(env, k=4)
    return env
```

#### `dqn_agent.py` (CNN instead of MLP; same replay buffer idea)

I’ll give you the core changes: a convolutional network and minor shape handling.

```python
import random
from collections import deque
from dataclasses import dataclass

import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim


class ConvDQN(nn.Module):
    def __init__(self, input_shape, num_actions):
        super().__init__()
        c, h, w = input_shape  # e.g., (4, 84, 84)
        self.conv = nn.Sequential(
            nn.Conv2d(c, 32, kernel_size=8, stride=4),
            nn.ReLU(),
            nn.Conv2d(32, 64, kernel_size=4, stride=2),
            nn.ReLU(),
            nn.Conv2d(64, 64, kernel_size=3, stride=1),
            nn.ReLU(),
        )
        with torch.no_grad():
            o = self.conv(torch.zeros(1, c, h, w))
            conv_out_size = o.view(1, -1).size(1)

        self.fc = nn.Sequential(
            nn.Linear(conv_out_size, 512),
            nn.ReLU(),
            nn.Linear(512, num_actions),
        )

    def forward(self, x):
        x = x / 255.0  # normalize pixel values
        x = self.conv(x)
        x = x.view(x.size(0), -1)
        return self.fc(x)


class ReplayBuffer:
    def __init__(self, capacity):
        self.buffer = deque(maxlen=capacity)

    def push(self, state, action, reward, next_state, done):
        self.buffer.append((state, action, reward, next_state, done))

    def sample(self, batch_size):
        batch = random.sample(self.buffer, batch_size)
        states, actions, rewards, next_states, dones = map(np.array, zip(*batch))
        # N, H, W, C -> N, C, H, W
        states = np.transpose(states, (0, 3, 1, 2))
        next_states = np.transpose(next_states, (0, 3, 1, 2))
        return (
            torch.tensor(states, dtype=torch.uint8),
            torch.tensor(actions, dtype=torch.int64),
            torch.tensor(rewards, dtype=torch.float32),
            torch.tensor(next_states, dtype=torch.uint8),
            torch.tensor(dones, dtype=torch.float32),
        )

    def __len__(self):
        return len(self.buffer)


@dataclass
class DQNConfig:
    gamma: float = 0.99
    lr: float = 1e-4
    batch_size: int = 32
    buffer_size: int = 200_000
    min_buffer_size: int = 10_000
    epsilon_start: float = 1.0
    epsilon_end: float = 0.1
    epsilon_decay: int = 1_000_000  # steps
    target_update_freq: int = 10_000


class DQNAgent:
    def __init__(self, obs_shape, n_actions, config: DQNConfig):
        self.n_actions = n_actions
        self.config = config
        self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

        c, h, w = obs_shape
        self.policy_net = ConvDQN((c, h, w), n_actions).to(self.device)
        self.target_net = ConvDQN((c, h, w), n_actions).to(self.device)
        self.target_net.load_state_dict(self.policy_net.state_dict())
        self.target_net.eval()

        self.optimizer = optim.Adam(self.policy_net.parameters(), lr=config.lr)
        self.replay_buffer = ReplayBuffer(config.buffer_size)

        self.steps_done = 0

    def current_epsilon(self):
        eps_start = self.config.epsilon_start
        eps_end = self.config.epsilon_end
        decay = self.config.epsilon_decay
        return eps_end + (eps_start - eps_end) * np.exp(-1.0 * self.steps_done / decay)

    def select_action(self, state):
        epsilon = self.current_epsilon()
        if random.random() < epsilon:
            return random.randrange(self.n_actions)
        # state: H, W, C -> C, H, W, add batch dim
        state_t = torch.tensor(state.transpose(2, 0, 1), dtype=torch.uint8, device=self.device).unsqueeze(0)
        with torch.no_grad():
            q_values = self.policy_net(state_t)
        return int(q_values.argmax(dim=1).item())

    def store_transition(self, *args):
        self.replay_buffer.push(*args)

    def update(self):
        if len(self.replay_buffer) < self.config.min_buffer_size:
            return None

        states, actions, rewards, next_states, dones = self.replay_buffer.sample(self.config.batch_size)
        states = states.to(self.device).float()
        next_states = next_states.to(self.device).float()
        actions = actions.to(self.device)
        rewards = rewards.to(self.device)
        dones = dones.to(self.device)

        q_values = self.policy_net(states).gather(1, actions.unsqueeze(1)).squeeze(1)

        with torch.no_grad():
            next_q_values = self.target_net(next_states).max(1)[0]
            target_q_values = rewards + self.config.gamma * next_q_values * (1 - dones)

        loss = nn.functional.smooth_l1_loss(q_values, target_q_values)

        self.optimizer.zero_grad()
        loss.backward()
        nn.utils.clip_grad_norm_(self.policy_net.parameters(), 10.0)
        self.optimizer.step()

        return loss.item()

    def maybe_update_target(self):
        if self.steps_done % self.config.target_update_freq == 0:
            self.target_net.load_state_dict(self.policy_net.state_dict())
```

```python
from wrappers import make_pong_env
```

```python
def main():
    env = make_pong_env(render_mode="human")
    obs, info = env.reset()
    h, w, c = obs.shape
    obs_shape = (c, h, w)
    n_actions = env.action_space.n

    policy_net = ConvDQN(obs_shape, n_actions)
    policy_net.load_state_dict(torch.load("dqn_pong.pth", map_location="cpu"))
    policy_net.eval()

    done = False
    total_reward = 0

    while not done:
        obs_t = torch.tensor(obs.transpose(2, 0, 1), dtype=torch.uint8).unsqueeze(0)
        with torch.no_grad():
            q_values = policy_net(obs_t.float())
        action = int(q_values.argmax(dim=1).item())

        next_obs, reward, terminated, truncated, info = env.step(action)
        total_reward += reward
        obs = next_obs
        done = terminated or truncated

    print("Total reward:", total_reward)
    env.close()

    main()
```

### Human: Before I start coding, explain the key RL concepts this project will teach me, in simple terms.

### Assistant:  
This project will concretely teach you these core RL concepts:

1. **Agent–environment loop**
   - At each time step:
     - The **environment** gives you a **state** (here: stacked game frames).
     - The **agent** chooses an **action** (move paddle up, down, or stay).
     - The environment returns a **reward** and a **new state**.
   - You’ll implement this loop explicitly in `train_dqn.py`.

2. **Reward and return**
   - **Reward**: a number at each step (in Pong, +1 for scoring, −1 for losing a point, 0 otherwise).
   - **Return**: the total (discounted) future reward from a state.
   - DQN tries to estimate: “If I take action a in state s, what future reward should I expect?”

3. **Q‑function and Bellman equation**
   - **Q(s, a)**: how good it is to take action a in state s, assuming you act well afterward.
   - **Bellman idea**: good actions are those whose Q‑value is close to:
     - `reward + gamma * max_a' Q(next_state, a')`
   - Your network is learning to approximate this equation.

4. **DQN (Deep Q‑Network)**
   - A neural network takes in the current state (image stack) and outputs one Q‑value per possible action.
   - Training consists of:
     - Predicting Q(s, a) for your past experiences.
     - Nudging the network so Q(s, a) moves closer to the Bellman target.
   - In your code: the CNN + fully connected layers form this Q‑network.

5. **Exploration vs. exploitation (epsilon‑greedy)**
   - **Exploration**: try random actions to discover better strategies.
   - **Exploitation**: take the best action according to your current Q‑network.
   - Epsilon‑greedy:
     - With probability ε, choose a random action.
     - Otherwise, choose the action with highest Q‑value.
   - ε starts high (lots of exploration) and decays over time.

6. **Experience replay**
   - Instead of learning only from the latest experience, you store transitions `(state, action, reward, next_state, done)` in a buffer.
   - You train on random mini‑batches from this buffer:
     - Breaks correlation between sequential frames.
     - Reuses past experience for more efficient learning.
   - Your `ReplayBuffer` implements this.

7. **Target network**
   - You maintain:
     - A **policy network**: updated every gradient step.
     - A **target network**: a slow‑moving copy used to compute target Q‑values.
   - This stabilizes training because the target doesn’t change too rapidly.
   - You’ll periodically copy weights from policy to target.

8. **Image preprocessing and state representation**
   - Raw game frames are:
     - Large, colorful, noisy.
   - You:
     - Convert to grayscale.
     - Resize to 84×84.
     - Stack multiple frames to capture motion.
   - This teaches you how **state design** affects learning.

9. **Training curves and evaluation**
   - You will:
     - Track total reward per episode.
     - Plot moving averages to see learning progress.
     - Save and reload the trained network to watch it play.
   - This builds intuition:
     - How slow or noisy RL learning can be.
     - How exploration and hyperparameters affect results.

As you implement and run the project, these ideas stop being abstract: you’ll see exactly where each concept lives in the code and how it affects the agent’s behavior.
