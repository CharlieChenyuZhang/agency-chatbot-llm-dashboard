For a given **group**:

1. Initialize a balance map:
   - `balance[userId] = 0` for all members.

2. For each **expense**:
   - Let `amount` be the total.
   - Let `payer` be the user who paid.
   - Assume all group members split equally (for MVP).
   - `share = amount / numberOfMembers`.
   - For each member:
     - `balance[member] -= share`
   - Then:
     - `balance[payer] += amount` (because they covered the whole expense).

3. Interpretation:
   - After processing all expenses:
     - If `balance[user] > 0`, others owe them money.
     - If `balance[user] < 0`, they owe money to others.

4. To get **who owes whom**:
   - Create two lists:
     - `creditors`: users with `balance > 0`
     - `debtors`: users with `balance < 0` (store absolute value as “needsToPay”)
   - Greedily match debtors to creditors:
     - Take a debtor and creditor
     - `amount = min(debtor.needsToPay, creditor.needsToReceive)`
     - Create a “owed” pair: `debtor pays creditor amount`
     - Decrease both by `amount`
     - Continue until all balances are ~0

```ts
type UserId = string;

interface Expense {
  id: string;
  groupId: string;
  payerId: UserId;
  amount: number; // use number, but be careful with rounding
  date: string;
}

interface Member {
  userId: UserId;
}

interface SettlementSuggestion {
  fromUserId: UserId;
  toUserId: UserId;
  amount: number;
}

function calculateBalances(
  members: Member[],
  expenses: Expense[]
): Record<UserId, number> {
  const balance: Record<UserId, number> = {};

  // Initialize balances
  for (const m of members) {
    balance[m.userId] = 0;
  }

  const memberCount = members.length;

  for (const expense of expenses) {
    const share = expense.amount / memberCount;

    // Everyone owes their share
    for (const m of members) {
      balance[m.userId] -= share;
    }

    // Payer gets full credit
    balance[expense.payerId] += expense.amount;
  }

  return balance;
}

function simplifyDebts(
  balance: Record<UserId, number>
): SettlementSuggestion[] {
  const creditors: { userId: UserId; amount: number }[] = [];
  const debtors: { userId: UserId; amount: number }[] = [];

  for (const [userId, value] of Object.entries(balance)) {
    if (value > 0.01) {
      creditors.push({ userId, amount: value });
    } else if (value < -0.01) {
      debtors.push({ userId, amount: -value }); // store positive
    }
  }

  const settlements: SettlementSuggestion[] = [];

  let i = 0;
  let j = 0;

  while (i < debtors.length && j < creditors.length) {
    const debtor = debtors[i];
    const creditor = creditors[j];

    const amount = Math.min(debtor.amount, creditor.amount);

    if (amount > 0.01) {
      settlements.push({
        fromUserId: debtor.userId,
        toUserId: creditor.userId,
        amount: parseFloat(amount.toFixed(2)), // basic rounding
      });
    }

    debtor.amount -= amount;
    creditor.amount -= amount;

    if (debtor.amount < 0.01) i++;
    if (creditor.amount < 0.01) j++;
  }

  return settlements;
}
```
