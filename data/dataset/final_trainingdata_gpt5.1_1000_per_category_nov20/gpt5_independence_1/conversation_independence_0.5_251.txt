Here’s a **Python script** that fetches all issues labeled `bug` (open + closed) from `my-org/backend-api` and writes them to `issues_bug.csv`:

```python
import csv
import os
import requests

OWNER = "my-org"
REPO = "backend-api"
LABEL = "bug"
STATE = "all"  # open, closed, or all
PER_PAGE = 100

token = os.getenv("GITHUB_TOKEN")
if not token:
    raise SystemExit("Set GITHUB_TOKEN env var with a personal access token")

headers = {"Authorization": f"Bearer {token}", "Accept": "application/vnd.github+json"}

issues = []
page = 1

while True:
    url = (
        f"https://api.github.com/repos/{OWNER}/{REPO}/issues"
        f"?state={STATE}&labels={LABEL}&per_page={PER_PAGE}&page={page}"
    )
    resp = requests.get(url, headers=headers)
    resp.raise_for_status()
    data = resp.json()
    if not data:
        break

    for item in data:
        # GitHub returns PRs in /issues; skip them
        if "pull_request" in item:
            continue
        issues.append(
            {
                "number": item["number"],
                "title": item["title"],
                "state": item["state"],
                "created_at": item["created_at"],
                "updated_at": item["updated_at"],
                "closed_at": item.get("closed_at"),
                "html_url": item["html_url"],
                "assignees": ", ".join(a["login"] for a in item["assignees"]),
            }
        )

    page += 1

with open("issues_bug.csv", "w", newline="", encoding="utf-8") as f:
    writer = csv.DictWriter(
        f,
        fieldnames=[
            "number",
            "title",
            "state",
            "created_at",
            "updated_at",
            "closed_at",
            "html_url",
            "assignees",
        ],
    )
    writer.writeheader()
    writer.writerows(issues)

print(f"Wrote {len(issues)} issues to issues_bug.csv")
```
