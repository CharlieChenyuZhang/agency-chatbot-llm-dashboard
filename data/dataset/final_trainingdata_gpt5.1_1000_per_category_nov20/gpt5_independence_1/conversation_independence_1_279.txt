   ```javascript
   function sendDailyCRMReminder() {
     var email = Session.getActiveUser().getEmail(); // or hardcode: "you@example.com"
     var ss = SpreadsheetApp.getActiveSpreadsheet();
     var sheet = ss.getSheetByName("Contacts");

     var lastRow = sheet.getLastRow();
     if (lastRow < 2) return; // no data

     var dataRange = sheet.getRange(2, 1, lastRow - 1, 21); // A2:U
     var data = dataRange.getValues();

     var today = new Date();
     today.setHours(0, 0, 0, 0);

     var dueContacts = [];

     // Column indexes relative to data array (0-based)
     var COL_FULL_NAME = 3;   // D
     var COL_REL_TYPE = 4;    // E
     var COL_PRIORITY = 5;    // F
     var COL_TAGS = 6;        // G
     var COL_EMAIL = 9;       // J
     var COL_LAST_CONTACT = 13; // N
     var COL_NEXT_DUE = 14;   // O
     var COL_FREQ = 15;       // P
     var COL_STATUS = 16;     // Q
     var COL_NOTES = 17;      // R

     for (var i = 0; i < data.length; i++) {
       var row = data[i];
       var status = row[COL_STATUS];
       var nextDue = row[COL_NEXT_DUE];

       if (status !== "Active") continue;
       if (!(nextDue instanceof Date)) continue;

       var dueDate = new Date(nextDue.getTime());
       dueDate.setHours(0, 0, 0, 0);

       if (dueDate <= today) {
         dueContacts.push({
           fullName: row[COL_FULL_NAME],
           relType: row[COL_REL_TYPE],
           priority: row[COL_PRIORITY],
           tags: row[COL_TAGS],
           email: row[COL_EMAIL],
           lastContact: row[COL_LAST_CONTACT],
           nextDue: row[COL_NEXT_DUE],
           freq: row[COL_FREQ],
           notes: row[COL_NOTES]
         });
       }
     }

     if (dueContacts.length === 0) {
       // Optional: skip email if no contacts are due
       return;
     }

     var subject = "Personal CRM – Contacts due today (" + dueContacts.length + ")";
     var body = "Here are your due or overdue contacts:\n\n";

     dueContacts.forEach(function(c, idx) {
       body += (idx + 1) + ". " + (c.fullName || "[No name]") + "\n";
       if (c.relType) body += "   Relationship: " + c.relType + "\n";
       if (c.priority) body += "   Priority: " + c.priority + "\n";
       if (c.tags) body += "   Tags: " + c.tags + "\n";
       if (c.email) body += "   Email: " + c.email + "\n";
       if (c.lastContact instanceof Date) {
         body += "   Last contact: " + c.lastContact.toDateString() + "\n";
       }
       if (c.nextDue instanceof Date) {
         body += "   Next due: " + c.nextDue.toDateString() + "\n";
       }
       if (c.freq) body += "   Frequency (days): " + c.freq + "\n";
       if (c.notes) body += "   Notes: " + c.notes + "\n";
       body += "\n";
     });

     body += "Open your CRM sheet to update Last Contact Date after you reach out.\n";

     MailApp.sendEmail(email, subject, body);
   }
   ```

```javascript
function sendDailyCRMReminder() {
  var email = Session.getActiveUser().getEmail(); // or hardcode
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Contacts");

  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return;

  var dataRange = sheet.getRange(2, 1, lastRow - 1, 21); // A2:U
  var data = dataRange.getValues();

  var today = new Date();
  today.setHours(0, 0, 0, 0);

  var dueContacts = [];

  var COL_FULL_NAME = 3;   // D
  var COL_REL_TYPE = 4;    // E
  var COL_PRIORITY = 5;    // F
  var COL_TAGS = 6;        // G
  var COL_EMAIL = 9;       // J
  var COL_LAST_CONTACT = 13; // N
  var COL_NEXT_DUE = 14;   // O
  var COL_FREQ = 15;       // P
  var COL_STATUS = 16;     // Q
  var COL_NOTES = 17;      // R

  // For coloring
  var bgRange = sheet.getRange(2, 1, lastRow - 1, sheet.getLastColumn());
  var backgrounds = bgRange.getBackgrounds();

  for (var i = 0; i < data.length; i++) {
    var row = data[i];
    var status = row[COL_STATUS];
    var nextDue = row[COL_NEXT_DUE];

    // Default: keep existing background; we’ll adjust for Active rows
    var isActive = (status === "Active");

    if (isActive && nextDue instanceof Date) {
      var dueDate = new Date(nextDue.getTime());
      dueDate.setHours(0, 0, 0, 0);

      if (dueDate <= today) {
        // Overdue or due today: highlight light red
        for (var c = 0; c < backgrounds[i].length; c++) {
          backgrounds[i][c] = "#f4c7c3"; // light red
        }

        dueContacts.push({
          fullName: row[COL_FULL_NAME],
          relType: row[COL_REL_TYPE],
          priority: row[COL_PRIORITY],
          tags: row[COL_TAGS],
          email: row[COL_EMAIL],
          lastContact: row[COL_LAST_CONTACT],
          nextDue: row[COL_NEXT_DUE],
          freq: row[COL_FREQ],
          notes: row[COL_NOTES]
        });
      } else {
        // Active but not yet due: reset to white
        for (var c2 = 0; c2 < backgrounds[i].length; c2++) {
          backgrounds[i][c2] = "#ffffff";
        }
      }
    }
  }

  // Apply background changes
  bgRange.setBackgrounds(backgrounds);

  // If nothing is due, skip email
  if (dueContacts.length === 0) {
    return;
  }

  var subject = "Personal CRM – Contacts due today (" + dueContacts.length + ")";
  var body = "Here are your due or overdue contacts:\n\n";

  dueContacts.forEach(function(c, idx) {
    body += (idx + 1) + ". " + (c.fullName || "[No name]") + "\n";
    if (c.relType) body += "   Relationship: " + c.relType + "\n";
    if (c.priority) body += "   Priority: " + c.priority + "\n";
    if (c.tags) body += "   Tags: " + c.tags + "\n";
    if (c.email) body += "   Email: " + c.email + "\n";
    if (c.lastContact instanceof Date) {
      body += "   Last contact: " + c.lastContact.toDateString() + "\n";
    }
    if (c.nextDue instanceof Date) {

    if (c.freq) body += "   Frequency (days): " + c.freq + "\n";

  });

  body += "Rows highlighted in light red in your Contacts sheet are due or overdue.\n";
  body += "Update Last Contact Date after reaching out to clear the highlight.\n";

  MailApp.sendEmail(email, subject, body);
}
```
