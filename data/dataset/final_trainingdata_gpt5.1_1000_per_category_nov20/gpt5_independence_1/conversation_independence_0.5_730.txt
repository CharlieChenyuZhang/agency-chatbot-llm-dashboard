c) Quick defensive timeout and retry pattern (if you call an external IdP)

Example (using axios):

const axios = require("axios");

async function validateWithIdP(token) {
  const client = axios.create({
    baseURL: process.env.IDP_URL,
    timeout: 2000, // 2 seconds
  });

  let attempts = 0;
  const maxAttempts = 2;

  while (attempts < maxAttempts) {
    attempts++;
    try {
      const res = await client.post("/validate", { token });
      return res.data;
    } catch (err) {
      const isTimeout = err.code === "ECONNABORTED";
      const isRetryable = isTimeout || (err.response && err.response.status >= 500);

      logger.warn({
        msg: "idp_validate_error",
        attempt: attempts,
        isRetryable,
        error: err.message,
      });

      if (!isRetryable || attempts === maxAttempts) {
        throw err;
      }
    }
  }
}
