**scripts/list_oscar_winners.py:**
```python
import json
from pathlib import Path

DATA_PATH = Path(__file__).resolve().parents[1] / "data" / "films.json"

def main():
    with DATA_PATH.open("r", encoding="utf-8") as f:
        films = json.load(f)

    print("Oscar-Winning Christopher Nolan Films:\n")
    for film in sorted(films, key=lambda x: x["year"]):
        title = film["title"]
        year = film["year"]
        oscars = film.get("oscars_won", 0)
        categories = film.get("oscar_categories", [])
        print(f"- {title} ({year}): {oscars} Oscar(s)")
        if categories:
            print(f"  Categories: {', '.join(categories)}")
        print()

if __name__ == "__main__":
    main()
```

**Updated script: `scripts/list_oscar_winners.py`:**
```python
import json
from pathlib import Path

DATA_PATH = Path(__file__).resolve().parents[1] / "data" / "films.json"

def main():
    with DATA_PATH.open("r", encoding="utf-8") as f:
        films = json.load(f)

    print("Christopher Nolan – Oscar Performance by Film\n")

    for film in sorted(films, key=lambda x: x["year"]):
        title = film["title"]
        year = film["year"]

        oscars_won = film.get("oscars_won", 0)
        oscars_nominated = film.get("oscars_nominated", 0)

        won_cats = film.get("oscar_categories_won", [])
        nom_cats = film.get("oscar_categories_nominated", [])

        print(f"- {title} ({year})")
        print(f"  Wins: {oscars_won}  |  Nominations: {oscars_nominated}")

        if won_cats:
            print(f"  Won in: {', '.join(won_cats)}")
        if nom_cats:
            print(f"  Nominated in: {', '.join(nom_cats)}")
        print()

if __name__ == "__main__":
    main()
```
