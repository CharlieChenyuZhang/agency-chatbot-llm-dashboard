  - ε-greedy exploration.

Phase 2: Improved DQN Variants
- Double DQN: Reduce overestimation bias by decoupling action selection and evaluation in the target.
- Dueling DQN: Separate state-value and advantage streams to improve learning of state importance.
- Optional: Prioritized Experience Replay for sampling more informative transitions.

  - Store (state, action, reward, next_state, done) tuples.
  - Sample random mini-batches for training.

Milestone 3 – Training Loop (Vanilla DQN)
- Implement:

  - Loss values.

Milestone 4 – Add Double DQN

  - Use online network to select argmax action for next state.
  - Use target network to evaluate that action.

Milestone 5 – Add Dueling Architecture
- Modify the network:

  - Speed of learning.
  - Final performance.

import gymnasium as gym
import numpy as np
from collections import deque
import cv2

class PreprocessAtari(gym.ObservationWrapper):
    def __init__(self, env, width=84, height=84):
        super().__init__(env)
        self.width = width
        self.height = height
        self.observation_space = gym.spaces.Box(
            low=0, high=255, shape=(self.height, self.width), dtype=np.uint8
        )

    def observation(self, obs):
        # obs: (210, 160, 3) uint8
        gray = cv2.cvtColor(obs, cv2.COLOR_RGB2GRAY)
        resized = cv2.resize(gray, (self.width, self.height), interpolation=cv2.INTER_AREA)
        return resized

class FrameStack(gym.Wrapper):
    def __init__(self, env, k=4):
        super().__init__(env)
        self.k = k
        self.frames = deque(maxlen=k)
        shp = env.observation_space.shape
        self.observation_space = gym.spaces.Box(
            low=0,
            high=255,
            shape=(k, shp[0], shp[1]),
            dtype=np.uint8,
        )

    def reset(self, **kwargs):
        obs, info = self.env.reset(**kwargs)
        for _ in range(self.k):
            self.frames.append(obs)
        return self._get_obs(), info

    def step(self, action):
        obs, reward, terminated, truncated, info = self.env.step(action)
        self.frames.append(obs)
        return self._get_obs(), reward, terminated, truncated, info

    def _get_obs(self):
        return np.stack(self.frames, axis=0)

def make_breakout_env(seed=0):
    env = gym.make("BreakoutNoFrameskip-v4", render_mode=None)
    env.reset(seed=seed)
    # Add wrappers (no-op reset, max-pool, etc. if desired)
    env = PreprocessAtari(env)
    env = FrameStack(env, k=4)
    return env
```

import numpy as np

class ReplayBuffer:
    def __init__(self, capacity, obs_shape, n_frames=4):
        self.capacity = capacity
        self.idx = 0
        self.full = False

        self.states = np.zeros((capacity, *obs_shape), dtype=np.uint8)
        self.actions = np.zeros((capacity,), dtype=np.int64)
        self.rewards = np.zeros((capacity,), dtype=np.float32)
        self.next_states = np.zeros((capacity, *obs_shape), dtype=np.uint8)
        self.dones = np.zeros((capacity,), dtype=bool)

    def add(self, state, action, reward, next_state, done):
        self.states[self.idx] = state
        self.actions[self.idx] = action
        self.rewards[self.idx] = reward
        self.next_states[self.idx] = next_state
        self.dones[self.idx] = done

        self.idx = (self.idx + 1) % self.capacity
        if self.idx == 0:
            self.full = True

    def sample(self, batch_size):
        max_idx = self.capacity if self.full else self.idx
        idxs = np.random.randint(0, max_idx, size=batch_size)
        return (
            self.states[idxs],
            self.actions[idxs],
            self.rewards[idxs],
            self.next_states[idxs],
            self.dones[idxs],
        )

    def __len__(self):
        return self.capacity if self.full else self.idx
```

```python
import torch
import torch.nn.functional as F
import numpy as np
from models import DQN

class DQNAgent:
    def __init__(self, num_actions, cfg, device):
        self.num_actions = num_actions
        self.device = device

        self.gamma = cfg["agent"]["gamma"]
        self.eps_start = cfg["agent"]["eps_start"]
        self.eps_end = cfg["agent"]["eps_end"]
        self.eps_decay_frames = cfg["agent"]["eps_decay_frames"]
        self.learning_rate = cfg["agent"]["lr"]
        self.target_update_freq = cfg["agent"]["target_update_freq"]

        self.policy_net = DQN(num_actions).to(device)
        self.target_net = DQN(num_actions).to(device)
        self.target_net.load_state_dict(self.policy_net.state_dict())
        self.target_net.eval()

        self.optimizer = torch.optim.Adam(self.policy_net.parameters(), lr=self.learning_rate)
        self.step_count = 0

    def select_action(self, state):
        eps = self._epsilon()
        if np.random.rand() < eps:
            return np.random.randint(self.num_actions)
        state_t = torch.from_numpy(state).unsqueeze(0).to(self.device, dtype=torch.float32)
        with torch.no_grad():
            q_values = self.policy_net(state_t)
        return int(q_values.argmax(dim=1).item())

    def _epsilon(self):
        return max(
            self.eps_end,
            self.eps_start - (self.eps_start - self.eps_end) * self.step_count / self.eps_decay_frames
        )

    def update(self, batch, cfg):
        states, actions, rewards, next_states, dones = batch
        self.step_count += 1

        states = torch.from_numpy(states).to(self.device, dtype=torch.float32)
        next_states = torch.from_numpy(next_states).to(self.device, dtype=torch.float32)
        actions = torch.from_numpy(actions).to(self.device, dtype=torch.long)
        rewards = torch.from_numpy(rewards).to(self.device, dtype=torch.float32)
        dones = torch.from_numpy(dones.astype(np.float32)).to(self.device, dtype=torch.float32)

        # Compute current Q
        q_values = self.policy_net(states)
        q_values = q_values.gather(1, actions.unsqueeze(1)).squeeze(1)

        # Compute target Q
        with torch.no_grad():
            next_q_values = self.target_net(next_states).max(dim=1)[0]
            target = rewards + (1.0 - dones) * cfg["agent"]["gamma"] * next_q_values

        loss = F.smooth_l1_loss(q_values, target)

        self.optimizer.zero_grad()
        loss.backward()
        torch.nn.utils.clip_grad_norm_(self.policy_net.parameters(), cfg["agent"]["grad_clip"])
        self.optimizer.step()

        if self.step_count % self.target_update_freq == 0:
            self.target_net.load_state_dict(self.policy_net.state_dict())

        return loss.item()
```

src/train.py
```python
import os
from pathlib import Path
import time

import gymnasium as gym
import numpy as np
import torch
from tqdm import tqdm

from env_wrappers import make_cartpole_env
from replay_buffer import ReplayBuffer
from dqn_agent import DQNAgent

def train(cfg):
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

    # Environment
    env = make_cartpole_env(seed=cfg["env"]["seed"])
    state_dim = env.observation_space.shape[0]
    num_actions = env.action_space.n

    # Replay buffer
    buffer = ReplayBuffer(
        capacity=cfg["replay_buffer"]["capacity"],
        state_dim=state_dim,
    )

    # Agent
    agent = DQNAgent(
        state_dim=state_dim,
        num_actions=num_actions,
        cfg=cfg,
        device=device,
    )

    # Checkpoints / logs
    checkpoints_dir = Path(cfg["logging"]["checkpoints_dir"])
    checkpoints_dir.mkdir(parents=True, exist_ok=True)

    total_steps = 0
    episode_rewards = []

    observation, info = env.reset()
    state = observation
    episode_reward = 0.0
    episode = 0

    pbar = tqdm(total=cfg["train"]["max_frames"], desc="Training", ncols=100)

    while total_steps < cfg["train"]["max_frames"]:
        # Select action
        action = agent.select_action(state)

        # Step environment
        next_state, reward, terminated, truncated, info = env.step(action)
        done = terminated or truncated

        # Store transition
        buffer.add(state, action, reward, next_state, done)

        state = next_state
        episode_reward += reward
        total_steps += 1
        pbar.update(1)

        # Learning step
        if len(buffer) > cfg["train"]["learning_starts"] and \
           total_steps % cfg["train"]["train_freq"] == 0:
            batch = buffer.sample(cfg["train"]["batch_size"])
            loss = agent.update(batch, cfg)

        # End of episode
        if done:
            episode += 1
            episode_rewards.append(episode_reward)

            if episode % cfg["logging"]["log_interval_episodes"] == 0:
                avg_reward = np.mean(episode_rewards[-cfg["logging"]["log_interval_episodes"]:])
                print(
                    f"Episode {episode} | Step {total_steps} | "
                    f"Episode Reward: {episode_reward:.2f} | "
                    f"Avg Reward: {avg_reward:.2f}"
                )

            # Reset env
            observation, info = env.reset()
            state = observation
            episode_reward = 0.0

        # Checkpointing
        if total_steps % cfg["logging"]["checkpoint_interval"] == 0:
            ckpt_path = checkpoints_dir / f"dqn_cartpole_{total_steps}.pt"
            torch.save(
                {
                    "policy_net": agent.policy_net.state_dict(),
                    "target_net": agent.target_net.state_dict(),
                    "optimizer": agent.optimizer.state_dict(),
                    "total_steps": total_steps,
                    "episode": episode,
                },
                ckpt_path,
            )

    pbar.close()
    env.close()
```

- Loads a saved checkpoint.
- Runs a fixed number of episodes with near-greedy policy (ε ≈ 0).

src/eval.py
```python
from pathlib import Path
import torch
import numpy as np

from env_wrappers import make_cartpole_env
from dqn_agent import DQNAgent

def evaluate(cfg, checkpoint_path, num_episodes=20):
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

    env = make_cartpole_env(seed=cfg["env"]["seed"])
    state_dim = env.observation_space.shape[0]
    num_actions = env.action_space.n

    # Initialize agent
    agent = DQNAgent(
        state_dim=state_dim,
        num_actions=num_actions,
        cfg=cfg,
        device=device,
    )

    # Load checkpoint
    checkpoint_path = Path(checkpoint_path)
    ckpt = torch.load(checkpoint_path, map_location=device)
    agent.policy_net.load_state_dict(ckpt["policy_net"])
    agent.target_net.load_state_dict(ckpt["target_net"])

    # Disable exploration by overriding epsilon schedule
    agent.eps_start = 0.0
    agent.eps_end = 0.0
    agent.eps_decay_frames = 1  # irrelevant now

    rewards = []

    for ep in range(num_episodes):
        obs, info = env.reset()
        state = obs
        done = False
        ep_reward = 0.0

        while not done:
            action = agent.select_action(state)
            next_state, reward, terminated, truncated, info = env.step(action)
            done = terminated or truncated
            state = next_state
            ep_reward += reward

        rewards.append(ep_reward)
        print(f"Eval Episode {ep + 1}: Reward = {ep_reward:.2f}")

    avg_reward = np.mean(rewards)
    std_reward = np.std(rewards)
    print(f"\nAverage Reward over {num_episodes} episodes: {avg_reward:.2f} ± {std_reward:.2f}")

    env.close()
```

- Train first:

- Evaluate a checkpoint (e.g., at 50000 steps):
  - `python -m src.main --config config/default.yaml --mode eval --checkpoint checkpoints/dqn_cartpole_50000.pt`
