popup.js (very basic wiring + status message)
```js
let selectedMinutes = 20; // default

document.addEventListener("DOMContentLoaded", () => {
  const durationButtons = document.querySelectorAll(".durations button");
  const startButton = document.getElementById("start-session");
  const goalInput = document.getElementById("goal");
  const statusEl = document.getElementById("status");

  durationButtons.forEach((btn) => {
    btn.addEventListener("click", () => {
      selectedMinutes = parseInt(btn.dataset.minutes, 10);
      durationButtons.forEach((b) => b.classList.remove("selected"));
      btn.classList.add("selected");
    });
  });

  startButton.addEventListener("click", () => {
    const goal = goalInput.value.trim();
    if (!goal) {
      statusEl.textContent = "Please enter a goal.";
      return;
    }

    const durationMs = selectedMinutes * 60 * 1000;

    chrome.storage.sync.set(
      {
        activeSession: {
          goal,
          durationMs,
          startTime: Date.now()
        }
      },
      () => {
        statusEl.textContent = `Session started for ${selectedMinutes} minutes.`;
      }
    );
  });
});
```
