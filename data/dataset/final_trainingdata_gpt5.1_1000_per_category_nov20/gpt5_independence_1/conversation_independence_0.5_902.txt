```powershell
# backup-weekly.ps1

$UserProfile = "C:\Users\John"
$OneDrivePath = Join-Path $UserProfile "OneDrive"
$BackupRoot = Join-Path $OneDrivePath "Backups"
$LogFile = Join-Path $BackupRoot "backup-log.txt"

# Simple function to log messages
function Write-Log {
    param (
        [string]$Message
    )
    $timestamp = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
    $entry = "[$timestamp] $Message"
    Add-Content -Path $LogFile -Value $entry
}

# Try/catch to handle errors
try {
    if (-not (Test-Path $BackupRoot)) {
        New-Item -ItemType Directory -Path $BackupRoot | Out-Null
    }

    $FoldersToBackup = @(
        Join-Path $UserProfile "Documents"
        Join-Path $UserProfile "Desktop"
        Join-Path $UserProfile "Pictures"
        Join-Path $UserProfile "Work"
    )

    $DateStamp = Get-Date -Format "yyyy-MM-dd_HH-mm"
    $BackupFileName = "weekly-backup-$DateStamp.zip"
    $BackupFilePath = Join-Path $BackupRoot $BackupFileName

    Compress-Archive -Path $FoldersToBackup -DestinationPath $BackupFilePath -Force

    $MaxBackups = 12
    $existingBackups = Get-ChildItem -Path $BackupRoot -Filter "weekly-backup-*.zip" |
                       Sort-Object LastWriteTime -Descending

    if ($existingBackups.Count -gt $MaxBackups) {
        $toDelete = $existingBackups | Select-Object -Skip $MaxBackups
        $toDelete | Remove-Item -Force
    }

    Write-Log "Backup succeeded: $BackupFileName"

    # Show popup if user is logged in (best-effort)
    try {
        [reflection.assembly]::LoadWithPartialName('System.Windows.Forms') | Out-Null
        [System.Windows.Forms.MessageBox]::Show(
            "Weekly backup completed successfully.`nFile: $BackupFileName",
            "Backup Completed",
            'OK',
            'Information'
        ) | Out-Null
    } catch {
        # Ignore popup failures
    }

} catch {
    $errorMessage = $_.Exception.Message
    Write-Log "Backup FAILED: $errorMessage"

    # Show popup if possible
    try {
        [reflection.assembly]::LoadWithPartialName('System.Windows.Forms') | Out-Null
        [System.Windows.Forms.MessageBox]::Show(
            "Weekly backup FAILED.`nError: $errorMessage",
            "Backup Failed",
            'OK',
            'Error'
        ) | Out-Null
    } catch {
        # Ignore popup failures
    }

    exit 1
}
```
