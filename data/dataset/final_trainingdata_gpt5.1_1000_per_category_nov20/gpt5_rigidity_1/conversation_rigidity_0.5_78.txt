**Example code for Q1: Age groups and subscription rate**

```python
import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt

# Assuming df is already loaded:
# df = pd.read_csv("bank.csv")

# 1. Create age groups
bins = [0, 29, 39, 49, 59, 120]
labels = ['<30', '30-39', '40-49', '50-59', '60+']
df['age_group'] = pd.cut(df['age'], bins=bins, labels=labels, right=True)

# 2. Encode target as 0/1 for easier averaging
df['subscribed'] = (df['y'] == 'yes').astype(int)

# 3. Compute subscription rate by age group
age_conv = (
    df.groupby('age_group')['subscribed']
      .agg(['mean', 'count'])
      .reset_index()
      .rename(columns={'mean': 'conversion_rate', 'count': 'n_customers'})
)

print(age_conv)

# 4. Plot conversion rate by age group
plt.figure(figsize=(8, 5))
sns.barplot(
    data=age_conv,
    x='age_group',
    y='conversion_rate',
    color='steelblue'
)
plt.title('Subscription Rate by Age Group')
plt.xlabel('Age Group')
plt.ylabel('Subscription Rate')
plt.ylim(0, age_conv['conversion_rate'].max() * 1.2)
plt.show()
```
