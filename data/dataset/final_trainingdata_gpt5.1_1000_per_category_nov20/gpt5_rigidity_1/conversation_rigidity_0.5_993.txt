```python
# dqn_agent.py
import random
from collections import deque

import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim


class QNetwork(nn.Module):
    def __init__(self, state_dim, action_dim, hidden_dim=128):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(state_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, action_dim),
        )

    def forward(self, x):
        return self.net(x)


class DQNAgent:
    def __init__(
        self,
        state_dim,
        action_dim,
        gamma=0.99,
        lr=1e-3,
        batch_size=64,
        buffer_size=50_000,
        min_replay_size=1_000,
        eps_start=1.0,
        eps_end=0.05,
        eps_decay=500,  # roughly episodes or steps depending how you use it
        target_update_freq=1000,
        device=None,
    ):
        self.action_dim = action_dim
        self.gamma = gamma
        self.batch_size = batch_size
        self.min_replay_size = min_replay_size
        self.target_update_freq = target_update_freq

        self.device = device or ("cuda" if torch.cuda.is_available() else "cpu")

        self.q_net = QNetwork(state_dim, action_dim).to(self.device)
        self.target_q_net = QNetwork(state_dim, action_dim).to(self.device)
        self.target_q_net.load_state_dict(self.q_net.state_dict())
        self.target_q_net.eval()

        self.optimizer = optim.Adam(self.q_net.parameters(), lr=lr)

        self.replay_buffer = deque(maxlen=buffer_size)
        self.steps_done = 0

        # epsilon-greedy parameters
        self.eps_start = eps_start
        self.eps_end = eps_end
        self.eps_decay = eps_decay

    def select_action(self, state, eval_mode=False):
        """state: np.array shape (state_dim,)"""
        if eval_mode:
            eps = 0.0
        else:
            eps = self.eps_end + (self.eps_start - self.eps_end) * \
                  np.exp(-1.0 * self.steps_done / self.eps_decay)

        self.steps_done += 1

        if random.random() < eps:
            return random.randrange(self.action_dim)
        else:
            state_t = torch.as_tensor(state, dtype=torch.float32, device=self.device).unsqueeze(0)
            with torch.no_grad():
                q_values = self.q_net(state_t)
            return int(q_values.argmax(dim=1).item())

    def store_transition(self, state, action, reward, next_state, done):
        self.replay_buffer.append((state, action, reward, next_state, done))

    def can_update(self):
        return len(self.replay_buffer) >= self.min_replay_size

    def update(self):
        if not self.can_update():
            return

        batch = random.sample(self.replay_buffer, self.batch_size)
        states, actions, rewards, next_states, dones = map(np.array, zip(*batch))

        states_t = torch.as_tensor(states, dtype=torch.float32, device=self.device)
        actions_t = torch.as_tensor(actions, dtype=torch.int64, device=self.device).unsqueeze(-1)
        rewards_t = torch.as_tensor(rewards, dtype=torch.float32, device=self.device).unsqueeze(-1)
        next_states_t = torch.as_tensor(next_states, dtype=torch.float32, device=self.device)
        dones_t = torch.as_tensor(dones, dtype=torch.float32, device=self.device).unsqueeze(-1)

        # Q(s, a)
        q_values = self.q_net(states_t).gather(1, actions_t)

        # max_a' Q_target(s', a')
        with torch.no_grad():
            next_q_values = self.target_q_net(next_states_t).max(dim=1, keepdim=True)[0]
            target_q = rewards_t + self.gamma * (1 - dones_t) * next_q_values

        loss = nn.functional.mse_loss(q_values, target_q)

        self.optimizer.zero_grad()
        loss.backward()
        self.optimizer.step()

        # Periodically update target network
        if self.steps_done % self.target_update_freq == 0:
            self.target_q_net.load_state_dict(self.q_net.state_dict())
```

For simplicity, this uses a single `steps_done` counter for both epsilon decay and target network updates. If you’d rather tie those to environment steps instead of action selections, you can increment it in your training loop instead and pass it in.

```python
# main.py
import gym
import numpy as np
import torch

from dqn_agent import DQNAgent


def train_cartpole(
    env_id="CartPole-v1",
    num_episodes=500,
    max_steps_per_episode=500,
    render=False,  # set True only to watch, it slows training
):
    env = gym.make(env_id)
    state_dim = env.observation_space.shape[0]
    action_dim = env.action_space.n

    agent = DQNAgent(
        state_dim=state_dim,
        action_dim=action_dim,
        gamma=0.99,
        lr=1e-3,
        batch_size=64,
        buffer_size=50_000,
        min_replay_size=1_000,
        eps_start=1.0,
        eps_end=0.05,
        eps_decay=10_000,   # decay over ~10k steps
        target_update_freq=1000,
    )

    rewards_history = []
    best_mean_reward = -float("inf")
    best_model_path = "best_cartpole_dqn.pt"

    for episode in range(num_episodes):
        state, _ = env.reset(return_info=True) if hasattr(env, "reset") else (env.reset(), {})
        episode_reward = 0

        for t in range(max_steps_per_episode):
            if render:
                env.render()

            action = agent.select_action(state)
            next_state, reward, done, info = env.step(action)

            # CartPole gives +1 for each timestep alive
            agent.store_transition(state, action, reward, next_state, done)

            agent.update()

            state = next_state
            episode_reward += reward

            if done:
                break

        rewards_history.append(episode_reward)

        # Simple moving average over last 20 episodes
        if len(rewards_history) >= 20:
            mean_reward = np.mean(rewards_history[-20:])
        else:
            mean_reward = np.mean(rewards_history)

        print(
            f"Episode {episode+1}/{num_episodes} | "
            f"Reward: {episode_reward:.1f} | "
            f"Mean(20): {mean_reward:.1f} | "
            f"Buffer: {len(agent.replay_buffer)}"
        )

        # Save best model
        if mean_reward > best_mean_reward and len(rewards_history) >= 20:
            best_mean_reward = mean_reward
            torch.save(agent.q_net.state_dict(), best_model_path)

    env.close()
    print(f"Training finished. Best 20-ep mean reward: {best_mean_reward:.1f}")
    print(f"Best model saved to: {best_model_path}")


if __name__ == "__main__":
    train_cartpole()
```

### Assistant: Sure, I’ll focus on the structure rather than full code, so you implement it yourself and really internalize it.

**1. Environment**

- Use `FrozenLake-v1` from Gym/Gymnasium.
- Start with `is_slippery=True` (the default), even though it’s a bit harder; it shows stochastic transitions.

Key properties:

- Observation space: discrete states `0..(N-1)`.
- Action space: discrete actions `0..3` (LEFT, DOWN, RIGHT, UP).

**2. Q-table initialization**

- Create a 2D NumPy array: `Q = np.zeros((num_states, num_actions))`.
- `num_states = env.observation_space.n`
- `num_actions = env.action_space.n`

**3. Hyperparameters**

Typical starting values:

- Learning rate: `alpha = 0.8`
- Discount factor: `gamma = 0.95`
- Exploration:
  - `epsilon_start = 1.0`
  - `epsilon_end = 0.01`
  - Some decay schedule over episodes (e.g., linear or exponential).

**4. Action selection (ε-greedy)**

For each step:

- With probability ε: choose a random action.
- Otherwise: choose `argmax_a Q[state, a]`.

You can decay ε per episode: `epsilon = max(epsilon_end, epsilon_start * decay_rate**episode)`.

**5. Q-update rule**

After taking action `a` in state `s`:

- You observe `reward`, `next_state`, and `done`.
- Update:
  - `best_next = max_a' Q[next_state, a']`
  - `td_target = reward + gamma * best_next * (1 - done)`
  - `td_error = td_target - Q[s, a]`
  - `Q[s, a] += alpha * td_error`

**6. Training loop**

For each episode:

1. `state = env.reset()`
2. For each step (up to a max step count):
   - Choose action with ε-greedy.
   - Step env: `next_state, reward, done, info = env.step(action)`.
   - Apply Q-update.
   - Set `state = next_state`.
   - If `done`, break.
3. Optionally track total reward per episode to see learning progress.

**7. Evaluation**

- After training, run several episodes with `epsilon = 0` (always greedy).
- Count the success rate (reaching the goal).
- Optionally, render a few episodes to see the agent’s behavior.

**Baseline hyperparameters for FrozenLake-v1 (4x4, slippery):**

- Episodes: `num_episodes = 5000`
- Max steps per episode: `max_steps = 100`
- Learning rate: `alpha = 0.8`
- Discount factor: `gamma = 0.95`

**Epsilon schedule (simple and effective):**

Option A – Exponential decay (common and easy):

- `epsilon_start = 1.0`
- `epsilon_end = 0.01`
- `decay_rate = 0.001` (tune this if exploration lasts too long or too short)

Per episode:

```python
epsilon = epsilon_end + (epsilon_start - epsilon_end) * np.exp(-decay_rate * episode)
```

Option B – Piecewise (even simpler to reason about):

- Start high, then step down in chunks:

```python
if episode < 1000:
    epsilon = 1.0
elif episode < 3000:
    epsilon = 0.1
else:
    epsilon = 0.01
```

This isn’t fancy but is very understandable: lots of exploration early, then mostly exploitation with some randomness, then almost fully greedy near the end.

**What to look for:**

- If after ~2000–3000 episodes your success rate is still basically random:
  - Try lowering `alpha` to `0.5` (sometimes too high LR causes instability).
  - Try making `epsilon` drop faster (less exploration late), but keep it nonzero.
- If learning is extremely slow:
  - Increase `num_episodes` to 10,000.
  - Slightly increase `alpha` (e.g., 0.9), but monitor for instability.

You can start with:

- `alpha = 0.8`
- `gamma = 0.95`
- Piecewise epsilon (Option B)
