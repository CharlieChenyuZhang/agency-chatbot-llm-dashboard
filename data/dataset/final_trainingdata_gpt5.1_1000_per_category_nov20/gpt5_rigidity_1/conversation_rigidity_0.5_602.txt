Given that, I suggest using the API rather than raw scraping. Here’s a conceptual Python approach using the API (simplified, not production-ready):

```python
# Pseudocode – requires valid Amazon Product Advertising API keys
from amazon_paapi import AmazonApi

api = AmazonApi(ACCESS_KEY, SECRET_KEY, PARTNER_TAG, "US")

search_term = "bluetooth headphones"
max_price_cents = 5000  # $50.00

items = api.search_items(
    keywords=search_term,
    item_count=50,
    resources=[
        "ItemInfo.Title",
        "Offers.Listings.Price",
        "CustomerReviews.Count",
        "CustomerReviews.StarRating"
    ]
)

filtered = []
for item in items:
    price = item.offers.listings[0].price.amount
    rating = item.customer_reviews.star_rating  # 0–5
    if price <= 50 and rating >= 4.0:
        filtered.append({
            "title": item.item_info.title,
            "price": price,
            "rating": rating,
            "reviews": item.customer_reviews.count
        })

# Sort by rating then review count
filtered.sort(key=lambda x: (-x["rating"], -x["reviews"]))

for i, p in enumerate(filtered[:10], start=1):
    print(f"{i}. {p['title']} - ${p['price']} - {p['rating']}★ ({p['reviews']} reviews)")
```
