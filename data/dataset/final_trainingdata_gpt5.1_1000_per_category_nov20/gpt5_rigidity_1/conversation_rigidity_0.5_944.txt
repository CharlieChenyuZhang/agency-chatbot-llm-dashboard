2) Example script: `auto_tag_docs.py`

```python
import os
import re
import yaml
from pathlib import Path
from textwrap import dedent
from openai import OpenAI

# Configure
DOCS_DIR = Path("docs")
MODEL = "gpt-4o-mini"  # adjust as needed

TAG_LIST = [
    "Onboarding", "Product", "Architecture", "DevOps",
    "Backend", "Frontend", "Operations", "Sales",
    "Support", "How-To", "Concept", "API", "Security"
]

client = OpenAI()

FRONT_MATTER_PATTERN = re.compile(r"^---\n(.*?)\n---\n(.*)$", re.DOTALL)

def extract_front_matter(content):
    match = FRONT_MATTER_PATTERN.match(content)
    if match:
        return yaml.safe_load(match.group(1)) or {}, match.group(2)
    else:
        return {}, content

def build_content_for_model(body, max_chars=3000):
    # Truncate to avoid huge prompts
    return body[:max_chars]

def suggest_tags(body):
    prompt = dedent(f"""
    You are helping classify documentation pages for an internal knowledge base.

    You are given:
    - A fixed list of allowed tags.
    - The raw content of a documentation page.

    Allowed tags:
    {", ".join(TAG_LIST)}

    Task:
    - Choose up to 3 tags from the allowed list that best describe the page.
    - Return ONLY a JSON array of strings, e.g.: ["Onboarding", "Product"]

    Content:
    \"\"\"
    {body}
    \"\"\"
    """)

    resp = client.chat.completions.create(
        model=MODEL,
        messages=[{"role": "user", "content": prompt}],
        temperature=0
    )

    text = resp.choices[0].message.content.strip()
    # Very simple parse, assuming the model returns a JSON-like array
    try:
        tags = yaml.safe_load(text)
        if isinstance(tags, list):
            # Keep only allowed tags
            return [t for t in tags if t in TAG_LIST][:3]
    except Exception:
        pass

    return []

def update_file(path: Path):
    content = path.read_text(encoding="utf-8")
    fm, body = extract_front_matter(content)

    short_body = build_content_for_model(body)
    tags = suggest_tags(short_body)

    if not tags:
        print(f"[SKIP] No tags suggested for {path}")
        return

    fm["tags"] = tags

    new_front_matter = "---\n" + yaml.safe_dump(fm, sort_keys=False) + "---\n"
    new_content = new_front_matter + body

    path.write_text(new_content, encoding="utf-8")
    print(f"[OK] Updated tags for {path}: {tags}")

def main():
    for ext in ("*.md", "*.mdx"):
        for path in DOCS_DIR.rglob(ext):
            update_file(path)

if __name__ == "__main__":
    main()
```
