pnpm add -D typescript ts-node @types/node @types/express nodemon

import express from "express";

const app = express();
const port = process.env.PORT || 4000;

app.use(express.json());

app.get("/api/health", (_req, res) => {
  res.json({ status: "ok" });
});

app.listen(port, () => {
  console.log(`API server listening on port ${port}`);
});
EOF

# 6. Basic scripts in backend package.json
node -e "const fs=require('fs');const pkg=JSON.parse(fs.readFileSync('package.json','utf8'));pkg.scripts={dev:'nodemon src/index.ts',build:'tsc',start:'node dist/index.js'};fs.writeFileSync('package.json',JSON.stringify(pkg,null,2));"

# 7. (Optional) Root scripts to run both later
cd ../..
node -e "const fs=require('fs');const pkg=JSON.parse(fs.readFileSync('package.json','utf8'));pkg.scripts={dev:'pnpm --filter frontend dev & pnpm --filter backend dev'};fs.writeFileSync('package.json',JSON.stringify(pkg,null,2));"
```

import os
import asyncpg
from typing import Optional

DATABASE_URL = os.getenv("DATABASE_URL", "postgresql://user:password@localhost:5432/mydb")

_pool: Optional[asyncpg.pool.Pool] = None

async def get_pool() -> asyncpg.pool.Pool:
    global _pool
    if _pool is None:
        _pool = await asyncpg.create_pool(DATABASE_URL)
    return _pool

async def close_pool() -> None:
    global _pool
    if _pool is not None:
        await _pool.close()
        _pool = None
```

Short explanation: Connection pool utilities for PostgreSQL using asyncpg.
