`backend/src/modules/auth/auth.service.ts`:

```ts
import bcrypt from 'bcryptjs';
import jwt from 'jsonwebtoken';
import { prisma } from '../../db/client';

const JWT_SECRET = process.env.JWT_SECRET || 'dev-secret-change-me';

interface AuthPayload {
  email: string;
  password: string;
  name?: string;
}

async function register(data: AuthPayload) {
  const existing = await prisma.user.findUnique({ where: { email: data.email } });
  if (existing) {
    const err: any = new Error('Email already in use');
    err.status = 400;
    throw err;
  }

  const hashed = await bcrypt.hash(data.password, 10);
  const user = await prisma.user.create({
    data: {
      email: data.email,
      password: hashed,
      name: data.name
    },
    select: { id: true, email: true, name: true, createdAt: true }
  });

  const token = jwt.sign({ sub: user.id }, JWT_SECRET, { expiresIn: '7d' });

  return { user, token };
}

async function login(data: AuthPayload) {
  const user = await prisma.user.findUnique({ where: { email: data.email } });
  if (!user) {
    const err: any = new Error('Invalid credentials');
    err.status = 401;
    throw err;
  }

  const match = await bcrypt.compare(data.password, user.password);
  if (!match) {
    const err: any = new Error('Invalid credentials');
    err.status = 401;
    throw err;
  }

  const token = jwt.sign({ sub: user.id }, JWT_SECRET, { expiresIn: '7d' });

  return {
    user: {
      id: user.id,
      email: user.email,
      name: user.name,
      createdAt: user.createdAt
    },
    token
  };
}

export const authService = { register, login };
```

`backend/src/middleware/authMiddleware.ts`:

```ts
import { Request, Response, NextFunction } from 'express';
import jwt from 'jsonwebtoken';

const JWT_SECRET = process.env.JWT_SECRET || 'dev-secret-change-me';

export interface AuthRequest extends Request {
  userId?: string;
}

export function requireAuth(req: AuthRequest, res: Response, next: NextFunction) {
  const header = req.headers.authorization;
  if (!header?.startsWith('Bearer ')) {
    return res.status(401).json({ message: 'Missing or invalid authorization header' });
  }

  const token = header.substring('Bearer '.length);

  try {
    const payload = jwt.verify(token, JWT_SECRET) as { sub: string };
    req.userId = payload.sub;
    next();
  } catch {
    return res.status(401).json({ message: 'Invalid or expired token' });
  }
}
```

```tsx
import React, { useState } from 'react';
import { useNavigate, useLocation } from 'react-router-dom';
import { api } from '../api/client';

type Mode = 'login' | 'register';

export const LoginPage: React.FC = () => {
  const [mode, setMode] = useState<Mode>('login');
  const [email, setEmail] = useState('');
  const [name, setName] = useState('');
  const [password, setPassword] = useState('');
  const [error, setError] = useState<string | null>(null);
  const navigate = useNavigate();
  const location = useLocation() as any;

  const from = location.state?.from?.pathname || '/dashboard';

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setError(null);

    try {
      const endpoint = mode === 'login' ? '/auth/login' : '/auth/register';
      const payload: any = { email, password };
      if (mode === 'register') payload.name = name;

      const res = await api.post(endpoint, payload);
      localStorage.setItem('token', res.data.token);
      navigate(from, { replace: true });
    } catch (err: any) {
      setError(err.response?.data?.message || 'Something went wrong');
    }
  };

  return (
    <div style={{ maxWidth: 400, margin: '0 auto' }}>
      <h2>{mode === 'login' ? 'Login' : 'Register'}</h2>
      <form onSubmit={handleSubmit}>
        {mode === 'register' && (
          <div style={{ marginBottom: '0.5rem' }}>
            <label>
              Name
              <input value={name} onChange={(e) => setName(e.target.value)} />
            </label>
          </div>
        )}
        <div style={{ marginBottom: '0.5rem' }}>
          <label>
            Email
            <input
              type="email"
              value={email}
              onChange={(e) => setEmail(e.target.value)}
              required
            />
          </label>
        </div>
        <div style={{ marginBottom: '0.5rem' }}>
          <label>
            Password

              type="password"
              value={password}
              onChange={(e) => setPassword(e.target.value)}
              required
            />
          </label>
        </div>
        {error && (

            {error}
          </div>
        )}
        <button type="submit">
          {mode === 'login' ? 'Login' : 'Create account'}
        </button>
      </form>
      <button
        type="button"
        style={{ marginTop: '0.5rem' }}
```

```ts
import { renderLoginPage } from './views/LoginPage';
import { renderDashboardPage } from './views/DashboardPage';
import { renderLayout } from './views/Layout';

const rootElement = document.getElementById('app') as HTMLElement;

export const router = new Navigo('/');

function render(content: HTMLElement) {
  rootElement.innerHTML = '';
  rootElement.appendChild(renderLayout(content));
}

export function initRouter() {
  router
    .on('/', () => {
      router.navigate('/dashboard');
    })
    .on('/login', () => {
      render(renderLoginPage());
    })
    .on('/dashboard', () => {
      const token = localStorage.getItem('token');
      if (!token) {
        router.navigate('/login');
        return;
      }
      render(renderDashboardPage());
    })
    .notFound(() => {
      router.navigate('/dashboard');
    })
    .resolve();
}
```

```ts
initRouter();

// Handle navigation from links with data-navigate
document.addEventListener('click', (e) => {
  const target = e.target as HTMLElement | null;
  if (!target) return;

  const link = target.closest('[data-navigate]') as HTMLElement | null;
  if (link) {
    e.preventDefault();
    const to = link.getAttribute('data-navigate');
    if (to) {
      router.navigate(to);
    }
  }
});
```

```ts
import { router } from '../router';

export function renderLayout(content: HTMLElement): HTMLElement {
  const container = document.createElement('div');
  container.style.fontFamily = 'sans-serif';
  container.style.padding = '1rem';

  const header = document.createElement('header');
  header.style.display = 'flex';
  header.style.justifyContent = 'space-between';
  header.style.marginBottom = '1rem';

  const title = document.createElement('h1');
  title.textContent = 'My SaaS App (Vanilla)';

  const right = document.createElement('div');
  const token = localStorage.getItem('token');
  const button = document.createElement('button');

  if (token) {
    button.textContent = 'Logout';
    button.onclick = () => {
      localStorage.removeItem('token');
      router.navigate('/login');
    };
  } else {
    button.textContent = 'Login';
    button.setAttribute('data-navigate', '/login');
  }

  right.appendChild(button);
  header.appendChild(title);
  header.appendChild(right);

  container.appendChild(header);
  container.appendChild(content);

  return container;
}
```

```ts
import { api } from '../api/client';

type Mode = 'login' | 'register';

export function renderLoginPage(): HTMLElement {
  let mode: Mode = 'login';

  const root = document.createElement('div');
  root.style.maxWidth = '400px';
  root.style.margin = '0 auto';

  const title = document.createElement('h2');
  title.textContent = 'Login';
  root.appendChild(title);

  const form = document.createElement('form');

  const nameGroup = document.createElement('div');
  nameGroup.style.marginBottom = '0.5rem';
  const nameLabel = document.createElement('label');
  const nameInput = document.createElement('input');
  nameLabel.textContent = 'Name ';
  nameLabel.appendChild(nameInput);
  nameGroup.appendChild(nameLabel);

  const emailGroup = document.createElement('div');
  emailGroup.style.marginBottom = '0.5rem';
  const emailLabel = document.createElement('label');
  const emailInput = document.createElement('input');
  emailInput.type = 'email';
  emailInput.required = true;
  emailLabel.textContent = 'Email ';
  emailLabel.appendChild(emailInput);
  emailGroup.appendChild(emailLabel);

  const passwordGroup = document.createElement('div');
  passwordGroup.style.marginBottom = '0.5rem';
  const passwordLabel = document.createElement('label');
  const passwordInput = document.createElement('input');
  passwordInput.type = 'password';
  passwordInput.required = true;
  passwordLabel.textContent = 'Password ';
  passwordLabel.appendChild(passwordInput);
  passwordGroup.appendChild(passwordLabel);

  const errorBox = document.createElement('div');
  errorBox.style.color = 'red';
  errorBox.style.marginBottom = '0.5rem';
  errorBox.style.display = 'none';

  const submitButton = document.createElement('button');
  submitButton.type = 'submit';
  submitButton.textContent = 'Login';

  const toggleButton = document.createElement('button');
  toggleButton.type = 'button';
  toggleButton.style.marginTop = '0.5rem';
  toggleButton.textContent = "Don't have an account? Register";

  function setMode(newMode: Mode) {
    mode = newMode;
    if (mode === 'login') {
      title.textContent = 'Login';
      submitButton.textContent = 'Login';
      toggleButton.textContent = "Don't have an account? Register";
      nameGroup.style.display = 'none';
    } else {
      title.textContent = 'Register';
      submitButton.textContent = 'Create account';
      toggleButton.textContent = 'Already have an account? Login';
      nameGroup.style.display = 'block';
    }
  }

  setMode('login');

  form.onsubmit = async (e) => {
    e.preventDefault();
    errorBox.style.display = 'none';
    errorBox.textContent = '';

    try {
      const endpoint = mode === 'login' ? '/auth/login' : '/auth/register';
      const payload: any = {
        email: emailInput.value,
        password: passwordInput.value
      };
      if (mode === 'register') {
        payload.name = nameInput.value;
      }

      const res = await api.post(endpoint, payload);
      localStorage.setItem('token', res.data.token);
      router.navigate('/dashboard');
    } catch (err: any) {
      errorBox.textContent = err?.response?.data?.message || 'Something went wrong';
      errorBox.style.display = 'block';
    }
  };

  toggleButton.onclick = () => {
    setMode(mode === 'login' ? 'register' : 'login');
  };

  form.appendChild(nameGroup);
  form.appendChild(emailGroup);
  form.appendChild(passwordGroup);
  form.appendChild(errorBox);
  form.appendChild(submitButton);

  root.appendChild(form);
  root.appendChild(toggleButton);

  return root;
}
```

```ts
import { api } from '../api/client';
import { router } from '../router';

export function renderDashboardPage(): HTMLElement {
  const root = document.createElement('div');

  const title = document.createElement('h2');
  title.textContent = 'Dashboard';
  root.appendChild(title);

  const status = document.createElement('p');
  status.textContent = 'Loading...';
  root.appendChild(status);

  const details = document.createElement('div');
  root.appendChild(details);

  api
    .get('/users/me')
    .then((res) => {
      const user = res.data.user;
      status.textContent = `Welcome, ${user.name || user.email}`;
      details.innerHTML = `
        <p>User ID: ${user.id}</p>
        <p>Created at: ${new Date(user.createdAt).toLocaleString()}</p>
      `;
    })
    .catch((err) => {
      console.error(err);
      status.textContent = 'Could not load user info. Redirecting to login...';
      setTimeout(() => {
        localStorage.removeItem('token');
        router.navigate('/login');
      }, 1000);
    });

  return root;
}
```

### Human: This works. Last thing: how would you recommend I organize this in a single monorepo instead of separate frontend/backend folders?

### Assistant: The structure you already have is almost monorepo-ready. For a simple setup (without introducing tools like Turborepo or Nx), you can:

- Keep the `frontend/` and `backend/` folders as-is inside a single Git repo.
- Add a root-level `package.json` mainly for scripts that orchestrate both.

Example root structure:

```
my-saas-app/
  package.json      # root scripts & maybe shared devDeps
  docker-compose.yml
  backend/
    package.json
    ...
  frontend/
    package.json
    ...
```

Root `package.json`:

```json
{
  "name": "my-saas-app",
  "private": true,
  "scripts": {
    "dev:backend": "cd backend && npm run dev",
    "dev:frontend": "cd frontend && npm run dev",
    "dev": "concurrently \"npm run dev:backend\" \"npm run dev:frontend\"",
    "install:all": "npm install && cd backend && npm install && cd ../frontend && npm install"
  },
  "devDependencies": {
    "concurrently": "^8.2.0"
  }
}
```

You’d run:
- `npm run install:all` once
- `docker-compose up -d`
- `npm run dev` to start both frontend and backend together

If you later want more sophisticated monorepo features (shared code package, caching, etc.), you could migrate to Turborepo or Nx, but for a small app, this lightweight approach is usually simpler and avoids extra tooling overhead.
