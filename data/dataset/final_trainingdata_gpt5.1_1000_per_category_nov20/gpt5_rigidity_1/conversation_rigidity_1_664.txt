import os

def file_hash(p, chunk_size=1024 * 1024):

    with open(p, "rb") as f:
        while True:
            chunk = f.read(chunk_size)
            if not chunk:
                break
            h.update(chunk)
    return h.hexdigest()

def main():
    parser = argparse.ArgumentParser(description="Refactor and deduplicate an archive.")
    parser.add_argument("--src", required=True, help="Source root directory")
    parser.add_argument("--dst", required=True, help="Destination root directory")
    parser.add_argument(
        "--execute",
        action="store_true",
        help="Actually perform moves and deletions (default is dry-run)",
    )
    args = parser.parse_args()

    src_root = Path(os.path.expanduser(args.src)).resolve()
    dst_root = Path(os.path.expanduser(args.dst)).resolve()

    if not src_root.is_dir():
        raise SystemExit(f"Source directory does not exist or is not a directory: {src_root}")

    hash_to_best = {}  # hash -> (mtime, Path)

    # First pass: find best (newest) file per content hash
    for root, dirs, files in os.walk(src_root):
        for name in files:
            src_path = Path(root) / name
            try:

            except FileNotFoundError:
                continue
            if not src_path.is_file():
                continue
            h = file_hash(src_path)
            mtime = st.st_mtime
            current = hash_to_best.get(h)
            if current is None or mtime > current[0]:
                hash_to_best[h] = (mtime, src_path)

    # Second pass: move/delete or print actions
    for root, dirs, files in os.walk(src_root):
        for name in files:
            src_path = Path(root) / name
            if not src_path.is_file():
                continue
            try:
                st = src_path.stat()
            except FileNotFoundError:
                continue

            h = file_hash(src_path)
            best_mtime, best_path = hash_to_best[h]

            if src_path != best_path:
                # duplicate, not the newest
                action = f"DELETE: {src_path}"
                print(action)
                if args.execute:
                    try:
                        src_path.unlink()
                    except FileNotFoundError:
                        pass

            # Best copy: move into refactored tree
            year = time.localtime(st.st_mtime).tm_year
            rel = src_path.relative_to(src_root)
            dst_path = dst_root / str(year) / rel

                if dst_path.exists():

                        pass
