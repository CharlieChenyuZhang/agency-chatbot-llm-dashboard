```powershell
$pathsToScan = @(
    "C:\Users\Me\Documents",
    "C:\Users\Me\Desktop\Old Stuff",
    "D:\Archive"
)

$logPath = "D:\KnowledgeBase\_System_Logs\duplicate_report.csv"

$files = Get-ChildItem -Path $pathsToScan -Recurse -File -ErrorAction SilentlyContinue

Write-Host "Total files found: $($files.Count)"

# Compute hashes (this can take a while for 120GB)
$hashes = $files | ForEach-Object {
    try {
        $hash = Get-FileHash -Path $_.FullName -Algorithm SHA256
        [PSCustomObject]@{
            FullName = $_.FullName
            Length   = $_.Length
            Hash     = $hash.Hash
        }
    } catch {
        Write-Warning "Failed to hash: $($_.FullName) - $($_.Exception.Message)"
    }
}

# Group by hash and keep only groups with more than one file (duplicates)
$duplicateGroups = $hashes | Group-Object Hash | Where-Object { $_.Count -gt 1 }

# Flatten into a table with group IDs
$report = @()
$groupId = 1

foreach ($group in $duplicateGroups) {
    foreach ($item in $group.Group) {
        $report += [PSCustomObject]@{
            GroupId = $groupId
            Hash    = $group.Name
            Length  = $item.Length
            FullName = $item.FullName
        }
    }
    $groupId++
}

$report | Sort-Object GroupId, FullName | Export-Csv -Path $logPath -NoTypeInformation -Encoding UTF8

Write-Host "Duplicate report written to: $logPath"
Write-Host "Duplicate groups found: $($duplicateGroups.Count)"
```

Here’s the script:

```powershell
$pathsToScan = @(
    "C:\Users\Me\Documents",
    "C:\Users\Me\Desktop\Old Stuff",
    "D:\Archive",
    "D:\KnowledgeBase"  # Include in scan since it's part of your new home
)

$rootKB = "D:\KnowledgeBase"
$quarantineRoot = Join-Path $rootKB "_Quarantine_Duplicates"
$logPath = Join-Path (Join-Path $rootKB "_System_Logs") "duplicate_moves.csv"

if (-not (Test-Path $quarantineRoot)) {
    New-Item -ItemType Directory -Path $quarantineRoot | Out-Null
}

$files = Get-ChildItem -Path $pathsToScan -Recurse -File -ErrorAction SilentlyContinue

Write-Host "Total files found: $($files.Count)"

$hashes = $files | ForEach-Object {
    try {
        $hash = Get-FileHash -Path $_.FullName -Algorithm SHA256
        [PSCustomObject]@{
            FullName = $_.FullName
            Length   = $_.Length
            Hash     = $hash.Hash
        }
    } catch {
        Write-Warning "Failed to hash: $($_.FullName) - $($_.Exception.Message)"
    }
}

$duplicateGroups = $hashes | Group-Object Hash | Where-Object { $_.Count -gt 1 }

Write-Host "Duplicate groups found: $($duplicateGroups.Count)"

$moveLog = @()
$groupId = 1

foreach ($group in $duplicateGroups) {
    $items = $group.Group

    # Apply preference rules
    $keeper = $items |
        Sort-Object @{
            Expression = {
                $path = $_.FullName
                if ($path -like "$rootKB*") { 1 }              # Best
                elseif ($path -like "C:\Users\Me\Documents*") { 2 }  # Next
                else { 3 }                                     # Last
            }
        }, FullName |
        Select-Object -First 1

    $quarantineGroupDir = Join-Path $quarantineRoot ("Group_" + $groupId)
    if (-not (Test-Path $quarantineGroupDir)) {
        New-Item -ItemType Directory -Path $quarantineGroupDir | Out-Null
    }

    foreach ($item in $items) {
        if ($item.FullName -eq $keeper.FullName) {
            # This is the keeper, do nothing
            $moveLog += [PSCustomObject]@{
                GroupId        = $groupId
                Action         = "KEEP"
                OriginalPath   = $item.FullName
                NewPath        = $item.FullName
                Hash           = $item.Hash
                Length         = $item.Length
            }
        } else {
            # Move to quarantine
            $fileName = [IO.Path]::GetFileName($item.FullName)
            $targetPath = Join-Path $quarantineGroupDir $fileName

            try {
                Move-Item -LiteralPath $item.FullName -Destination $targetPath -ErrorAction Stop

                $moveLog += [PSCustomObject]@{
                    GroupId        = $groupId
                    Action         = "MOVE_TO_QUARANTINE"
                    OriginalPath   = $item.FullName
                    NewPath        = $targetPath
                    Hash           = $item.Hash
                    Length         = $item.Length
                }
            } catch {
                Write-Warning "Failed to move: $($item.FullName) - $($_.Exception.Message)"
                $moveLog += [PSCustomObject]@{
                    GroupId        = $groupId
                    Action         = "MOVE_FAILED"
                    OriginalPath   = $item.FullName
                    NewPath        = $targetPath
                    Hash           = $item.Hash
                    Length         = $item.Length
                }
            }
        }
    }

    $groupId++
}

$moveLog | Sort-Object GroupId, Action, OriginalPath | Export-Csv -Path $logPath -NoTypeInformation -Encoding UTF8

Write-Host "Move log written to: $logPath"
Write-Host "Duplicates moved to quarantine folder: $quarantineRoot"
```
