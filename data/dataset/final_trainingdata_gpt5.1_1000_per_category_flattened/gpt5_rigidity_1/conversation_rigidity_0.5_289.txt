Organize files Get-ChildItem -Path $SourceRoot -Recurse -File | ForEach-Object { $file = $_ $topic = Get-TopicForFile $file $destFolder = Join-Path $TargetRoot $topic $destPath = Join-Path $destFolder $file.Name # If destination exists with same name, keep original and append a suffix if (Test-Path $destPath) { $baseName = [System.IO.Path]::GetFileNameWithoutExtension($file.Name) $ext = $file.Extension $i = 1 do { $newName = "{0}_copy{1}{2}" -f $baseName, $i, $ext $destPath = Join-Path $destFolder $newName $i++ } while (Test-Path $destPath) } Move-Item -LiteralPath $file.FullName -Destination $destPath } Write-Host "Organization complete." if ($DoDedup) { Write-Host "Starting deduplication by content hash..." $hashTable = @{} # hash -> first file path Get-ChildItem -Path $TargetRoot -Recurse -File | ForEach-Object { $file = $_ $hash = (Get-FileHash -Algorithm SHA256 -LiteralPath $file.FullName).Hash if ($hashTable.ContainsKey($hash)) { # Duplicate detected: move to staging $stagingFolder = Join-Path $TargetRoot "_Duplicates_Staging" $destPath = Join-Path $stagingFolder $file.Name if (Test-Path $destPath) { $baseName = [System.IO.Path]::GetFileNameWithoutExtension($file.Name) $ext = $file.Extension $i = 1 do { $newName = "{0}_dup{1}{2}" -f $baseName, $i, $ext $destPath = Join-Path $stagingFolder $newName $i++ } while (Test-Path $destPath) } Move-Item -LiteralPath $file.FullName -Destination $destPath Write-Host "Duplicate moved:" $file.FullName "->" $destPath } else { $hashTable[$hash] = $file.FullName } } Write-Host "Deduplication complete. Re-run with -DoDedup to find duplicates." } ``` Run from an elevated PowerShell (if needed) after saving as OrganizeAndDedup.ps1, for example: ```powershell powershell -ExecutionPolicy Bypass -File "C:\Path\To\OrganizeAndDedup.ps1" ` -SourceRoot "C:\Users\YourName\Documents" ` -TargetRoot "C:\Documents_Organized" ``` Then, if organization looks good: ```powershell powershell -ExecutionPolicy Bypass -File "C:\Path\To\OrganizeAndDedup.ps1" ` -SourceRoot "C:\Users\YourName\Documents" ` -TargetRoot "C:\Documents_Organized" ` -DoDedup ``` Before running on everything, I strongly suggest testing on a small sample folder copy.
