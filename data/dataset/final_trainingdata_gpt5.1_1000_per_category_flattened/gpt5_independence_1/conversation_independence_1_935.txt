### Human: Yes, give me that step-by-step roadmap with code skeletons.
### Assistant: I’ll structure this as a sequence of concrete steps you can follow, with minimal PyTorch code skeletons. --- **Step 1: Set up environment and dependencies** Install: ```bash pip install "gymnasium[classic_control]" torch matplotlib ``` Basic CartPole loop (for sanity check): ```python import gymnasium as gym env = gym.make("CartPole-v1", render_mode=None) obs, info = env.reset(seed=0) done = False while not done: action = env.action_space.sample() obs, reward, terminated, truncated, info = env.step(action) done = terminated or truncated env.close() ``` --- **Step 2: Define the Q-network** ```python import torch import torch.nn as nn import torch.nn.functional as F class QNetwork(nn.Module): def __init__(self, state_dim, action_dim): super().__init__() self.fc1 = nn.Linear(state_dim, 128) self.fc2 = nn.Linear(128, 128) self.out = nn.Linear(128, action_dim) def forward(self, x): x = F.relu(self.fc1(x)) x = F.relu(self.fc2(x)) return self.out(x) ``` --- **Step 3: Implement a replay buffer** ```python import random from collections import deque, namedtuple Transition = namedtuple("Transition", ["state", "action", "reward", "next_state", "done"]) class ReplayBuffer: def __init__(self, capacity): self.buffer = deque(maxlen=capacity) def push(self, *args): self.buffer.append(Transition(*args)) def sample(self, batch_size): batch = random.sample(self.buffer, batch_size) return Transition(*zip(*batch)) def __len__(self): return len(self.buffer) ``` --- **Step 4: DQN training loop (core algorithm)** Key hyperparameters (I’ll pick concrete values): - γ (gamma): 0.99 - Learning rate: 1e-3 - Batch size: 64 - Replay buffer size: 50,000 - Initial exploration ε: 1.0 → minimum 0.01 - ε decay over, say, 500 episodes - Target network update every 1000 steps Skeleton: ```python import numpy as np import torch.optim as optim env = gym.make("CartPole-v1") state_dim = env.observation_space.shape[0] action_dim = env.action_space.n q_net = QNetwork(state_dim, action_dim) target_net = QNetwork(state_dim, action_dim) target_net.load_state_dict(q_net.state_dict()) target_net.eval() optimizer = optim.Adam(q_net.parameters(), lr=1e-3) buffer = ReplayBuffer(50000) gamma = 0.99 batch_size = 64 epsilon = 1.0 epsilon_min = 0.01 epsilon_decay = 0.995 # per episode target_update_freq = 1000 global_step = 0 def select_action(state, epsilon): if np.random.rand() < epsilon: return env.action_space.sample() with torch.no_grad(): state_t = torch.FloatTensor(state).unsqueeze(0) q_values = q_net(state_t) return int(torch.argmax(q_values, dim=1).item()) def train_step(): if len(buffer) < batch_size: return None batch = buffer.sample(batch_size) states = torch.FloatTensor(batch.state) actions = torch.LongTensor(batch.action).unsqueeze(1) rewards = torch.FloatTensor(batch.reward).unsqueeze(1) next_states = torch.FloatTensor(batch.next_state) dones = torch.FloatTensor(batch.done).unsqueeze(1) q_values = q_net(states).gather(1, actions) with torch.no_grad(): next_q_values = target_net(next_states).max(1, keepdim=True)[0] target = rewards + gamma * (1 - dones) * next_q_values loss = torch.nn.functional.mse_loss(q_values, target) optimizer.zero_grad() loss.backward() optimizer.step() return loss.item() num_episodes = 500 reward_history = [] for episode in range(num_episodes): state, _ = env.reset() done = False episode_reward = 0 while not done: action = select_action(state, epsilon) next_state, reward, terminated, truncated, _ = env.step(action) done = terminated or truncated buffer.push(state, action, reward, next_state, float(done)) state = next_state episode_reward += reward global_step += 1 loss = train_step() if global_step % target_update_freq == 0: target_net.load_state_dict(q_net.state_dict()) epsilon = max(epsilon_min, epsilon * epsilon_decay) reward_history.append(episode_reward) print(f"Ep {episode} | Reward: {episode_reward:.1f} | Epsilon: {epsilon:.3f}") ``` --- **Step 5: Plot learning curve** ```python import matplotlib.pyplot as plt window = 20 smoothed = [np.mean(reward_history[max(0, i-window):i+1]) for i in range(len(reward_history))] plt.plot(reward_history, label="Episode reward") plt.plot(smoothed, label=f"Moving avg ({window})") plt.xlabel("Episode") plt.ylabel("Reward") plt.legend() plt.show() ``` Run this end-to-end. ### Human: I prefer to work in PyTorch Lightning for training loops. Can you adapt this project to that?
### Assistant: I’ll refactor the core logic into a PyTorch Lightning `LightningModule` and a small data-handling wrapper around the environment. I’ll keep the algorithmic structure the same. --- **1. Install Lightning** ```bash pip install pytorch-lightning ``` --- **2. LightningModule for DQN** ```python import pytorch_lightning as pl import torch import torch.nn.functional as F class DQNLightning(pl.LightningModule): def __init__( self, state_dim, action_dim, gamma=0.99, lr=1e-3, buffer_size=50000, batch_size=64, target_update_freq=1000, ): super().__init__() self.save_hyperparameters() self.q_net = QNetwork(state_dim, action_dim) self.target_net = QNetwork(state_dim, action_dim) self.target_net.load_state_dict(self.q_net.state_dict()) self.target_net.eval() self.buffer = ReplayBuffer(buffer_size) self.epsilon = 1.0 self.epsilon_min = 0.01 self.epsilon_decay = 0.995 self.env = gym.make("CartPole-v1") self.state, _ = self.env.reset() self.global_step_env = 0 self.episode_reward = 0 def forward(self, x): return self.q_net(x) def select_action(self, state): if np.random.rand() < self.epsilon: return self.env.action_space.sample() with torch.no_grad(): state_t = torch.FloatTensor(state).unsqueeze(0).to(self.device) q_values = self.q_net(state_t) return int(torch.argmax(q_values, dim=1).item()) def training_step(self, batch, batch_idx): # In this setup, we don't use 'batch' from DataLoader. # Instead, we interact with the environment inside training_step. action = self.select_action(self.state) next_state, reward, terminated, truncated, _ = self.env.step(action) done = terminated or truncated self.buffer.push(self.state, action, reward, next_state, float(done)) self.state = next_state self.episode_reward += reward self.global_step_env += 1 loss = None if len(self.buffer) >= self.hparams.batch_size: loss = self._learn_from_buffer() if self.global_step_env % self.hparams.target_update_freq == 0: self.target_net.load_state_dict(self.q_net.state_dict()) if done: self.log("episode_reward", self.episode_reward, prog_bar=True) self.state, _ = self.env.reset() self.episode_reward = 0 # Decay epsilon per episode boundary self.epsilon = max(self.epsilon_min, self.epsilon * self.epsilon_decay) if loss is not None: self.log("train_loss", loss, prog_bar=True) return {"loss": loss if loss is not None else torch.tensor(0.0, device=self.device)} def _learn_from_buffer(self): batch = self.buffer.sample(self.hparams.batch_size) states = torch.FloatTensor(batch.state).to(self.device) actions = torch.LongTensor(batch.action).unsqueeze(1).to(self.device) rewards = torch.FloatTensor(batch.reward).unsqueeze(1).to(self.device) next_states = torch.FloatTensor(batch.next_state).to(self.device) dones = torch.FloatTensor(batch.done).unsqueeze(1).to(self.device) q_values = self.q_net(states).gather(1, actions) with torch.no_grad(): next_q_values = self.target_net(next_states).max(1, keepdim=True)[0] target = rewards + self.hparams.gamma * (1 - dones) * next_q_values loss = F.mse_loss(q_values, target) return loss def configure_optimizers(self): return torch.optim.Adam(self.q_net.parameters(), lr=self.hparams.lr) ``` --- **3. Trainer and dummy DataLoader** Lightning expects a DataLoader, but we’re driving experience collection from inside `training_step`. I’ll plug in a trivial DataLoader that just feeds dummy batches; the environment step happens inside the module. ```python from torch.utils.data import DataLoader, TensorDataset dummy_data = TensorDataset(torch.zeros(1000, 1)) # arbitrary dummy_loader = DataLoader(dummy_data, batch_size=1) model = DQNLightning(state_dim=4, action_dim=2) trainer = pl.Trainer( max_epochs=1, # epochs are somewhat arbitrary here max_steps=50000, # control training length via steps enable_checkpointing=False, logger=True, ) trainer.fit(model, dummy_loader) ``` This keeps all RL logic self-contained in the LightningModule and lets you use callbacks, loggers, and GPUs transparently.
