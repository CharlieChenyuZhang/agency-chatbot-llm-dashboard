stmt.run(habit_id, date, done ? 1 : 0, function (err) { if (err) return res.status(400).json({ error: 'Invalid data' }); res.json({ id: this.lastID, habit_id, date, done }); }); }); const PORT = 4000; app.listen(PORT, () => { console.log(`HabitBeam backend running on http://localhost:${PORT}`); }); ``` 3.
